     1→     1→"""Deterministic, explainable rule evaluation.
     2→     2→
     3→     3→Semantics (documented in README, /api/v1/rule-types and /api/v1/meta):
     4→     4→
     5→     5→- ``price_level`` — ``above``/``below``: the **latest candle** in the
     6→     6→  lookback window satisfies the comparison; ``crosses_above``/
     7→     7→  ``crosses_below``: the latest candle completes the cross.
     8→     8→- ``candle_property`` — derived property of the latest candle vs a
     9→     9→  threshold.
    10→    10→- ``candle_pattern`` — classic pattern on the latest 1-2 candles.
    11→    11→- ``swing_point`` — a swing of the given strength CONFIRMED inside the
    12→    12→  window (confirmed = ``strength`` subsequent candles exist inside the
    13→    13→  supplied data; nothing from the future is assumed).
    14→    14→- ``structure_pattern`` — the last N confirmed swings are strictly
    15→    15→  rising/falling (HH / HL / LH / LL).
    16→    16→- ``market_structure_break`` — latest close beyond the most recent
    17→    17→  confirmed opposite swing (generic, precisely-defined BOS).
    18→    18→- ``moving_average`` — SMA/EMA vs a threshold or a price field, with
    19→    19→  fresh-cross semantics for ``crosses_*``.
    20→    20→- ``price_zone`` — interaction with a user-defined [low, high] zone.
    21→    21→- Group logic: ``AND`` = every required child satisfied (unsatisfied
    22→    22→  **optional** children do not fail the group but are reported);
    23→    23→  ``OR`` = at least one child satisfied.
    24→    24→- Overall status:
    25→    25→    * ``invalidated``            — the invalidation group is satisfied
    26→    26→    * ``conditions_met``         — core satisfied AND confirmation (if any) satisfied
    27→    27→    * ``conditions_missing``     — core or confirmation unmet, no invalidation
    28→    28→    * ``insufficient_data``      — at least one referenced timeframe has no usable data
    29→    29→
    30→    30→**No look-ahead bias**: every evaluator is a pure function of the candles
    31→    31→supplied at evaluation time. A condition never uses information that was
    32→    32→unavailable when the last supplied candle closed (swing confirmation
    33→    33→explicitly requires ``strength`` candles AFTER the swing inside the data).
    34→    34→
    35→    35→The evaluator never fetches or invents market data.
    36→    36→"""

   142→   142→def _crossover_satisfied(
   143→   143→    operator: str,
   144→   144→    series_a: list[float],
   145→   145→    series_b: list[float],
   146→   146→) -> tuple[bool, list[int]]:
   147→   147→    """Shared fresh-cross semantics. Returns (satisfied, cross_indices)."""
   148→   148→
   149→   149→    crosses: list[int] = []
   150→   150→    for i in range(1, len(series_a)):
   151→   151→        prev, cur = series_a[i - 1], series_a[i]
   152→   152→        if operator == "crosses_above":
   153→   153→            hit = prev <= series_b[i - 1] and cur > series_b[i]
   154→   154→        else:
   155→   155→            hit = prev >= series_b[i - 1] and cur < series_b[i]
   156→   156→        if hit:
   157→   157→            crosses.append(i)
   158→   158→    satisfied = bool(crosses) and crosses[-1] == len(series_a) - 1
   159→   159→    return satisfied, crosses

   167→   167→def evaluate_price_level_condition(
   168→   168→    condition: Condition, candles: list[dict[str, Any]]
   169→   169→) -> NodeResult:
   170→   170→    """Evaluate a ``price_level`` condition with full evidence."""
   171→   171→
   172→   172→    tf = condition.timeframe
   173→   173→    lookback = condition.lookback
   174→   174→    window = _window(candles, lookback)
   175→   175→    field = str(condition.params["field"])
   176→   176→    operator = str(condition.params["operator"])
   177→   177→    threshold = float(condition.params["value"])
   178→   178→    needs_two = operator in ("crosses_above", "crosses_below")
   179→   179→    series = _series_from(window, field)
   180→   180→
   181→   181→    description = (
   182→   182→        f"{field} {operator.replace('_', ' ')} "
   183→   183→        f"{threshold:g} on {tf} (last {len(window)} candle(s))"
   184→   184→    )
   185→   185→    evidence: dict[str, Any] = {
   186→   186→        "condition_type": condition.type,
   187→   187→        "timeframe": tf,
   188→   188→        "field": field,
   189→   189→        "operator": operator,
   190→   190→        "threshold": threshold,
   191→   191→        "lookback": lookback,
   192→   192→        "candles_evaluated": len(window),
   193→   193→        "candles_available": len(candles),
   194→   194→    }
   195→   195→
   196→   196→    if not window or (needs_two and len(window) < 2):
   197→   197→        return _unsatisfied(
   198→   198→            condition,
   199→   199→            description,
   200→   200→            evidence,
   201→   201→            reason=(
   202→   202→                "Cross conditions need at least 2 candles in the window."
   203→   203→                if needs_two
   204→   204→                else "No usable candles for this timeframe."
   205→   205→            ),
   206→   206→        )
   207→   207→
   208→   208→    if operator in ("above", "below"):
   209→   209→        comparisons = [
   210→   210→            (v > threshold) if operator == "above" else (v < threshold) for v in series
   211→   211→        ]
   212→   212→        satisfied = comparisons[-1]
   213→   213→        matched = sum(comparisons)
   214→   214→        first_hit = next(
   215→   215→            (window[i]["time"] for i, hit in enumerate(comparisons) if hit), None
   216→   216→        )
   217→   217→        last_hit = next(
   218→   218→            (window[i]["time"] for i in range(len(comparisons) - 1, -1, -1) if comparisons[i]),
   219→   219→            None,
   220→   220→        )
   221→   221→        evidence.update(
   222→   222→            {
   223→   223→                "current_value": series[-1],
   224→   224→                "matched_count": matched,
   225→   225→                "first_satisfied_time": first_hit,
   226→   226→                "last_satisfied_time": last_hit,
   227→   227→                "criterion": "latest candle must satisfy the comparison",
   228→   228→            }
   229→   229→        )
   230→   230→    else:  # crosses_above / crosses_below
   231→   231→        crosses: list[dict[str, Any]] = []
   232→   232→        for i in range(1, len(series)):
   233→   233→            prev, cur = series[i - 1], series[i]
   234→   234→            if operator == "crosses_above":
   235→   235→                hit = prev <= threshold < cur
   236→   236→            else:
   237→   237→                hit = prev >= threshold > cur
   238→   238→            if hit:
   239→   239→                crosses.append({"time": window[i]["time"], "value": cur})
   240→   240→        # Documented semantics: the FRESH cross decides — the latest candle
   241→   241→        # must be the one completing the cross.
   242→   242→        satisfied = bool(crosses) and crosses[-1]["time"] == window[-1]["time"]
   243→   243→        evidence.update(
   244→   244→            {
   245→   245→                "current_value": series[-1],
   246→   246→                "previous_value": series[-2],
   247→   247→                "crosses_in_window": crosses[-10:],  # cap for payload size
   248→   248→                "total_crosses_in_window": len(crosses),
   249→   249→                "last_cross_time": crosses[-1]["time"] if crosses else None,
   250→   250→                "criterion": "latest candle must complete the cross",
   251→   251→            }
   252→   252→        )
   253→   253→
   254→   254→    return _node(condition, description, evidence, satisfied)

   261→   261→
   262→   262→def _candle_property_value(candle: dict[str, Any], prop: str) -> float:
   263→   263→    o, h, l, c = (float(candle[k]) for k in ("open", "high", "low", "close"))
   264→   264→    if prop in ("open", "high", "low", "close"):
   265→   265→        return {"open": o, "high": h, "low": l, "close": c}[prop]
   266→   266→    if prop == "body":
   267→   267→        return abs(c - o)
   268→   268→    if prop == "range":
   269→   269→        return h - l
   270→   270→    if prop == "upper_wick":
   271→   271→        return h - max(o, c)
   272→   272→    if prop == "lower_wick":
   273→   273→        return min(o, c) - l
   274→   274→    raise ValueError(f"Unknown candle property '{prop}'")  # pragma: no cover
   275→   275→

   277→   277→def evaluate_candle_property(
   278→   278→    condition: Condition, candles: list[dict[str, Any]]
   279→   279→) -> NodeResult:
   280→   280→    tf = condition.timeframe
   281→   281→    prop = str(condition.params["property"])
   282→   282→    operator = str(condition.params["operator"])
   283→   283→    threshold = float(condition.params["value"])
   284→   284→    window = _window(candles, condition.lookback)
   285→   285→
   286→   286→    description = (
   287→   287→        f"{prop.replace('_', ' ')} {operator} {threshold:g} on {tf} "
   288→   288→        "(latest candle)"
   289→   289→    )
   290→   290→    evidence: dict[str, Any] = {
   291→   291→        "condition_type": condition.type,
   292→   292→        "timeframe": tf,
   293→   293→        "property": prop,
   294→   294→        "operator": operator,
   295→   295→        "threshold": threshold,
   296→   296→        "candles_evaluated": len(window),
   297→   297→        "candles_available": len(candles),
   298→   298→    }
   299→   299→    if not window:
   300→   300→        return _unsatisfied(
   301→   301→            condition, description, evidence, "No usable candles for this timeframe."
   302→   302→        )
   303→   303→    value = _candle_property_value(window[-1], prop)
   304→   304→    satisfied = value > threshold if operator == "above" else value < threshold
   305→   305→    evidence.update(
   306→   306→        {
   307→   307→            "current_value": value,
   308→   308→            "criterion": "property of the latest candle must satisfy the comparison",
   309→   309→        }
   310→   310→    )
   311→   311→    return _node(condition, description, evidence, satisfied)

   318→   318→
   319→   319→def evaluate_candle_pattern(
   320→   320→    condition: Condition, candles: list[dict[str, Any]]
   321→   321→) -> NodeResult:
   322→   322→    tf = condition.timeframe
   323→   323→    pattern = str(condition.params["pattern"])
   324→   324→    max_body_ratio = float(condition.params.get("max_body_ratio", 0.1))
   325→   325→    window = _window(candles, condition.lookback)
   326→   326→
   327→   327→    description = f"{pattern.replace('_', ' ')} on {tf} (latest candle)"
   328→   328→    evidence: dict[str, Any] = {
   329→   329→        "condition_type": condition.type,
   330→   330→        "timeframe": tf,
   331→   331→        "pattern": pattern,
   332→   332→        "candles_evaluated": len(window),
   333→   333→        "candles_available": len(candles),
   334→   334→    }
   335→   335→    if not window:
   336→   336→        return _unsatisfied(
   337→   337→            condition, description, evidence, "No usable candles for this timeframe."
   338→   338→        )
   339→   339→    needs_two = pattern in ("bullish_engulfing", "bearish_engulfing", "inside_bar")
   340→   340→    if needs_two and len(window) < 2:
   341→   341→        return _unsatisfied(
   342→   342→            condition,
   343→   343→            description,
   344→   344→            evidence,
   345→   345→            reason=f"Pattern '{pattern}' needs at least 2 candles.",
   346→   346→        )
   347→   347→
   348→   348→    cur, prev = window[-1], window[-2] if len(window) > 1 else None
   349→   349→
   350→   350→    def body(c: dict[str, Any]) -> float:
   351→   351→        return abs(float(c["close"]) - float(c["open"]))
   352→   352→
   353→   353→    def is_bull(c: dict[str, Any]) -> bool:
   354→   354→        return float(c["close"]) > float(c["open"])
   355→   355→
   356→   356→    def is_bear(c: dict[str, Any]) -> bool:
   357→   357→        return float(c["close"]) < float(c["open"])
   358→   358→
   359→   359→    rng = float(cur["high"]) - float(cur["low"])
   360→   360→    body_cur = body(cur)
   361→   361→    if rng <= 0:
   362→   362→        matched = pattern == "doji"  # a zero-range candle is a perfect doji
   363→   363→        evidence.update({"pattern_range": 0.0, "criterion": pattern})
   364→   364→        return _node(condition, description, evidence, bool(matched))
   365→   365→
   366→   366→    matched = False
   367→   367→    detail: dict[str, Any] = {"criterion": pattern}
   368→   368→    if pattern == "bullish_engulfing" and prev is not None:
   369→   369→        matched = (
   370→   370→            is_bear(prev)
   371→   371→            and is_bull(cur)
   372→   372→            and float(cur["open"]) <= float(prev["close"])
   373→   373→            and float(cur["close"]) >= float(prev["open"])
   374→   374→        )
   375→   375→    elif pattern == "bearish_engulfing" and prev is not None:
   376→   376→        matched = (
   377→   377→            is_bull(prev)
   378→   378→            and is_bear(cur)
   379→   379→            and float(cur["open"]) >= float(prev["close"])
   380→   380→            and float(cur["close"]) <= float(prev["open"])
   381→   381→        )
   382→   382→    elif pattern == "hammer":
   383→   383→        lower_wick = min(float(cur["open"]), float(cur["close"])) - float(cur["low"])
   384→   384→        upper_wick = float(cur["high"]) - max(float(cur["open"]), float(cur["close"]))
   385→   385→        mid = (float(cur["high"]) + float(cur["low"])) / 2
   386→   386→        matched = (
   387→   387→            lower_wick >= 2 * body_cur
   388→   388→            and upper_wick <= body_cur
   389→   389→            and float(cur["close"]) >= mid
   390→   390→        )
   391→   391→        detail.update({"lower_wick": lower_wick, "upper_wick": upper_wick})
   392→   392→    elif pattern == "shooting_star":
   393→   393→        lower_wick = min(float(cur["open"]), float(cur["close"])) - float(cur["low"])
   394→   394→        upper_wick = float(cur["high"]) - max(float(cur["open"]), float(cur["close"]))
   395→   395→        mid = (float(cur["high"]) + float(cur["low"])) / 2
   396→   396→        matched = (
   397→   397→            upper_wick >= 2 * body_cur
   398→   398→            and lower_wick <= body_cur
   399→   399→            and float(cur["close"]) <= mid
   400→   400→        )
   401→   401→        detail.update({"lower_wick": lower_wick, "upper_wick": upper_wick})
   402→   402→    elif pattern == "doji":
   403→   403→        matched = body_cur <= max_body_ratio * rng
   404→   404→        detail.update({"body": body_cur, "range": rng, "max_body_ratio": max_body_ratio})
   405→   405→    elif pattern == "inside_bar" and prev is not None:
   406→   406→        matched = float(cur["high"]) <= float(prev["high"]) and float(cur["low"]) >= float(prev["low"])
   407→   407→        if prev is not None:
   408→   408→            detail.update(
   409→   409→                {
   410→   410→                    "previous_high": float(prev["high"]),
   411→   411→                    "previous_low": float(prev["low"]),
   412→   412→                }
   413→   413→            )
   414→   414→
   415→   415→    evidence.update(detail)
   416→   416→    evidence["matched_pattern"] = matched
   417→   417→    return _node(condition, description, evidence, matched)
   418→   418→

   421→   421→# swing points (shared by three rule types)
   422→   422→# --------------------------------------------------------------------------- #

   424→   424→
   425→   425→def _confirmed_swings(
   426→   426→    window: list[dict[str, Any]], kind: str, strength: int
   427→   427→) -> list[dict[str, Any]]:
   428→   428→    """Swings confirmed strictly inside the supplied window (no look-ahead).
   429→   429→
   430→   430→    A candle at index ``i`` is a confirmed swing high when its high is
   431→   431→    strictly greater than the highs of the ``strength`` candles before AND
   432→   432→    after it, and at least ``strength`` candles exist after it inside the
   433→   433→    window. Returns a list of {"index", "time", "price"} in chronological
   434→   434→    order.
   435→   435→    """
   436→   436→
   437→   437→    swings: list[dict[str, Any]] = []
   438→   438→    n = len(window)
   439→   439→    for i in range(strength, n - strength):
   440→   440→        if kind == "high":
   441→   441→            value = float(window[i]["high"])
   442→   442→            is_swing = all(
   443→   443→                value > float(window[j]["high"])
   444→   444→                for j in range(i - strength, i + strength + 1)
   445→   445→                if j != i
   446→   446→            )
   447→   447→        else:
   448→   448→            value = float(window[i]["low"])
   449→   449→            is_swing = all(
   450→   450→                value < float(window[j]["low"])
   451→   451→                for j in range(i - strength, i + strength + 1)
   452→   452→                if j != i
   453→   453→            )
   454→   454→        if is_swing:
   455→   455→            swings.append(
   456→   456→                {"index": i, "time": window[i]["time"], "price": value}
   457→   457→            )
   458→   458→    return swings

   461→   461→def evaluate_swing_point(
   462→   462→    condition: Condition, candles: list[dict[str, Any]]
   463→   463→) -> NodeResult:
   464→   464→    tf = condition.timeframe
   465→   465→    kind = str(condition.params["kind"])
   466→   466→    strength = int(condition.params["strength"])
   467→   467→    window = _window(candles, condition.lookback)
   468→   468→
   469→   469→    description = (
   470→   470→        f"confirmed swing {kind} (strength {strength}) within last "
   471→   471→        f"{len(window)} candle(s) on {tf}"
   472→   472→    )
   473→   473→    evidence: dict[str, Any] = {
   474→   474→        "condition_type": condition.type,
   475→   475→        "timeframe": tf,
   476→   476→        "kind": kind,
   477→   477→        "strength": strength,
   478→   478→        "lookback": condition.lookback,
   479→   479→        "candles_evaluated": len(window),
   480→   480→        "candles_available": len(candles),
   481→   481→        "criterion": (
   482→   482→            "a swing high/low confirmed by `strength` subsequent candles "
   483→   483→            "inside the window (no look-ahead)"
   484→   484→        ),
   485→   485→    }
   486→   486→    minimum = 2 * strength + 1
   487→   487→    if len(window) < minimum:
   488→   488→        return _unsatisfied(
   489→   489→            condition,
   490→   490→            description,
   491→   491→            evidence,
   492→   492→            reason=(
   493→   493→                f"Detecting strength-{strength} swings needs at least "
   494→   494→                f"{minimum} candles in the window."
   495→   495→            ),
   496→   496→        )
   497→   497→    swings = _confirmed_swings(window, kind, strength)
   498→   498→    evidence["swings_found"] = len(swings)
   499→   499→    evidence["swings"] = [
   500→   500→        {"time": s["time"], "price": s["price"]} for s in swings[-5:]
   501→   501→    ]
   502→   502→    return _node(condition, description, evidence, bool(swings))
   503→   503→
   504→   504→
   505→   505→# --------------------------------------------------------------------------- #
   506→   506→# structure_pattern (HH / HL / LH / LL)
   507→   507→# --------------------------------------------------------------------------- #
   508→   508→
   509→   509→
   510→   510→def evaluate_structure_pattern(
   511→   511→    condition: Condition, candles: list[dict[str, Any]]
   512→   512→) -> NodeResult:
   513→   513→    tf = condition.timeframe
   514→   514→    pattern = str(condition.params["pattern"])
   515→   515→    count = int(condition.params["count"])
   516→   516→    strength = int(condition.params.get("strength", 2))
   517→   517→    window = _window(candles, condition.lookback)
   518→   518→
   519→   519→    kind = "high" if pattern in ("higher_highs", "lower_highs") else "low"
   520→   520→    rising = pattern in ("higher_highs", "higher_lows")
   521→   521→    description = (
   522→   522→        f"{pattern.replace('_', ' ')}: last {count} confirmed swing {kind}s "
   523→   523→        f"{'strictly rising' if rising else 'strictly falling'} on {tf}"
   524→   524→    )
   525→   525→    evidence: dict[str, Any] = {
   526→   526→        "condition_type": condition.type,
   527→   527→        "timeframe": tf,
   528→   528→        "pattern": pattern,
   529→   529→        "count": count,
   530→   530→        "strength": strength,
   531→   531→        "candles_evaluated": len(window),
   532→   532→        "candles_available": len(candles),
   533→   533→        "criterion": (
   534→   534→            f"the most recent {count} confirmed swing {kind}s must be "
   535→   535→            f"{'strictly rising' if rising else 'strictly falling'}"
   536→   536→        ),
   537→   537→    }
   538→   538→    minimum = 2 * strength + count
   539→   539→    if len(window) < minimum:
   540→   540→        return _unsatisfied(
   541→   541→            condition,
   542→   542→            description,
   543→   543→            evidence,
   544→   544→            reason=(
   545→   545→                f"Confirming {count} strength-{strength} swings needs at "
   546→   546→                f"least {minimum} candles in the window."
   547→   547→            ),
   548→   548→        )
   549→   549→    swings = _confirmed_swings(window, kind, strength)[-count:]
   550→   550→    evidence["swings_found"] = len(swings)
   551→   551→    evidence["swing_prices"] = [s["price"] for s in swings]
   552→   552→    if len(swings) < count:
   553→   553→        evidence["reason"] = (
   554→   554→            f"Only {len(swings)} confirmed swing {kind}(s) in the window; "
   555→   555→            f"{count} are required."
   556→   556→        )
   557→   557→        return _node(condition, description, evidence, False)
   558→   558→    if rising:
   559→   559→        satisfied = all(
   560→   560→            swings[i]["price"] > swings[i - 1]["price"] for i in range(1, len(swings))
   561→   561→        )
   562→   562→    else:
   563→   563→        satisfied = all(
   564→   564→            swings[i]["price"] < swings[i - 1]["price"] for i in range(1, len(swings))
   565→   565→        )
   566→   566→    return _node(condition, description, evidence, satisfied)
   567→   567→

   569→   569→# --------------------------------------------------------------------------- #
   570→   570→# market_structure_break (generic BOS)
   571→   571→# --------------------------------------------------------------------------- #
   572→   572→
   573→   573→
   574→   574→def evaluate_market_structure_break(
   575→   575→    condition: Condition, candles: list[dict[str, Any]]
   576→   576→) -> NodeResult:
   577→   577→    tf = condition.timeframe
   578→   578→    side = str(condition.params["side"])
   579→   579→    strength = int(condition.params.get("strength", 2))
   580→   580→    window = _window(candles, condition.lookback)
   581→   581→
   582→   582→    swing_kind = "high" if side == "bullish" else "low"
   583→   583→    description = (
   584→   584→        f"{side} structure break: close beyond the most recent confirmed "
   585→   585→        f"swing {swing_kind} (strength {strength}) on {tf}"
   586→   586→    )
   587→   587→    evidence: dict[str, Any] = {
   588→   588→        "condition_type": condition.type,
   589→   589→        "timeframe": tf,
   590→   590→        "side": side,
   591→   591→        "strength": strength,
   592→   592→        "candles_evaluated": len(window),
   593→   593→        "candles_available": len(candles),
   594→   594→        "criterion": (
   595→   595→            f"latest close must be {'above' if side == 'bullish' else 'below'} "
   596→   596→            f"the most recent confirmed swing {swing_kind} in the window"
   597→   597→        ),
   598→   598→    }
   599→   599→    if len(window) < 2 * strength + 2:
   600→   600→        return _unsatisfied(
   601→   601→            condition,
   602→   602→            description,
   603→   603→            evidence,
   604→   604→            reason=(
   605→   605→                f"A break evaluation needs at least {2 * strength + 2} "
   606→   606→                "candles (swing + the breaking candle)."
   607→   607→            ),
   608→   608→        )
   609→   609→
   610→   610→    # Swings confirmed strictly BEFORE the latest candle: the last
   611→   611→    # `strength` candles cannot be confirmed swings, and the breaking
   612→   612→    # candle itself is the latest one.
   613→   613→    swing_zone = window[:-1]
   614→   614→    swings = _confirmed_swings(swing_zone, swing_kind, strength)
   615→   615→    evidence["reference_swings"] = [
   616→   616→        {"time": s["time"], "price": s["price"]} for s in swings[-3:]
   617→   617→    ]
   618→   618→    if not swings:
   619→   619→        return _unsatisfied(
   620→   620→            condition,
   621→   621→            description,
   622→   622→            evidence,
   623→   623→            reason=f"No confirmed swing {swing_kind} available as the break reference.",
   624→   624→        )
   625→   625→    reference = swings[-1]
   626→   626→    close = float(window[-1]["close"])
   627→   627→    satisfied = close > reference["price"] if side == "bullish" else close < reference["price"]
   628→   628→    evidence.update(
   629→   629→        {
   630→   630→            "reference_swing_price": reference["price"],
   631→   631→            "reference_swing_time": reference["time"],
   632→   632→            "current_close": close,
   633→   633→        }
   634→   634→    )
   635→   635→    return _node(condition, description, evidence, satisfied)
   636→   636→

   638→   638→# --------------------------------------------------------------------------- #
   639→   639→# moving_average
   640→   640→# --------------------------------------------------------------------------- #
   641→   641→
   642→   642→
   643→   643→def _sma(series: list[float], period: int) -> list[float] | None:
   644→   644→    if len(series) < period:
   645→   645→        return None
   646→   646→    out = [math.nan] * len(series)
   647→   647→    running = sum(series[:period])
   648→   648→    out[period - 1] = running / period
   649→   649→    for i in range(period, len(series)):
   650→   650→        running += series[i] - series[i - period]
   651→   651→        out[i] = running / period
   652→   652→    return out
   653→   653→
   654→   654→
   655→   655→def _ema(series: list[float], period: int) -> list[float] | None:
   656→   656→    if len(series) < period:
   657→   657→        return None
   658→   658→    out = [math.nan] * len(series)
   659→   659→    alpha = 2.0 / (period + 1)
   660→   660→    prev = sum(series[:period]) / period  # SMA seed (documented)
   661→   661→    out[period - 1] = prev
   662→   662→    for i in range(period, len(series)):
   663→   663→        prev = alpha * series[i] + (1 - alpha) * prev
   664→   664→        out[i] = prev
   665→   665→    return out
   666→   666→
   667→   667→
   668→   668→def evaluate_moving_average(
   669→   669→    condition: Condition, candles: list[dict[str, Any]]
   670→   670→) -> NodeResult:
   671→   671→    tf = condition.timeframe
   672→   672→    ma_type = str(condition.params["ma_type"])
   673→   673→    period = int(condition.params["period"])
   674→   674→    source = str(condition.params.get("source", "close"))
   675→   675→    compare_to = str(condition.params["compare_to"])
   676→   676→    operator = str(condition.params["operator"])
   677→   677→    window = _window(candles, condition.lookback)
   678→   678→
   679→   679→    needs_two = operator in ("crosses_above", "crosses_below")
   680→   680→    target_desc = (
   681→   681→        f"{float(condition.params['value']):g}"
   682→   682→        if compare_to == "value" and "value" in condition.params
   683→   683→        else str(condition.params.get("price_field", "price field"))
   684→   684→    )
   685→   685→    description = (
   686→   686→        f"{ma_type.upper()}({period}) of {source} {operator.replace('_', ' ')} "
   687→   687→        f"{target_desc} on {tf} (last {len(window)} candle(s))"
   688→   688→    )
   689→   689→    evidence: dict[str, Any] = {
   690→   690→        "condition_type": condition.type,
   691→   691→        "timeframe": tf,
   692→   692→        "ma_type": ma_type,
   693→   693→        "period": period,
   694→   694→        "source": source,
   695→   695→        "compare_to": compare_to,
   696→   696→        "operator": operator,
   697→   697→        "lookback": condition.lookback,
   698→   698→        "candles_evaluated": len(window),
   699→   699→        "candles_available": len(candles),
   700→   700→    }
   701→   701→
   702→   702→    if not window or (needs_two and len(window) < 2):
   703→   703→        return _unsatisfied(
   704→   704→            condition,
   705→   705→            description,
   706→   706→            evidence,
   707→   707→            reason="Cross conditions need at least 2 candles in the window.",
   708→   708→        )
   709→   709→    series = _series_from(window, source)
   710→   710→    ma_series = (_ema if ma_type == "ema" else _sma)(series, period)
   711→   711→    if ma_series is None or math.isnan(ma_series[-1]) or (
   712→   712→        needs_two and math.isnan(ma_series[-2])
   713→   713→    ):
   714→   714→        return _unsatisfied(
   715→   715→            condition,
   716→   716→            description,
   717→   717→            evidence,
   718→   718→            reason=(
   719→   719→                f"{ma_type.upper()}({period}) needs at least {period} "
   720→   720→                "candles in the window."
   721→   721→            ),
   722→   722→        )
   723→   723→
   724→   724→    if compare_to == "value":
   725→   725→        threshold = float(condition.params["value"])
   726→   726→        target_series = [threshold] * len(series)
   727→   727→        evidence["threshold"] = threshold
   728→   728→        target_label = f"{threshold:g}"
   729→   729→    else:
   730→   730→        target_field = str(condition.params["price_field"])
   731→   731→        target_series = _series_from(window, target_field)
   732→   732→        evidence["price_field"] = target_field
   733→   733→        target_label = target_field
   734→   734→
   735→   735→    description = (
   736→   736→        f"{ma_type.upper()}({period}) of {source} "
   737→   737→        f"{operator.replace('_', ' ')} {target_label} on {tf} "
   738→   738→        f"(last {len(window)} candle(s))"
   739→   739→    )
   740→   740→
   741→   741→    if operator in ("above", "below"):
   742→   742→        last_ma = ma_series[-1]
   743→   743→        satisfied = last_ma > target_series[-1] if operator == "above" else last_ma < target_series[-1]
   744→   744→        evidence.update(
   745→   745→            {
   746→   746→                "ma_value": last_ma,
   747→   747→                "target_value": target_series[-1],
   748→   748→                "criterion": "MA on the latest candle must satisfy the comparison",
   749→   749→            }
   750→   750→        )
   751→   751→    else:
   752→   752→        # Compare the MA against the target series at aligned indices; skip
   753→   753→        # the MA warm-up (NaN) region — the first valid MA value sits at
   754→   754→        # index period-1 (the SMA/EMA seed).
   755→   755→        start = period - 1
   756→   756→        ma_valid = ma_series[start:]
   757→   757→        target_valid = target_series[start:]
   758→   758→        satisfied, crosses = _crossover_satisfied(operator, ma_valid, target_valid)
   759→   759→        evidence.update(
   760→   760→            {
   761→   761→                "ma_value": ma_series[-1],
   762→   762→                "previous_ma_value": ma_series[-2],
   763→   763→                "target_value": target_series[-1],
   764→   764→                "previous_target_value": target_series[-2],
   765→   765→                "crosses_in_window": [
   766→   766→                    {"time": window[start + i]["time"], "value": ma_valid[i]}
   767→   767→                    for i in crosses[-10:]
   768→   768→                ],
   769→   769→                "total_crosses_in_window": len(crosses),
   770→   770→                "criterion": "latest candle must complete the cross",
   771→   771→            }
   772→   772→        )
   773→   773→    return _node(condition, description, evidence, satisfied)
   774→   774→

   776→   776→# --------------------------------------------------------------------------- #
   777→   777→# price_zone
   778→   778→# --------------------------------------------------------------------------- #
   779→   779→
   780→   780→
   781→   781→def evaluate_price_zone(
   782→   782→    condition: Condition, candles: list[dict[str, Any]]
   783→   783→) -> NodeResult:
   784→   784→    tf = condition.timeframe
   785→   785→    zone_low = float(condition.params["zone_low"])
   786→   786→    zone_high = float(condition.params["zone_high"])
   787→   787→    operator = str(condition.params["operator"])
   788→   788→    window = _window(candles, condition.lookback)
   789→   789→
   790→   790→    description = (
   791→   791→        f"zone [{zone_low:g}, {zone_high:g}] {operator.replace('_', ' ')} "
   792→   792→        f"on {tf} (last {len(window)} candle(s))"
   793→   793→    )
   794→   794→    evidence: dict[str, Any] = {
   795→   795→        "condition_type": condition.type,
   796→   796→        "timeframe": tf,
   797→   797→        "zone_low": zone_low,
   798→   798→        "zone_high": zone_high,
   799→   799→        "operator": operator,
   800→   800→        "lookback": condition.lookback,
   801→   801→        "candles_evaluated": len(window),
   802→   802→        "candles_available": len(candles),
   803→   803→    }
   804→   804→    if not window:
   805→   805→        return _unsatisfied(
   806→   806→            condition, description, evidence, "No usable candles for this timeframe."
   807→   807→        )
   808→   808→
   809→   809→    def touches(c: dict[str, Any]) -> bool:
   810→   810→        return float(c["low"]) <= zone_high and float(c["high"]) >= zone_low
   811→   811→
   812→   812→    if operator == "touched":
   813→   813→        hits = [c for c in window if touches(c)]
   814→   814→        evidence.update(
   815→   815→            {
   816→   816→                "touches": len(hits),
   817→   817→                "first_touch_time": hits[0]["time"] if hits else None,
   818→   818→                "last_touch_time": hits[-1]["time"] if hits else None,
   819→   819→                "criterion": "any candle in the window overlaps the zone",
   820→   820→            }
   821→   821→        )
   822→   822→        return _node(condition, description, evidence, bool(hits))
   823→   823→
   824→   824→    latest = window[-1]
   825→   825→    close = float(latest["close"])
   826→   826→    if operator == "closed_inside":
   827→   827→        satisfied = zone_low <= close <= zone_high
   828→   828→        evidence.update(
   829→   829→            {
   830→   830→                "current_close": close,
   831→   831→                "criterion": "latest candle's close lies inside the zone (inclusive)",
   832→   832→            }
   833→   833→        )
   834→   834→        return _node(condition, description, evidence, satisfied)
   835→   835→
   836→   836→    # rejected: latest candle touched the zone intrabar but closed outside.
   837→   837→    touched = touches(latest)
   838→   838→    closed_inside = zone_low <= close <= zone_high
   839→   839→    satisfied = touched and not closed_inside
   840→   840→    evidence.update(
   841→   841→        {
   842→   842→            "current_close": close,
   843→   843→            "touched_zone": touched,
   844→   844→            "criterion": "latest candle touched the zone but closed outside it",
   845→   845→        }
   846→   846→    )
   847→   847→    return _node(condition, description, evidence, satisfied)
   848→   848→

   857→   857→    "price_level": evaluate_price_level_condition,

   859→   859→    "candle_pattern": evaluate_candle_pattern,

   867→   867→
   868→   868→def _minimum_candles(condition: Condition) -> int:
   869→   869→    """Worst-case minimum window size implied by the condition's type."""
   870→   870→
   871→   871→    p = condition.params
   872→   872→    t = condition.type
   873→   873→    if t == "moving_average":
   874→   874→        return int(p.get("period", 2)) + (2 if str(p.get("operator", "")).startswith("crosses") else 1)
   875→   875→    if t == "swing_point":
   876→   876→        return 2 * int(p.get("strength", 1)) + 1
   877→   877→    if t == "structure_pattern":
   878→   878→        return 2 * int(p.get("strength", 2)) + int(p.get("count", 2))
   879→   879→    if t == "market_structure_break":
   880→   880→        return 2 * int(p.get("strength", 2)) + 2
   881→   881→    if t == "candle_pattern":
   882→   882→        return 2
   883→   883→    return 1

   886→   886→# --------------------------------------------------------------------------- #
   887→   887→# Group + rule set evaluation
   888→   888→# --------------------------------------------------------------------------- #
   889→   889→
   890→   890→
   891→   891→def evaluate_group(
   892→   892→    group: Group, candles_by_timeframe: dict[str, list[dict[str, Any]]]
   893→   893→) -> NodeResult:
   894→   894→    children: list[NodeResult] = []
   895→   895→    for cond in group.conditions:
   896→   896→        candles = candles_by_timeframe.get(cond.timeframe, [])
   897→   897→        evaluator = CONDITION_EVALUATORS.get(cond.type)
   898→   898→        if evaluator is None:
   899→   899→            # Defensive: validation already rejects unknown/unavailable
   900→   900→            # types. If one ever reaches the evaluator, fail CLOSED with an
   901→   901→            # explicit reason instead of pretending it was checked.
   902→   902→            children.append(
   903→   903→                NodeResult(
   904→   904→                    kind="condition",
   905→   905→                    id=cond.id,
   906→   906→                    logic=None,
   907→   907→                    satisfied=False,
   908→   908→                    description=f"{cond.type} on {cond.timeframe}",
   909→   909→                    evidence={
   910→   910→                        "condition_type": cond.type,
   911→   911→                        "reason": (
   912→   912→                            "No evaluator registered for this rule type. "
   913→   913→                            "The platform does not pretend unsupported rules "
   914→   914→                            "are satisfied."
   915→   915→                        ),
   916→   916→                    },
   917→   917→                    optional=cond.optional,
   918→   918→                )
   919→   919→            )
   920→   920→            continue
   921→   921→        children.append(evaluator(cond, candles))
   922→   922→    for nested in group.groups:
   923→   923→        children.append(evaluate_group(nested, candles_by_timeframe))
   924→   924→
   925→   925→    required = [child for child in children if not child.optional]
   926→   926→    if group.logic == "AND":
   927→   927→        satisfied = bool(required) and all(child.satisfied for child in required)
   928→   928→    else:  # OR
   929→   929→        satisfied = any(child.satisfied for child in children)
   930→   930→
   931→   931→    optional_unmet = sum(1 for c in children if c.optional and not c.satisfied)
   932→   932→    suffix = (
   933→   933→        f" ({optional_unmet} optional rule(s) currently unmet)"
   934→   934→        if optional_unmet
   935→   935→        else ""
   936→   936→    )
   937→   937→    return NodeResult(
   938→   938→        kind="group",
   939→   939→        id=f"group_{group.logic.lower()}_{id(group) % 100000:05d}",
   940→   940→        logic=group.logic,
   941→   941→        satisfied=satisfied,
   942→   942→        description=(
   943→   943→            f"ALL of {len(children)} condition(s)/group(s) must hold{suffix}"
   944→   944→            if group.logic == "AND"
   945→   945→            else f"AT LEAST ONE of {len(children)} condition(s)/group(s) must hold"
   946→   946→        ),
   947→   947→        children=children,
   948→   948→    )

   960→   960→def evaluate_rule_set(
   961→   961→    rule_set: RuleSet,
   962→   962→    candles_by_timeframe: dict[str, list[dict[str, Any]]],
   963→   963→) -> EvaluationResult:
   964→   964→    """Evaluate a validated rule set against caller-supplied candles."""
   965→   965→
   966→   966→    from datetime import datetime, timezone
   967→   967→
   968→   968→    evaluated_at = datetime.now(tz=timezone.utc).isoformat(timespec="milliseconds")
   969→   969→    warnings: list[dict[str, Any]] = []
   970→   970→    used_timeframes: set[str] = set()
   971→   971→    collect_timeframes(rule_set.conditions, used_timeframes)
   972→   972→    collect_timeframes(rule_set.confirmation, used_timeframes)
   973→   973→    collect_timeframes(rule_set.invalidation, used_timeframes)
   974→   974→
   975→   975→    # Normalise + assess every referenced timeframe; keep usable series only.
   976→   976→    usable: dict[str, list[dict[str, Any]]] = {}
   977→   977→    missing_timeframes: list[str] = []
   978→   978→    for tf in sorted(used_timeframes):
   979→   979→        raw = candles_by_timeframe.get(tf)
   980→   980→        if not raw:
   981→   981→            missing_timeframes.append(tf)
   982→   982→            continue
   983→   983→        series_warnings, rows = assess_candle_series(
   984→   984→            raw, tf, _required_count_for_timeframe(rule_set, tf)
   985→   985→        )
   986→   986→        warnings.extend(series_warnings)
   987→   987→        # Drop candles with invalid OHLC from evaluation (warned above).
   988→   988→        clean = [row for row in rows if _row_valid(row)]
   989→   989→        if clean:
   990→   990→            usable[tf] = clean
   991→   991→        elif rows:
   992→   992→            missing_timeframes.append(tf)
   993→   993→
   994→   994→    if missing_timeframes:
   995→   995→        core = NodeResult(
   996→   996→            kind="group",
   997→   997→            id="group_core",
   998→   998→            logic="AND",
   999→   999→            satisfied=False,
  1000→  1000→            description="Not evaluated: required candle data is missing.",
  1001→  1001→            evidence={
  1002→  1002→                "missing_timeframes": missing_timeframes,
  1003→  1003→                "reason": "The request did not include usable candles for "
  1004→  1004→                "every timeframe referenced by the rule set. The platform "
  1005→  1005→                "does not fabricate market data.",
  1006→  1006→            },
  1007→  1007→        )
  1008→  1008→        return EvaluationResult(
  1009→  1009→            direction=rule_set.direction,
  1010→  1010→            setup_status="insufficient_data",
  1011→  1011→            core=core,
  1012→  1012→            confirmation=None,
  1013→  1013→            invalidation=None,
  1014→  1014→            warnings=warnings,
  1015→  1015→            summary=[
  1016→  1016→                f"Missing candle data for timeframe(s): {', '.join(missing_timeframes)}.",
  1017→  1017→                "Provide candles for all referenced timeframes to obtain a "
  1018→  1018→                "full evaluation.",
  1019→  1019→            ],
  1020→  1020→            evaluated_at=evaluated_at,
  1021→  1021→        )
  1022→  1022→
  1023→  1023→    core_result = evaluate_group(rule_set.conditions, usable)
  1024→  1024→    confirmation_result = (
  1025→  1025→        evaluate_group(rule_set.confirmation, usable)
  1026→  1026→        if rule_set.confirmation is not None
  1027→  1027→        else None
  1028→  1028→    )
  1029→  1029→    invalidation_result = (
  1030→  1030→        evaluate_group(rule_set.invalidation, usable)
  1031→  1031→        if rule_set.invalidation is not None
  1032→  1032→        else None
  1033→  1033→    )
  1034→  1034→
  1035→  1035→    if invalidation_result is not None and invalidation_result.satisfied:
  1036→  1036→        setup_status = "invalidated"
  1037→  1037→    elif (
  1038→  1038→        core_result.satisfied
  1039→  1039→        and (confirmation_result is None or confirmation_result.satisfied)
  1040→  1040→    ):
  1041→  1041→        setup_status = "conditions_met"
  1042→  1042→    else:
  1043→  1043→        setup_status = "conditions_missing"
  1044→  1044→
  1045→  1045→    return EvaluationResult(
  1046→  1046→        direction=rule_set.direction,
  1047→  1047→        setup_status=setup_status,
  1048→  1048→        core=core_result,
  1049→  1049→        confirmation=confirmation_result,
  1050→  1050→        invalidation=invalidation_result,
  1051→  1051→        warnings=warnings,
  1052→  1052→        summary=_build_summary(
  1053→  1053→            rule_set, setup_status, core_result, confirmation_result, invalidation_result
  1054→  1054→        ),
  1055→  1055→        evaluated_at=datetime.now(tz=timezone.utc).isoformat(timespec="milliseconds"),
  1056→  1056→    )

  1059→  1059→def _required_count(group: Group) -> int:
  1060→  1060→    """Worst-case candles needed by any condition in the tree."""
  1061→  1061→
  1062→  1062→    needed = 1
  1063→  1063→    for cond in group.conditions:
  1064→  1064→        needed = max(needed, cond.lookback, _minimum_candles(cond))
  1065→  1065→    for nested in group.groups:
  1066→  1066→        needed = max(needed, _required_count(nested))
  1067→  1067→    return needed

  1070→  1070→def _required_count_for_timeframe(rule_set: RuleSet, timeframe: str) -> int:
  1071→  1071→    required = 1
  1072→  1072→
  1073→  1073→    def scan(group: Group | None) -> None:
  1074→  1074→        nonlocal required
  1075→  1075→        if group is None:
  1076→  1076→            return
  1077→  1077→        for cond in group.conditions:
  1078→  1078→            if cond.timeframe != timeframe:
  1079→  1079→                continue
  1080→  1080→            required = max(required, cond.lookback, _minimum_candles(cond))
  1081→  1081→        for nested in group.groups:
  1082→  1082→            scan(nested)
  1083→  1083→
  1084→  1084→    scan(rule_set.conditions)
  1085→  1085→    scan(rule_set.confirmation)
  1086→  1086→    scan(rule_set.invalidation)
  1087→  1087→    return required

  1090→  1090→def _row_valid(row: dict[str, Any]) -> bool:
  1091→  1091→    values = [row.get(k) for k in ("open", "high", "low", "close")]
  1092→  1092→    if any(
  1093→  1093→        v is None or isinstance(v, bool) or not isinstance(v, (int, float))
  1094→  1094→        or not math.isfinite(float(v)) or float(v) <= 0
  1095→  1095→        for v in values
  1096→  1096→    ):
  1097→  1097→        return False
  1098→  1098→    o, h, l, c = (float(v) for v in values)
  1099→  1099→    return h >= l and h >= o and h >= c and l <= o and l <= c

  1102→  1102→def _build_summary(
  1103→  1103→    rule_set: RuleSet,
  1104→  1104→    status: str,
  1105→  1105→    core: NodeResult,
  1106→  1106→    confirmation: NodeResult | None,
  1107→  1107→    invalidation: NodeResult | None,
  1108→  1108→) -> list[str]:
  1109→  1109→    lines: list[str] = []
  1110→  1110→    direction_label = {
  1111→  1111→        "bullish": "BULLISH (look for long setups)",
  1112→  1112→        "bearish": "BEARISH (look for short setups)",
  1113→  1113→        "both": "BOTH DIRECTIONS (long and short setups)",
  1114→  1114→    }[rule_set.direction]
  1115→  1115→    lines.append(f"Strategy direction: {direction_label}.")
  1116→  1116→    if status == "invalidated":
  1117→  1117→        lines.append(
  1118→  1118→            "The invalidation rules are currently satisfied — the setup is "
  1119→  1119→            "considered invalid regardless of the core conditions."
  1120→  1120→        )
  1121→  1121→    elif status == "conditions_met":
  1122→  1122→        extra = " and all confirmation rules" if confirmation is not None else ""
  1123→  1123→        lines.append(f"Core conditions{extra} are met on the provided data.")
  1124→  1124→    elif status == "conditions_missing":
  1125→  1125→        failed = _count_failed(core) + (_count_failed(confirmation) if confirmation else 0)
  1126→  1126→        optional_unmet = _count_optional_unmet(core) + (
  1127→  1127→            _count_optional_unmet(confirmation) if confirmation else 0
  1128→  1128→        )
  1129→  1129→        if failed:
  1130→  1130→            lines.append(
  1131→  1131→                f"{failed} required condition(s)/group(s) are not satisfied "
  1132→  1132→                "yet — the setup is not confirmed by the current data."
  1133→  1133→            )
  1134→  1134→        if optional_unmet:
  1135→  1135→            lines.append(
  1136→  1136→                f"{optional_unmet} optional rule(s) are unmet (they do not "
  1137→  1137→                "block the setup)."
  1138→  1138→            )
  1139→  1139→    lines.append(
  1140→  1140→        f"Evaluated {rule_set.mode.replace('_', ' ')} rule set v"
  1141→  1141→        f"{rule_set.schema_version} on timeframes: "
  1142→  1142→        f"{', '.join(rule_set.timeframes)}."
  1143→  1143→    )
  1144→  1144→    if invalidation is not None and not invalidation.satisfied:
  1145→  1145→        lines.append("Invalidation rules are NOT triggered (good for this setup).")
  1146→  1146→    return lines

  1149→  1149→def _count_failed(node: NodeResult | None) -> int:
  1150→  1150→    """Count unsatisfied REQUIRED conditions/groups (optional excluded)."""
  1151→  1151→
  1152→  1152→    if node is None:
  1153→  1153→        return 0
  1154→  1154→    if node.optional and not node.satisfied:
  1155→  1155→        return 0
  1156→  1156→    if node.kind == "condition":
  1157→  1157→        return 0 if node.satisfied else 1
  1158→  1158→    return sum(_count_failed(child) for child in node.children)

  1161→  1161→def _count_optional_unmet(node: NodeResult | None) -> int:
  1162→  1162→    if node is None:
  1163→  1163→        return 0
  1164→  1164→    if node.kind == "condition":
  1165→  1165→        return 1 if (node.optional and not node.satisfied) else 0
  1166→  1166→    return sum(_count_optional_unmet(child) for child in node.children)
