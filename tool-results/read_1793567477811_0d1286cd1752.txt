     1→"""Deterministic, explainable rule evaluation.
     2→
     3→Semantics (documented in README, /api/v1/rule-types and /api/v1/meta):
     4→
     5→- ``price_level`` — ``above``/``below``: the **latest candle** in the
     6→  lookback window satisfies the comparison; ``crosses_above``/
     7→  ``crosses_below``: the latest candle completes the cross.
     8→- ``candle_property`` — derived property of the latest candle vs a
     9→  threshold.
    10→- ``candle_pattern`` — classic pattern on the latest 1-2 candles.
    11→- ``swing_point`` — a swing of the given strength CONFIRMED inside the
    12→  window (confirmed = ``strength`` subsequent candles exist inside the
    13→  supplied data; nothing from the future is assumed).
    14→- ``structure_pattern`` — the last N confirmed swings are strictly
    15→  rising/falling (HH / HL / LH / LL).
    16→- ``market_structure_break`` — latest close beyond the most recent
    17→  confirmed opposite swing (generic, precisely-defined BOS).
    18→- ``moving_average`` — SMA/EMA vs a threshold or a price field, with
    19→  fresh-cross semantics for ``crosses_*``.
    20→- ``price_zone`` — interaction with a user-defined [low, high] zone.
    21→- Group logic: ``AND`` = every required child satisfied (unsatisfied
    22→  **optional** children do not fail the group but are reported);
    23→  ``OR`` = at least one child satisfied.
    24→- Overall status:
    25→    * ``invalidated``            — the invalidation group is satisfied
    26→    * ``conditions_met``         — core satisfied AND confirmation (if any) satisfied
    27→    * ``conditions_missing``     — core or confirmation unmet, no invalidation
    28→    * ``insufficient_data``      — at least one referenced timeframe has no usable data
    29→
    30→**No look-ahead bias**: every evaluator is a pure function of the candles
    31→supplied at evaluation time. A condition never uses information that was
    32→unavailable when the last supplied candle closed (swing confirmation
    33→explicitly requires ``strength`` candles AFTER the swing inside the data).
    34→
    35→The evaluator never fetches or invents market data.
    36→"""
    37→
    38→from __future__ import annotations
    39→
    40→import math
    41→from dataclasses import dataclass, field
    42→from typing import Any
    43→
    44→from app.domains.analysis.quality import assess_candle_series

    50→    """Evaluation result for a condition OR a group (explainability tree)."""

    53→    id: str
    54→    logic: str | None  # groups only
    55→    satisfied: bool
    56→    description: str
    57→    evidence: dict[str, Any] = field(default_factory=dict)
    58→    children: list["NodeResult"] = field(default_factory=list)
    59→    optional: bool = False

    62→        data: dict[str, Any] = {

    74→        return data

   141→
   142→def _crossover_satisfied(
   143→    operator: str,
   144→    series_a: list[float],
   145→    series_b: list[float],
   146→) -> tuple[bool, list[int]]:
   147→    """Shared fresh-cross semantics. Returns (satisfied, cross_indices)."""
   148→
   149→    crosses: list[int] = []
   150→    for i in range(1, len(series_a)):
   151→        prev, cur = series_a[i - 1], series_a[i]
   152→        if operator == "crosses_above":
   153→            hit = prev <= series_b[i - 1] and cur > series_b[i]
   154→        else:
   155→            hit = prev >= series_b[i - 1] and cur < series_b[i]
   156→        if hit:
   157→            crosses.append(i)
   158→    satisfied = bool(crosses) and crosses[-1] == len(series_a) - 1
   159→    return satisfied, crosses

   166→
   167→def evaluate_price_level_condition(
   168→    condition: Condition, candles: list[dict[str, Any]]
   169→) -> NodeResult:
   170→    """Evaluate a ``price_level`` condition with full evidence."""
   171→
   172→    tf = condition.timeframe
   173→    lookback = condition.lookback
   174→    window = _window(candles, lookback)
   175→    field = str(condition.params["field"])
   176→    operator = str(condition.params["operator"])
   177→    threshold = float(condition.params["value"])
   178→    needs_two = operator in ("crosses_above", "crosses_below")
   179→    series = _series_from(window, field)
   180→
   181→    description = (
   182→        f"{field} {operator.replace('_', ' ')} "
   183→        f"{threshold:g} on {tf} (last {len(window)} candle(s))"
   184→    )
   185→    evidence: dict[str, Any] = {
   186→        "condition_type": condition.type,
   187→        "timeframe": tf,
   188→        "field": field,
   189→        "operator": operator,
   190→        "threshold": threshold,
   191→        "lookback": lookback,
   192→        "candles_evaluated": len(window),
   193→        "candles_available": len(candles),
   194→    }
   195→
   196→    if not window or (needs_two and len(window) < 2):
   197→        return _unsatisfied(
   198→            condition,
   199→            description,
   200→            evidence,
   201→            reason=(
   202→                "Cross conditions need at least 2 candles in the window."
   203→                if needs_two
   204→                else "No usable candles for this timeframe."
   205→            ),
   206→        )
   207→
   208→    if operator in ("above", "below"):
   209→        comparisons = [
   210→            (v > threshold) if operator == "above" else (v < threshold) for v in series
   211→        ]
   212→        satisfied = comparisons[-1]
   213→        matched = sum(comparisons)
   214→        first_hit = next(
   215→            (window[i]["time"] for i, hit in enumerate(comparisons) if hit), None
   216→        )
   217→        last_hit = next(
   218→            (window[i]["time"] for i in range(len(comparisons) - 1, -1, -1) if comparisons[i]),
   219→            None,
   220→        )
   221→        evidence.update(
   222→            {
   223→                "current_value": series[-1],
   224→                "matched_count": matched,
   225→                "first_satisfied_time": first_hit,
   226→                "last_satisfied_time": last_hit,
   227→                "criterion": "latest candle must satisfy the comparison",
   228→            }
   229→        )
   230→    else:  # crosses_above / crosses_below
   231→        crosses: list[dict[str, Any]] = []
   232→        for i in range(1, len(series)):
   233→            prev, cur = series[i - 1], series[i]
   234→            if operator == "crosses_above":
   235→                hit = prev <= threshold < cur
   236→            else:
   237→                hit = prev >= threshold > cur
   238→            if hit:
   239→                crosses.append({"time": window[i]["time"], "value": cur})
   240→        # Documented semantics: the FRESH cross decides — the latest candle
   241→        # must be the one completing the cross.
   242→        satisfied = bool(crosses) and crosses[-1]["time"] == window[-1]["time"]
   243→        evidence.update(
   244→            {
   245→                "current_value": series[-1],
   246→                "previous_value": series[-2],
   247→                "crosses_in_window": crosses[-10:],  # cap for payload size
   248→                "total_crosses_in_window": len(crosses),
   249→                "last_cross_time": crosses[-1]["time"] if crosses else None,
   250→                "criterion": "latest candle must complete the cross",
   251→            }
   252→        )
   253→
   254→    return _node(condition, description, evidence, satisfied)

   257→# --------------------------------------------------------------------------- #
   258→# candle_property
   259→# --------------------------------------------------------------------------- #
   260→
   261→
   262→def _candle_property_value(candle: dict[str, Any], prop: str) -> float:
   263→    o, h, l, c = (float(candle[k]) for k in ("open", "high", "low", "close"))
   264→    if prop in ("open", "high", "low", "close"):
   265→        return {"open": o, "high": h, "low": l, "close": c}[prop]
   266→    if prop == "body":
   267→        return abs(c - o)
   268→    if prop == "range":
   269→        return h - l
   270→    if prop == "upper_wick":
   271→        return h - max(o, c)
   272→    if prop == "lower_wick":
   273→        return min(o, c) - l
   274→    raise ValueError(f"Unknown candle property '{prop}'")  # pragma: no cover
   275→
   276→
   277→def evaluate_candle_property(
   278→    condition: Condition, candles: list[dict[str, Any]]
   279→) -> NodeResult:
   280→    tf = condition.timeframe
   281→    prop = str(condition.params["property"])
   282→    operator = str(condition.params["operator"])
   283→    threshold = float(condition.params["value"])
   284→    window = _window(candles, condition.lookback)
   285→
   286→    description = (
   287→        f"{prop.replace('_', ' ')} {operator} {threshold:g} on {tf} "
   288→        "(latest candle)"
   289→    )
   290→    evidence: dict[str, Any] = {
   291→        "condition_type": condition.type,
   292→        "timeframe": tf,
   293→        "property": prop,
   294→        "operator": operator,
   295→        "threshold": threshold,
   296→        "candles_evaluated": len(window),
   297→        "candles_available": len(candles),
   298→    }
   299→    if not window:
   300→        return _unsatisfied(
   301→            condition, description, evidence, "No usable candles for this timeframe."
   302→        )
   303→    value = _candle_property_value(window[-1], prop)
   304→    satisfied = value > threshold if operator == "above" else value < threshold
   305→    evidence.update(
   306→        {
   307→            "current_value": value,
   308→            "criterion": "property of the latest candle must satisfy the comparison",
   309→        }
   310→    )
   311→    return _node(condition, description, evidence, satisfied)

   314→# --------------------------------------------------------------------------- #
   315→# candle_pattern
   316→# --------------------------------------------------------------------------- #
   317→
   318→
   319→def evaluate_candle_pattern(
   320→    condition: Condition, candles: list[dict[str, Any]]
   321→) -> NodeResult:
   322→    tf = condition.timeframe
   323→    pattern = str(condition.params["pattern"])
   324→    max_body_ratio = float(condition.params.get("max_body_ratio", 0.1))
   325→    window = _window(candles, condition.lookback)
   326→
   327→    description = f"{pattern.replace('_', ' ')} on {tf} (latest candle)"
   328→    evidence: dict[str, Any] = {
   329→        "condition_type": condition.type,
   330→        "timeframe": tf,
   331→        "pattern": pattern,
   332→        "candles_evaluated": len(window),
   333→        "candles_available": len(candles),
   334→    }
   335→    if not window:
   336→        return _unsatisfied(
   337→            condition, description, evidence, "No usable candles for this timeframe."
   338→        )
   339→    needs_two = pattern in ("bullish_engulfing", "bearish_engulfing", "inside_bar")
   340→    if needs_two and len(window) < 2:
   341→        return _unsatisfied(
   342→            condition,
   343→            description,
   344→            evidence,
   345→            reason=f"Pattern '{pattern}' needs at least 2 candles.",
   346→        )
   347→
   348→    cur, prev = window[-1], window[-2] if len(window) > 1 else None
   349→
   350→    def body(c: dict[str, Any]) -> float:
   351→        return abs(float(c["close"]) - float(c["open"]))
   352→
   353→    def is_bull(c: dict[str, Any]) -> bool:
   354→        return float(c["close"]) > float(c["open"])
   355→
   356→    def is_bear(c: dict[str, Any]) -> bool:
   357→        return float(c["close"]) < float(c["open"])
   358→
   359→    rng = float(cur["high"]) - float(cur["low"])
   360→    body_cur = body(cur)
   361→    if rng <= 0:
   362→        matched = pattern == "doji"  # a zero-range candle is a perfect doji
   363→        evidence.update({"pattern_range": 0.0, "criterion": pattern})
   364→        return _node(condition, description, evidence, bool(matched))
   365→
   366→    matched = False
   367→    detail: dict[str, Any] = {"criterion": pattern}
   368→    if pattern == "bullish_engulfing" and prev is not None:
   369→        matched = (
   370→            is_bear(prev)
   371→            and is_bull(cur)
   372→            and float(cur["open"]) <= float(prev["close"])
   373→            and float(cur["close"]) >= float(prev["open"])
   374→        )
   375→    elif pattern == "bearish_engulfing" and prev is not None:
   376→        matched = (
   377→            is_bull(prev)
   378→            and is_bear(cur)
   379→            and float(cur["open"]) >= float(prev["close"])
   380→            and float(cur["close"]) <= float(prev["open"])
   381→        )
   382→    elif pattern == "hammer":
   383→        lower_wick = min(float(cur["open"]), float(cur["close"])) - float(cur["low"])
   384→        upper_wick = float(cur["high"]) - max(float(cur["open"]), float(cur["close"]))
   385→        mid = (float(cur["high"]) + float(cur["low"])) / 2
   386→        matched = (
   387→            lower_wick >= 2 * body_cur
   388→            and upper_wick <= body_cur
   389→            and float(cur["close"]) >= mid
   390→        )
   391→        detail.update({"lower_wick": lower_wick, "upper_wick": upper_wick})
   392→    elif pattern == "shooting_star":
   393→        lower_wick = min(float(cur["open"]), float(cur["close"])) - float(cur["low"])
   394→        upper_wick = float(cur["high"]) - max(float(cur["open"]), float(cur["close"]))
   395→        mid = (float(cur["high"]) + float(cur["low"])) / 2
   396→        matched = (
   397→            upper_wick >= 2 * body_cur
   398→            and lower_wick <= body_cur
   399→            and float(cur["close"]) <= mid
   400→        )
   401→        detail.update({"lower_wick": lower_wick, "upper_wick": upper_wick})
   402→    elif pattern == "doji":
   403→        matched = body_cur <= max_body_ratio * rng
   404→        detail.update({"body": body_cur, "range": rng, "max_body_ratio": max_body_ratio})
   405→    elif pattern == "inside_bar" and prev is not None:
   406→        matched = float(cur["high"]) <= float(prev["high"]) and float(cur["low"]) >= float(prev["low"])
   407→        if prev is not None:
   408→            detail.update(
   409→                {
   410→                    "previous_high": float(prev["high"]),
   411→                    "previous_low": float(prev["low"]),
   412→                }
   413→            )
   414→
   415→    evidence.update(detail)
   416→    evidence["matched_pattern"] = matched
   417→    return _node(condition, description, evidence, matched)
   418→

   420→# --------------------------------------------------------------------------- #
   421→# swing points (shared by three rule types)
   422→# --------------------------------------------------------------------------- #
   423→
   424→
   425→def _confirmed_swings(
   426→    window: list[dict[str, Any]], kind: str, strength: int
   427→) -> list[dict[str, Any]]:
   428→    """Swings confirmed strictly inside the supplied window (no look-ahead).
   429→
   430→    A candle at index ``i`` is a confirmed swing high when its high is
   431→    strictly greater than the highs of the ``strength`` candles before AND
   432→    after it, and at least ``strength`` candles exist after it inside the
   433→    window. Returns a list of {"index", "time", "price"} in chronological
   434→    order.
   435→    """
   436→
   437→    swings: list[dict[str, Any]] = []
   438→    n = len(window)
   439→    for i in range(strength, n - strength):
   440→        if kind == "high":
   441→            value = float(window[i]["high"])
   442→            is_swing = all(
   443→                value > float(window[j]["high"])
   444→                for j in range(i - strength, i + strength + 1)
   445→                if j != i
   446→            )
   447→        else:
   448→            value = float(window[i]["low"])
   449→            is_swing = all(
   450→                value < float(window[j]["low"])
   451→                for j in range(i - strength, i + strength + 1)
   452→                if j != i
   453→            )
   454→        if is_swing:
   455→            swings.append(
   456→                {"index": i, "time": window[i]["time"], "price": value}
   457→            )
   458→    return swings
   459→
   460→
   461→def evaluate_swing_point(
   462→    condition: Condition, candles: list[dict[str, Any]]
   463→) -> NodeResult:
   464→    tf = condition.timeframe
   465→    kind = str(condition.params["kind"])
   466→    strength = int(condition.params["strength"])
   467→    window = _window(candles, condition.lookback)
   468→
   469→    description = (
   470→        f"confirmed swing {kind} (strength {strength}) within last "
   471→        f"{len(window)} candle(s) on {tf}"
   472→    )
   473→    evidence: dict[str, Any] = {
   474→        "condition_type": condition.type,
   475→        "timeframe": tf,
   476→        "kind": kind,
   477→        "strength": strength,
   478→        "lookback": condition.lookback,
   479→        "candles_evaluated": len(window),
   480→        "candles_available": len(candles),
   481→        "criterion": (
   482→            "a swing high/low confirmed by `strength` subsequent candles "
   483→            "inside the window (no look-ahead)"
   484→        ),
   485→    }
   486→    minimum = 2 * strength + 1
   487→    if len(window) < minimum:
   488→        return _unsatisfied(
   489→            condition,
   490→            description,
   491→            evidence,
   492→            reason=(
   493→                f"Detecting strength-{strength} swings needs at least "
   494→                f"{minimum} candles in the window."
   495→            ),
   496→        )
   497→    swings = _confirmed_swings(window, kind, strength)
   498→    evidence["swings_found"] = len(swings)
   499→    evidence["swings"] = [
   500→        {"time": s["time"], "price": s["price"]} for s in swings[-5:]
   501→    ]
   502→    return _node(condition, description, evidence, bool(swings))

   505→# --------------------------------------------------------------------------- #
   506→# structure_pattern (HH / HL / LH / LL)
   507→# --------------------------------------------------------------------------- #
   508→
   509→
   510→def evaluate_structure_pattern(
   511→    condition: Condition, candles: list[dict[str, Any]]
   512→) -> NodeResult:
   513→    tf = condition.timeframe
   514→    pattern = str(condition.params["pattern"])
   515→    count = int(condition.params["count"])
   516→    strength = int(condition.params.get("strength", 2))
   517→    window = _window(candles, condition.lookback)
   518→
   519→    kind = "high" if pattern in ("higher_highs", "lower_highs") else "low"
   520→    rising = pattern in ("higher_highs", "higher_lows")
   521→    description = (
   522→        f"{pattern.replace('_', ' ')}: last {count} confirmed swing {kind}s "
   523→        f"{'strictly rising' if rising else 'strictly falling'} on {tf}"
   524→    )
   525→    evidence: dict[str, Any] = {
   526→        "condition_type": condition.type,
   527→        "timeframe": tf,
   528→        "pattern": pattern,
   529→        "count": count,
   530→        "strength": strength,
   531→        "candles_evaluated": len(window),
   532→        "candles_available": len(candles),
   533→        "criterion": (
   534→            f"the most recent {count} confirmed swing {kind}s must be "
   535→            f"{'strictly rising' if rising else 'strictly falling'}"
   536→        ),
   537→    }
   538→    minimum = 2 * strength + count
   539→    if len(window) < minimum:
   540→        return _unsatisfied(
   541→            condition,
   542→            description,
   543→            evidence,
   544→            reason=(
   545→                f"Confirming {count} strength-{strength} swings needs at "
   546→                f"least {minimum} candles in the window."
   547→            ),
   548→        )
   549→    swings = _confirmed_swings(window, kind, strength)[-count:]
   550→    evidence["swings_found"] = len(swings)
   551→    evidence["swing_prices"] = [s["price"] for s in swings]
   552→    if len(swings) < count:
   553→        evidence["reason"] = (
   554→            f"Only {len(swings)} confirmed swing {kind}(s) in the window; "
   555→            f"{count} are required."
   556→        )
   557→        return _node(condition, description, evidence, False)
   558→    if rising:
   559→        satisfied = all(
   560→            swings[i]["price"] > swings[i - 1]["price"] for i in range(1, len(swings))
   561→        )
   562→    else:
   563→        satisfied = all(
   564→            swings[i]["price"] < swings[i - 1]["price"] for i in range(1, len(swings))
   565→        )
   566→    return _node(condition, description, evidence, satisfied)

   569→# --------------------------------------------------------------------------- #
   570→# market_structure_break (generic BOS)
   571→# --------------------------------------------------------------------------- #
   572→
   573→
   574→def evaluate_market_structure_break(
   575→    condition: Condition, candles: list[dict[str, Any]]
   576→) -> NodeResult:
   577→    tf = condition.timeframe
   578→    side = str(condition.params["side"])
   579→    strength = int(condition.params.get("strength", 2))
   580→    window = _window(candles, condition.lookback)
   581→
   582→    swing_kind = "high" if side == "bullish" else "low"
   583→    description = (
   584→        f"{side} structure break: close beyond the most recent confirmed "
   585→        f"swing {swing_kind} (strength {strength}) on {tf}"
   586→    )
   587→    evidence: dict[str, Any] = {
   588→        "condition_type": condition.type,
   589→        "timeframe": tf,
   590→        "side": side,
   591→        "strength": strength,
   592→        "candles_evaluated": len(window),
   593→        "candles_available": len(candles),
   594→        "criterion": (
   595→            f"latest close must be {'above' if side == 'bullish' else 'below'} "
   596→            f"the most recent confirmed swing {swing_kind} in the window"
   597→        ),
   598→    }
   599→    if len(window) < 2 * strength + 2:
   600→        return _unsatisfied(
   601→            condition,
   602→            description,
   603→            evidence,
   604→            reason=(
   605→                f"A break evaluation needs at least {2 * strength + 2} "
   606→                "candles (swing + the breaking candle)."
   607→            ),
   608→        )
   609→
   610→    # Swings confirmed strictly BEFORE the latest candle: the last
   611→    # `strength` candles cannot be confirmed swings, and the breaking
   612→    # candle itself is the latest one.
   613→    swing_zone = window[:-1]
   614→    swings = _confirmed_swings(swing_zone, swing_kind, strength)
   615→    evidence["reference_swings"] = [
   616→        {"time": s["time"], "price": s["price"]} for s in swings[-3:]
   617→    ]
   618→    if not swings:
   619→        return _unsatisfied(
   620→            condition,
   621→            description,
   622→            evidence,
   623→            reason=f"No confirmed swing {swing_kind} available as the break reference.",
   624→        )
   625→    reference = swings[-1]
   626→    close = float(window[-1]["close"])
   627→    satisfied = close > reference["price"] if side == "bullish" else close < reference["price"]
   628→    evidence.update(
   629→        {
   630→            "reference_swing_price": reference["price"],
   631→            "reference_swing_time": reference["time"],
   632→            "current_close": close,
   633→        }
   634→    )
   635→    return _node(condition, description, evidence, satisfied)
   636→

   638→# --------------------------------------------------------------------------- #
   639→# moving_average
   640→# --------------------------------------------------------------------------- #
   641→
   642→
   643→def _sma(series: list[float], period: int) -> list[float] | None:
   644→    if len(series) < period:
   645→        return None
   646→    out = [math.nan] * len(series)
   647→    running = sum(series[:period])
   648→    out[period - 1] = running / period
   649→    for i in range(period, len(series)):
   650→        running += series[i] - series[i - period]
   651→        out[i] = running / period
   652→    return out
   653→
   654→
   655→def _ema(series: list[float], period: int) -> list[float] | None:
   656→    if len(series) < period:
   657→        return None
   658→    out = [math.nan] * len(series)
   659→    alpha = 2.0 / (period + 1)
   660→    prev = sum(series[:period]) / period  # SMA seed (documented)
   661→    out[period - 1] = prev
   662→    for i in range(period, len(series)):
   663→        prev = alpha * series[i] + (1 - alpha) * prev
   664→        out[i] = prev
   665→    return out
   666→
   667→
   668→def evaluate_moving_average(
   669→    condition: Condition, candles: list[dict[str, Any]]
   670→) -> NodeResult:
   671→    tf = condition.timeframe
   672→    ma_type = str(condition.params["ma_type"])
   673→    period = int(condition.params["period"])
   674→    source = str(condition.params.get("source", "close"))
   675→    compare_to = str(condition.params["compare_to"])
   676→    operator = str(condition.params["operator"])
   677→    window = _window(candles, condition.lookback)
   678→
   679→    needs_two = operator in ("crosses_above", "crosses_below")
   680→    target_desc = (
   681→        f"{float(condition.params['value']):g}"
   682→        if compare_to == "value" and "value" in condition.params
   683→        else str(condition.params.get("price_field", "price field"))
   684→    )
   685→    description = (
   686→        f"{ma_type.upper()}({period}) of {source} {operator.replace('_', ' ')} "
   687→        f"{target_desc} on {tf} (last {len(window)} candle(s))"
   688→    )
   689→    evidence: dict[str, Any] = {
   690→        "condition_type": condition.type,
   691→        "timeframe": tf,
   692→        "ma_type": ma_type,
   693→        "period": period,
   694→        "source": source,
   695→        "compare_to": compare_to,
   696→        "operator": operator,
   697→        "lookback": condition.lookback,
   698→        "candles_evaluated": len(window),
   699→        "candles_available": len(candles),
   700→    }
   701→
   702→    if not window or (needs_two and len(window) < 2):
   703→        return _unsatisfied(
   704→            condition,
   705→            description,
   706→            evidence,
   707→            reason="Cross conditions need at least 2 candles in the window.",
   708→        )
   709→    series = _series_from(window, source)
   710→    ma_series = (_ema if ma_type == "ema" else _sma)(series, period)
   711→    if ma_series is None or math.isnan(ma_series[-1]) or (
   712→        needs_two and math.isnan(ma_series[-2])
   713→    ):
   714→        return _unsatisfied(
   715→            condition,
   716→            description,
   717→            evidence,
   718→            reason=(
   719→                f"{ma_type.upper()}({period}) needs at least {period} "
   720→                "candles in the window."
   721→            ),
   722→        )
   723→
   724→    if compare_to == "value":
   725→        threshold = float(condition.params["value"])
   726→        target_series = [threshold] * len(series)
   727→        evidence["threshold"] = threshold
   728→        target_label = f"{threshold:g}"
   729→    else:
   730→        target_field = str(condition.params["price_field"])
   731→        target_series = _series_from(window, target_field)
   732→        evidence["price_field"] = target_field
   733→        target_label = target_field
   734→
   735→    description = (
   736→        f"{ma_type.upper()}({period}) of {source} "
   737→        f"{operator.replace('_', ' ')} {target_label} on {tf} "
   738→        f"(last {len(window)} candle(s))"
   739→    )
   740→
   741→    if operator in ("above", "below"):
   742→        last_ma = ma_series[-1]
   743→        satisfied = last_ma > target_series[-1] if operator == "above" else last_ma < target_series[-1]
   744→        evidence.update(
   745→            {
   746→                "ma_value": last_ma,
   747→                "target_value": target_series[-1],
   748→                "criterion": "MA on the latest candle must satisfy the comparison",
   749→            }
   750→        )
   751→    else:
   752→        # Compare the MA against the target series at aligned indices; skip
   753→        # the MA warm-up (NaN) region — the first valid MA value sits at
   754→        # index period-1 (the SMA/EMA seed).
   755→        start = period - 1
   756→        ma_valid = ma_series[start:]
   757→        target_valid = target_series[start:]
   758→        satisfied, crosses = _crossover_satisfied(operator, ma_valid, target_valid)
   759→        evidence.update(
   760→            {
   761→                "ma_value": ma_series[-1],
   762→                "previous_ma_value": ma_series[-2],
   763→                "target_value": target_series[-1],
   764→                "previous_target_value": target_series[-2],
   765→                "crosses_in_window": [
   766→                    {"time": window[start + i]["time"], "value": ma_valid[i]}
   767→                    for i in crosses[-10:]
   768→                ],
   769→                "total_crosses_in_window": len(crosses),
   770→                "criterion": "latest candle must complete the cross",
   771→            }
   772→        )
   773→    return _node(condition, description, evidence, satisfied)
   774→

   780→
   781→def evaluate_price_zone(
   782→    condition: Condition, candles: list[dict[str, Any]]
   783→) -> NodeResult:
   784→    tf = condition.timeframe
   785→    zone_low = float(condition.params["zone_low"])
   786→    zone_high = float(condition.params["zone_high"])
   787→    operator = str(condition.params["operator"])
   788→    window = _window(candles, condition.lookback)
   789→
   790→    description = (
   791→        f"zone [{zone_low:g}, {zone_high:g}] {operator.replace('_', ' ')} "
   792→        f"on {tf} (last {len(window)} candle(s))"
   793→    )
   794→    evidence: dict[str, Any] = {
   795→        "condition_type": condition.type,
   796→        "timeframe": tf,
   797→        "zone_low": zone_low,
   798→        "zone_high": zone_high,
   799→        "operator": operator,
   800→        "lookback": condition.lookback,
   801→        "candles_evaluated": len(window),
   802→        "candles_available": len(candles),
   803→    }
   804→    if not window:
   805→        return _unsatisfied(
   806→            condition, description, evidence, "No usable candles for this timeframe."
   807→        )
   808→
   809→    def touches(c: dict[str, Any]) -> bool:
   810→        return float(c["low"]) <= zone_high and float(c["high"]) >= zone_low
   811→
   812→    if operator == "touched":
   813→        hits = [c for c in window if touches(c)]
   814→        evidence.update(
   815→            {
   816→                "touches": len(hits),
   817→                "first_touch_time": hits[0]["time"] if hits else None,
   818→                "last_touch_time": hits[-1]["time"] if hits else None,
   819→                "criterion": "any candle in the window overlaps the zone",
   820→            }
   821→        )
   822→        return _node(condition, description, evidence, bool(hits))
   823→
   824→    latest = window[-1]
   825→    close = float(latest["close"])
   826→    if operator == "closed_inside":
   827→        satisfied = zone_low <= close <= zone_high
   828→        evidence.update(
   829→            {
   830→                "current_close": close,
   831→                "criterion": "latest candle's close lies inside the zone (inclusive)",
   832→            }
   833→        )
   834→        return _node(condition, description, evidence, satisfied)
   835→
   836→    # rejected: latest candle touched the zone intrabar but closed outside.
   837→    touched = touches(latest)
   838→    closed_inside = zone_low <= close <= zone_high
   839→    satisfied = touched and not closed_inside
   840→    evidence.update(
   841→        {
   842→            "current_close": close,
   843→            "touched_zone": touched,
   844→            "criterion": "latest candle touched the zone but closed outside it",
   845→        }
   846→    )
   847→    return _node(condition, description, evidence, satisfied)

   860→    "swing_point": evaluate_swing_point,

   867→
   868→def _minimum_candles(condition: Condition) -> int:
   869→    """Worst-case minimum window size implied by the condition's type."""
   870→
   871→    p = condition.params
   872→    t = condition.type
   873→    if t == "moving_average":
   874→        return int(p.get("period", 2)) + (2 if str(p.get("operator", "")).startswith("crosses") else 1)
   875→    if t == "swing_point":
   876→        return 2 * int(p.get("strength", 1)) + 1
   877→    if t == "structure_pattern":
   878→        return 2 * int(p.get("strength", 2)) + int(p.get("count", 2))
   879→    if t == "market_structure_break":
   880→        return 2 * int(p.get("strength", 2)) + 2
   881→    if t == "candle_pattern":
   882→        return 2
   883→    return 1

   886→# --------------------------------------------------------------------------- #
   887→# Group + rule set evaluation
   888→# --------------------------------------------------------------------------- #
   889→
   890→
   891→def evaluate_group(
   892→    group: Group, candles_by_timeframe: dict[str, list[dict[str, Any]]]
   893→) -> NodeResult:
   894→    children: list[NodeResult] = []
   895→    for cond in group.conditions:
   896→        candles = candles_by_timeframe.get(cond.timeframe, [])
   897→        evaluator = CONDITION_EVALUATORS.get(cond.type)
   898→        if evaluator is None:
   899→            # Defensive: validation already rejects unknown/unavailable
   900→            # types. If one ever reaches the evaluator, fail CLOSED with an
   901→            # explicit reason instead of pretending it was checked.
   902→            children.append(
   903→                NodeResult(
   904→                    kind="condition",
   905→                    id=cond.id,
   906→                    logic=None,
   907→                    satisfied=False,
   908→                    description=f"{cond.type} on {cond.timeframe}",
   909→                    evidence={
   910→                        "condition_type": cond.type,
   911→                        "reason": (
   912→                            "No evaluator registered for this rule type. "
   913→                            "The platform does not pretend unsupported rules "
   914→                            "are satisfied."
   915→                        ),
   916→                    },
   917→                    optional=cond.optional,
   918→                )
   919→            )
   920→            continue
   921→        children.append(evaluator(cond, candles))
   922→    for nested in group.groups:
   923→        children.append(evaluate_group(nested, candles_by_timeframe))
   924→
   925→    required = [child for child in children if not child.optional]
   926→    if group.logic == "AND":
   927→        satisfied = bool(required) and all(child.satisfied for child in required)
   928→    else:  # OR
   929→        satisfied = any(child.satisfied for child in children)
   930→
   931→    optional_unmet = sum(1 for c in children if c.optional and not c.satisfied)
   932→    suffix = (
   933→        f" ({optional_unmet} optional rule(s) currently unmet)"
   934→        if optional_unmet
   935→        else ""
   936→    )
   937→    return NodeResult(
   938→        kind="group",
   939→        id=f"group_{group.logic.lower()}_{id(group) % 100000:05d}",
   940→        logic=group.logic,
   941→        satisfied=satisfied,
   942→        description=(
   943→            f"ALL of {len(children)} condition(s)/group(s) must hold{suffix}"
   944→            if group.logic == "AND"
   945→            else f"AT LEAST ONE of {len(children)} condition(s)/group(s) must hold"
   946→        ),
   947→        children=children,
   948→    )

   950→
   951→def collect_timeframes(group: Group | None, into: set[str]) -> None:
   952→    if group is None:
   953→        return
   954→    for cond in group.conditions:
   955→        into.add(cond.timeframe)
   956→    for nested in group.groups:
   957→        collect_timeframes(nested, into)

   960→def evaluate_rule_set(
   961→    rule_set: RuleSet,
   962→    candles_by_timeframe: dict[str, list[dict[str, Any]]],
   963→) -> EvaluationResult:
   964→    """Evaluate a validated rule set against caller-supplied candles."""
   965→
   966→    from datetime import datetime, timezone
   967→
   968→    evaluated_at = datetime.now(tz=timezone.utc).isoformat(timespec="milliseconds")
   969→    warnings: list[dict[str, Any]] = []
   970→    used_timeframes: set[str] = set()
   971→    collect_timeframes(rule_set.conditions, used_timeframes)
   972→    collect_timeframes(rule_set.confirmation, used_timeframes)
   973→    collect_timeframes(rule_set.invalidation, used_timeframes)
   974→
   975→    # Normalise + assess every referenced timeframe; keep usable series only.
   976→    usable: dict[str, list[dict[str, Any]]] = {}
   977→    missing_timeframes: list[str] = []
   978→    for tf in sorted(used_timeframes):
   979→        raw = candles_by_timeframe.get(tf)
   980→        if not raw:
   981→            missing_timeframes.append(tf)
   982→            continue
   983→        series_warnings, rows = assess_candle_series(
   984→            raw, tf, _required_count_for_timeframe(rule_set, tf)
   985→        )
   986→        warnings.extend(series_warnings)
   987→        # Drop candles with invalid OHLC from evaluation (warned above).
   988→        clean = [row for row in rows if _row_valid(row)]
   989→        if clean:
   990→            usable[tf] = clean
   991→        elif rows:
   992→            missing_timeframes.append(tf)
   993→
   994→    if missing_timeframes:
   995→        core = NodeResult(
   996→            kind="group",
   997→            id="group_core",
   998→            logic="AND",
   999→            satisfied=False,
  1000→            description="Not evaluated: required candle data is missing.",
  1001→            evidence={
  1002→                "missing_timeframes": missing_timeframes,
  1003→                "reason": "The request did not include usable candles for "
  1004→                "every timeframe referenced by the rule set. The platform "
  1005→                "does not fabricate market data.",
  1006→            },
  1007→        )
  1008→        return EvaluationResult(
  1009→            direction=rule_set.direction,
  1010→            setup_status="insufficient_data",
  1011→            core=core,
  1012→            confirmation=None,
  1013→            invalidation=None,
  1014→            warnings=warnings,
  1015→            summary=[
  1016→                f"Missing candle data for timeframe(s): {', '.join(missing_timeframes)}.",
  1017→                "Provide candles for all referenced timeframes to obtain a "
  1018→                "full evaluation.",
  1019→            ],
  1020→            evaluated_at=evaluated_at,
  1021→        )
  1022→
  1023→    core_result = evaluate_group(rule_set.conditions, usable)
  1024→    confirmation_result = (
  1025→        evaluate_group(rule_set.confirmation, usable)
  1026→        if rule_set.confirmation is not None
  1027→        else None
  1028→    )
  1029→    invalidation_result = (
  1030→        evaluate_group(rule_set.invalidation, usable)
  1031→        if rule_set.invalidation is not None
  1032→        else None
  1033→    )
  1034→
  1035→    if invalidation_result is not None and invalidation_result.satisfied:
  1036→        setup_status = "invalidated"
  1037→    elif (
  1038→        core_result.satisfied
  1039→        and (confirmation_result is None or confirmation_result.satisfied)
  1040→    ):
  1041→        setup_status = "conditions_met"
  1042→    else:
  1043→        setup_status = "conditions_missing"
  1044→
  1045→    return EvaluationResult(
  1046→        direction=rule_set.direction,
  1047→        setup_status=setup_status,
  1048→        core=core_result,
  1049→        confirmation=confirmation_result,
  1050→        invalidation=invalidation_result,
  1051→        warnings=warnings,
  1052→        summary=_build_summary(
  1053→            rule_set, setup_status, core_result, confirmation_result, invalidation_result
  1054→        ),
  1055→        evaluated_at=datetime.now(tz=timezone.utc).isoformat(timespec="milliseconds"),
  1056→    )
  1057→

  1059→def _required_count(group: Group) -> int:
  1060→    """Worst-case candles needed by any condition in the tree."""
  1061→
  1062→    needed = 1
  1063→    for cond in group.conditions:
  1064→        needed = max(needed, cond.lookback, _minimum_candles(cond))
  1065→    for nested in group.groups:
  1066→        needed = max(needed, _required_count(nested))
  1067→    return needed
  1068→
  1069→
  1070→def _required_count_for_timeframe(rule_set: RuleSet, timeframe: str) -> int:
  1071→    required = 1
  1072→
  1073→    def scan(group: Group | None) -> None:
  1074→        nonlocal required
  1075→        if group is None:
  1076→            return
  1077→        for cond in group.conditions:
  1078→            if cond.timeframe != timeframe:
  1079→                continue
  1080→            required = max(required, cond.lookback, _minimum_candles(cond))
  1081→        for nested in group.groups:
  1082→            scan(nested)
  1083→
  1084→    scan(rule_set.conditions)
  1085→    scan(rule_set.confirmation)
  1086→    scan(rule_set.invalidation)
  1087→    return required
  1088→

  1090→def _row_valid(row: dict[str, Any]) -> bool:
  1091→    values = [row.get(k) for k in ("open", "high", "low", "close")]
  1092→    if any(
  1093→        v is None or isinstance(v, bool) or not isinstance(v, (int, float))
  1094→        or not math.isfinite(float(v)) or float(v) <= 0
  1095→        for v in values
  1096→    ):
  1097→        return False
  1098→    o, h, l, c = (float(v) for v in values)
  1099→    return h >= l and h >= o and h >= c and l <= o and l <= c

  1102→def _build_summary(
  1103→    rule_set: RuleSet,
  1104→    status: str,
  1105→    core: NodeResult,
  1106→    confirmation: NodeResult | None,
  1107→    invalidation: NodeResult | None,
  1108→) -> list[str]:
  1109→    lines: list[str] = []
  1110→    direction_label = {
  1111→        "bullish": "BULLISH (look for long setups)",
  1112→        "bearish": "BEARISH (look for short setups)",
  1113→        "both": "BOTH DIRECTIONS (long and short setups)",
  1114→    }[rule_set.direction]
  1115→    lines.append(f"Strategy direction: {direction_label}.")
  1116→    if status == "invalidated":
  1117→        lines.append(
  1118→            "The invalidation rules are currently satisfied — the setup is "
  1119→            "considered invalid regardless of the core conditions."
  1120→        )
  1121→    elif status == "conditions_met":
  1122→        extra = " and all confirmation rules" if confirmation is not None else ""
  1123→        lines.append(f"Core conditions{extra} are met on the provided data.")
  1124→    elif status == "conditions_missing":
  1125→        failed = _count_failed(core) + (_count_failed(confirmation) if confirmation else 0)
  1126→        optional_unmet = _count_optional_unmet(core) + (
  1127→            _count_optional_unmet(confirmation) if confirmation else 0
  1128→        )
  1129→        if failed:
  1130→            lines.append(
  1131→                f"{failed} required condition(s)/group(s) are not satisfied "
  1132→                "yet — the setup is not confirmed by the current data."
  1133→            )
  1134→        if optional_unmet:
  1135→            lines.append(
  1136→                f"{optional_unmet} optional rule(s) are unmet (they do not "
  1137→                "block the setup)."
  1138→            )
  1139→    lines.append(
  1140→        f"Evaluated {rule_set.mode.replace('_', ' ')} rule set v"
  1141→        f"{rule_set.schema_version} on timeframes: "
  1142→        f"{', '.join(rule_set.timeframes)}."
  1143→    )
  1144→    if invalidation is not None and not invalidation.satisfied:
  1145→        lines.append("Invalidation rules are NOT triggered (good for this setup).")
  1146→    return lines
