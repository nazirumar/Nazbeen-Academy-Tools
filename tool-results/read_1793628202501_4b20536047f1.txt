     1→"""Research persistence + owner-scoped queries (Phase 8).
     2→
     3→Creation performs the FULL validation cascade BEFORE anything is queued
     4→(mirroring the Phase 6 backtest service): strategy version exists and is
     5→immutable, modes agree, timeframe is declared by the strategy, symbol is
     6→in scope, stored candles cover the window, the search space is bounded,
     7→the objective is computable for the configured simulation mode, the
     8→walk-forward windows fit into the period, and the total engine-walk
     9→budget covers candidates x phases + robustness scenarios.
    10→
    11→Every query is scoped by ``owner_id``; foreign rows answer 404 (no
    12→existence leak). New runs NEVER overwrite previous research results.
    13→"""

    53→def _resolve_timeframe(
    54→    requested: str | None, rule_set_timeframes: tuple[str, ...]
    55→) -> str:
    56→    if requested is not None:
    57→        if requested not in rule_set_timeframes:
    58→            raise ValidationAppError(
    59→                f"Timeframe '{requested}' is not declared by the selected "
    60→                f"strategy version (declared: {', '.join(rule_set_timeframes)}).",
    61→                code="TIMEFRAME_NOT_DECLARED",
    62→            )
    63→        return requested
    64→    return min(rule_set_timeframes, key=timeframe_seconds)

    66→
    67→def _walk_budget(
    68→    *,
    69→    combinations: int,
    70→    validation_mode: str,
    71→    windows: int,
    72→    cost_variants: int,
    73→    has_neighbors: bool,
    74→    param_count: int,
    75→    bootstrap_phases: int,
    76→) -> int:
    77→    """Upper bound on engine walks for the configuration (creation check)."""
    78→
    79→    if validation_mode == "walk_forward":
    80→        per_candidate = windows
    81→        baseline = windows
    82→        variants = windows * cost_variants
    83→    elif validation_mode == "holdout":
    84→        per_candidate = 1
    85→        baseline = 2  # IS walk + OOS walk of the winner
    86→        variants = 2 * cost_variants
    87→    else:
    88→        per_candidate = 1
    89→        baseline = 1
    90→        variants = cost_variants
    91→    # Neighbour bound: ±1 step per dimension (patched baseline) or every
    92→    # non-current value per dimension (original baseline) — take the max.
    93→    neighbours = (2 * param_count) if has_neighbors else 0
    94→    return combinations * per_candidate + baseline + variants + neighbours + bootstrap_phases

    97→async def validate_and_create(
    98→    session: AsyncSession,
    99→    owner_id: str,
   100→    settings: Settings,
   101→    request: ResearchCreateRequest,
   102→) -> ResearchRun:
   103→    """Validate everything, then persist the run as ``queued``."""
   104→
   105→    version_row = await strategy_service.get_version(
   106→        session, owner_id, request.strategy_id, request.strategy_version
   107→    )
   108→    strategy = await strategy_service.get_strategy(
   109→        session, owner_id, request.strategy_id
   110→    )
   111→    if version_row.status == "draft":
   112→        raise ValidationAppError(
   113→            f"Version {request.strategy_version} is a draft. Publish it (or "
   114→            "select a published/superseded version) before researching — "
   115→            "drafts can still be replaced, so their lifecycle is not "
   116→            "immutable.",
   117→            code="DRAFT_VERSION_NOT_RESEARCHABLE",
   118→        )
   119→    if strategy.mode != request.mode:
   120→        raise ValidationAppError(
   121→            f"Strategy mode '{strategy.mode}' does not match the requested "
   122→            f"mode '{request.mode}'.",
   123→            code="MODE_MISMATCH",
   124→        )
   125→    try:
   126→        rule_set = validate_rule_set(version_row.rule_set)
   127→    except Exception as exc:
   128→        raise ValidationAppError(
   129→            "The stored rule set failed revalidation and cannot be researched.",
   130→            code="RULE_SET_INVALID",
   131→        ) from exc
   132→    if rule_set.mode != request.mode:
   133→        raise ValidationAppError(
   134→            f"Rule set mode '{rule_set.mode}' does not match the requested "
   135→            f"mode '{request.mode}'.",
   136→            code="MODE_MISMATCH",
   137→        )
   138→    symbol = request.symbol.strip().upper()
   139→    if rule_set.symbols and symbol not in rule_set.symbols:
   140→        raise ValidationAppError(
   141→            f"Symbol '{symbol}' is outside the strategy's symbol scope "
   142→            f"({', '.join(rule_set.symbols)}).",
   143→            code="SYMBOL_NOT_IN_SCOPE",
   144→        )
   145→    timeframe = _resolve_timeframe(request.timeframe, rule_set.timeframes)
   146→
   147→    if request.mode == "forex" and request.binary is not None:
   148→        raise ValidationAppError(
   149→            "binary configuration cannot be combined with a Forex run.",
   150→            code="CONFIG_MISMATCH",
   151→        )
   152→    if request.mode == "binary_options" and request.forex_execution is not None:
   153→        raise ValidationAppError(
   154→            "forex_execution configuration cannot be combined with a Binary "
   155→            "Options run.",
   156→            code="CONFIG_MISMATCH",
   157→        )
   158→    if request.mode == "forex" and request.binary_cost_variants:
   159→        raise ValidationAppError(
   160→            "binary_cost_variants cannot be combined with a Forex run.",
   161→            code="CONFIG_MISMATCH",
   162→        )
   163→    if request.mode == "binary_options" and request.forex_cost_variants:
   164→        raise ValidationAppError(
   165→            "forex_cost_variants cannot be combined with a Binary Options run.",
   166→            code="CONFIG_MISMATCH",
   167→        )
   168→    if request.mode == "forex" and request.forex_cost_variants and not (
   169→        request.forex_execution and request.forex_execution.enabled
   170→    ):
   171→        raise ValidationAppError(
   172→            "forex_cost_variants require an enabled forex_execution "
   173→            "configuration (cost sensitivity is meaningless for a "
   174→            "setup-only run).",
   175→            code="COST_VARIANTS_REQUIRE_EXECUTION_CONFIG",
   176→        )
   177→
   178→    # ---- search space bounds ------------------------------------------------
   179→    from app.domains.research.params import parameter_inventory
   180→
   181→    snapshot = rule_set.to_dict()
   182→    validate_patch_specs(
   183→        snapshot, [p.model_dump() for p in request.parameters]
   184→    )
   185→    combos = combination_count([p.model_dump() for p in request.parameters])
   186→    if len(request.parameters) > settings.research_max_parameters:
   187→        raise ValidationAppError(
   188→            f"The run selects {len(request.parameters)} parameters; the "
   189→            f"platform limit is {settings.research_max_parameters}.",
   190→            code="TOO_MANY_PARAMETERS",
   191→        )
   192→    for p in request.parameters:
   193→        if len(p.values) > settings.research_max_values_per_parameter:
   194→            raise ValidationAppError(
   195→                f"Parameter {p.condition_id}.{p.field} lists "
   196→                f"{len(p.values)} values; the limit is "
   197→                f"{settings.research_max_values_per_parameter}.",
   198→                code="TOO_MANY_VALUES",
   199→            )
   200→    if combos > settings.research_max_combinations:
   201→        raise ValidationAppError(
   202→            f"The selected parameters produce {combos} combinations; the "
   203→            f"platform limit is {settings.research_max_combinations}. "
   204→            "Narrow the value lists.",
   205→            code="TOO_MANY_COMBINATIONS",
   206→        )
   207→
   208→    # ---- objective applicability -------------------------------------------
   209→    simulation_mode = "setup_only"
   210→    if request.mode == "forex":
   211→        if request.forex_execution is not None and request.forex_execution.enabled:
   212→            simulation_mode = "trade_simulation"
   213→    else:
   214→        simulation_mode = "trade_simulation"
   215→    if simulation_mode == "setup_only" and request.objective.metric in TRADE_LEVEL_METRICS:
   216→        raise ValidationAppError(
   217→            f"Objective '{request.objective.metric}' requires simulated "
   218→            "trades. Provide an enabled forex_execution configuration, or "
   219→            "choose the 'conditions_met_count' objective for a setup-only "
   220→            "research run.",
   221→            code="OBJECTIVE_REQUIRES_TRADE_SIMULATION",
   222→        )
   223→
   224→    # ---- period split / windows ---------------------------------------------
   225→    start = request.start.astimezone(timezone.utc)
   226→    end = request.end.astimezone(timezone.utc)
   227→    interval = timeframe_seconds(timeframe)
   228→    start_epoch = int(start.timestamp())
   229→    end_epoch = int(end.timestamp())
   230→    in_sample_end: str | None = None
   231→    windows_count = 0
   232→    wf_warnings: list[str] = []
   233→    if request.validation_mode == "holdout":
   234→        holdout_span = end_epoch - start_epoch
   235→        is_epoch = start_epoch + int(holdout_span * request.holdout.in_sample_pct)
   236→        # Align the boundary to the primary timeframe grid so the split is
   237→        # expressible in whole candles.
   238→        is_epoch -= is_epoch % interval
   239→        if not (start_epoch < is_epoch < end_epoch - interval):
   240→            raise ValidationAppError(
   241→                "The in-sample/out-of-sample split is degenerate for this "
   242→                "period; extend the date range.",
   243→                code="INVALID_SPLIT",
   244→            )
   245→        in_sample_end = datetime.fromtimestamp(is_epoch, tz=timezone.utc).isoformat(
   246→            timespec="milliseconds"
   247→        )
   248→    elif request.validation_mode == "walk_forward":
   249→        windows, wf_warnings = generate_windows(
   250→            start_epoch=start_epoch,
   251→            end_epoch=end_epoch,
   252→            interval_seconds=interval,
   253→            train_candles=request.walk_forward.train_candles,
   254→            test_candles=request.walk_forward.test_candles,
   255→            step_candles=request.walk_forward.step_candles,
   256→            window_mode=request.walk_forward.window_mode,
   257→            max_windows=settings.research_max_windows,
   258→        )
   259→        windows_count = len(windows)
   260→        if windows_count < 2:
   261→            raise ValidationAppError(
   262→                f"The configured period holds only {windows_count} "
   263→                "walk-forward window(s); at least 2 are required for "
   264→                "meaningful walk-forward validation. Extend the date range "
   265→                "or shrink the windows.",
   266→                code="INSUFFICIENT_HISTORY_FOR_WINDOWS",
   267→            )
   268→
   269→    # ---- walk budget ---------------------------------------------------------
   270→    cost_variants = len(request.forex_cost_variants) + len(request.binary_cost_variants)
   271→    bootstrap_phases = 0
   272→    if request.run_bootstrap and settings.research_bootstrap_samples > 0:
   273→        bootstrap_phases = 2 if request.validation_mode == "holdout" else 1
   274→    budget = _walk_budget(
   275→        combinations=combos + 1,  # + candidate 0 (original parameterization)
   276→        validation_mode=request.validation_mode,
   277→        windows=windows_count,
   278→        cost_variants=cost_variants,
   279→        has_neighbors=request.param_sensitivity and request.validation_mode != "walk_forward",
   280→        param_count=len(request.parameters),
   281→        bootstrap_phases=bootstrap_phases,
   282→    )
   283→    if budget > settings.research_max_walks:
   284→        raise ValidationAppError(
   285→            f"This configuration needs up to {budget} engine walks (grid + "
   286→            f"windows + robustness scenarios), above the platform limit of "
   287→            f"{settings.research_max_walks}. Reduce combinations, windows "
   288→            "or cost variants.",
   289→            code="RESEARCH_WALK_BUDGET_EXCEEDED",
   290→        )
   291→
   292→    # ---- data coverage (pre-run, honest) --------------------------------------
   293→    coverage = await verify_coverage(
   294→        session,
   295→        symbol=symbol,
   296→        primary_timeframe=timeframe,
   297→        start=start,
   298→        end=end,
   299→        max_candles=settings.backtest_max_candles,
   300→    )
   301→
   302→    warnings: list[dict[str, Any]] = []
   303→    if request.objective.metric == "win_rate":
   304→        warnings.append(
   305→            {
   306→                "code": "WIN_RATE_OBJECTIVE",
   307→                "message": OBJECTIVE_NOTES["win_rate"],
   308→            }
   309→        )
   310→    if request.objective.metric == "conditions_met_count":
   311→        warnings.append(
   312→            {
   313→                "code": "FREQUENCY_OBJECTIVE",
   314→                "message": OBJECTIVE_NOTES["conditions_met_count"],
   315→            }
   316→        )
   317→    for message in wf_warnings:
   318→        warnings.append({"code": message.split(":")[0], "message": message})
   319→
   320→    parameter_labels: dict[str, str] = {}
   321→    for p in request.parameters:
   322→        key = f"{p.condition_id}.{p.field}"
   323→        parameter_labels[key] = p.label or key
   324→
   325→    config_payload = ResearchConfigPayload(
   326→        mode=request.mode,
   327→        symbol=symbol,
   328→        timeframe=timeframe,
   329→        validation_mode=request.validation_mode,
   330→        start=start.isoformat(timespec="milliseconds"),
   331→        end=end.isoformat(timespec="milliseconds"),
   332→        in_sample_end=in_sample_end,
   333→        data_source="postgres_candle_cache",
   334→        warmup_candles=settings.backtest_warmup_candles,
   335→        max_candles=settings.backtest_max_candles,
   336→        strategy_id=str(request.strategy_id),
   337→        strategy_name=strategy.name,
   338→        strategy_version=version_row.version,
   339→        strategy_version_status=version_row.status,
   340→        rule_set=snapshot,
   341→        direction=rule_set.direction,
   342→        parameters=[p.model_dump() for p in request.parameters],
   343→        parameter_labels=parameter_labels,
   344→        objective={"metric": request.objective.metric, "note": OBJECTIVE_NOTES[request.objective.metric]},
   345→        objective_direction=request.objective.direction,
   346→        objective_note=OBJECTIVE_NOTES[request.objective.metric],
   347→        objective_constraints=request.objective.constraints.model_dump(),
   348→        holdout=(
   349→            request.holdout.model_dump() if request.validation_mode == "holdout" else None
   350→        ),
   351→        walk_forward=(
   352→            request.walk_forward.model_dump()
   353→            if request.validation_mode == "walk_forward"
   354→            else None
   355→        ),
   356→        forex_execution=(
   357→            request.forex_execution.model_dump() if request.forex_execution else None
   358→        ),
   359→        binary=request.binary.model_dump() if request.binary else None,
   360→        forex_cost_variants=[v.model_dump() for v in request.forex_cost_variants],
   361→        binary_cost_variants=[v.model_dump() for v in request.binary_cost_variants],
   362→        param_sensitivity=request.param_sensitivity,
   363→        run_bootstrap=request.run_bootstrap,
   364→        seed=request.seed,
   365→        combination_count=combos,
   366→        walk_budget=budget,
   367→        limits={
   368→            "max_combinations": settings.research_max_combinations,
   369→            "max_walks": settings.research_max_walks,
   370→            "max_windows": settings.research_max_windows,
   371→            "bootstrap_samples": settings.research_bootstrap_samples,
   372→            "max_cost_scenarios": settings.research_max_cost_scenarios,
   373→        },
   374→        simulation_mode=simulation_mode,
   375→        notes=request.notes,
   376→        warnings=warnings,
   377→        software={
   378→            "research_runner_version": RESEARCH_RUNNER_VERSION,
   379→            "engine_version": BACKTEST_ENGINE_VERSION,
   380→            "python": platform.python_version(),
   381→        },
   382→    )
   383→
   384→    if cost_variants > settings.research_max_cost_scenarios:
   385→        raise ValidationAppError(
   386→            f"{cost_variants} cost variants were selected; the platform "
   387→            f"limit is {settings.research_max_cost_scenarios}.",
   388→            code="TOO_MANY_COST_VARIANTS",
   389→        )
   390→
   391→    run = ResearchRun(
   392→        owner_id=owner_id,
   393→        mode=request.mode,
   394→        symbol=symbol,
   395→        timeframe=timeframe,
   396→        validation_mode=request.validation_mode,
   397→        strategy_id=request.strategy_id,
   398→        strategy_name=strategy.name,
   399→        strategy_version=version_row.version,
   400→        strategy_version_id=version_row.id,
   401→        status="queued",
   402→        start_time=start,
   403→        end_time=end,
   404→        config=config_payload.model_dump(),
   405→        research_schema_version="1.0",
   406→        research_version=RESEARCH_RUNNER_VERSION,
   407→        engine_version=BACKTEST_ENGINE_VERSION,
   408→        data_quality={"pre_run_coverage": coverage},
   409→    )
   410→    session.add(run)
   411→    await session.flush()
   412→    return run

   420→async def mark_running(session: AsyncSession, run_id: uuid.UUID) -> ResearchRun:
   421→    run = await session.get(ResearchRun, run_id)
   422→    if run is None:
   423→        raise NotFoundError(f"Research run '{run_id}' was not found.")
   424→    if run.status != "queued":
   425→        raise ConflictError(
   426→            f"Run is already '{run.status}'; it cannot be started again.",
   427→            code="INVALID_RUN_TRANSITION",
   428→        )
   429→    run.status = "running"
   430→    run.started_at = datetime.now(tz=timezone.utc)
   431→    await session.flush()
   432→    return run
   433→
   434→
   435→async def mark_completed(session: AsyncSession, run: ResearchRun) -> None:
   436→    run.status = "completed"
   437→    run.completed_at = datetime.now(tz=timezone.utc)
   438→    await session.flush()
   439→
   440→
   441→async def mark_failed(
   442→    session: AsyncSession, run_id: uuid.UUID, error: dict[str, Any]
   443→) -> None:
   444→    run = await session.get(ResearchRun, run_id)
   445→    if run is None:  # pragma: no cover - defensive
   446→        return
   447→    run.status = "failed"
   448→    run.error = error
   449→    run.completed_at = datetime.now(tz=timezone.utc)
   450→    await session.flush()
   451→
   452→
   453→async def mark_cancelled(session: AsyncSession, run_id: uuid.UUID) -> None:
   454→    run = await session.get(ResearchRun, run_id)
   455→    if run is None:  # pragma: no cover - defensive
   456→        return
   457→    run.status = "cancelled"
   458→    run.completed_at = datetime.now(tz=timezone.utc)
   459→    run.error = {
   460→        "code": "CANCELLED",
   461→        "message": "The run was cancelled by its owner before completion.",
   462→        "details": [],
   463→    }
   464→    await session.flush()

   466→
   467→async def recover_interrupted(session: AsyncSession) -> int:
   468→    """Startup recovery: queued/running research rows from a previous
   469→    process are marked failed (in-process runs do not survive restarts)."""
   470→
   471→    stmt = select(ResearchRun).where(
   472→        ResearchRun.status.in_(("queued", "running")),
   473→        ResearchRun.owner_id.isnot(None),
   474→    )
   475→    rows = list((await session.scalars(stmt)).all())
   476→    for run in rows:
   477→        run.status = "failed"
   478→        run.error = {
   479→            "code": "RUNNER_RESTARTED",
   480→            "message": "The backend process restarted while this research "
   481→            "run was queued/running; in-process runs do not survive "
   482→            "restarts. Create a new run — the configuration below is "
   483→            "retained.",
   484→            "details": [],
   485→        }
   486→        run.completed_at = datetime.now(tz=timezone.utc)
   487→    if rows:
   488→        await session.flush()
   489→    return len(rows)

   497→async def get_run(session: AsyncSession, owner_id: str, run_id: uuid.UUID) -> ResearchRun:

   574→async def list_runs(
   575→    session: AsyncSession,
   576→    owner_id: str,
   577→    *,
   578→    mode: str | None = None,
   579→    strategy_id: uuid.UUID | None = None,
   580→    status: str | None = None,
   581→    validation_mode: str | None = None,
   582→    page: int = 1,
   583→    page_size: int = _PAGE_SIZE_DEFAULT,
   584→) -> tuple[list[dict[str, Any]], dict[str, int]]:
   585→    page = max(1, page)
   586→    page_size = min(max(1, page_size), _PAGE_SIZE_MAX)
   587→    stmt = select(ResearchRun).where(ResearchRun.owner_id == owner_id)
   588→    if mode:
   589→        stmt = stmt.where(ResearchRun.mode == mode)
   590→    if strategy_id is not None:
   591→        stmt = stmt.where(ResearchRun.strategy_id == strategy_id)
   592→    if status:
   593→        stmt = stmt.where(ResearchRun.status == status)
   594→    if validation_mode:
   595→        stmt = stmt.where(ResearchRun.validation_mode == validation_mode)
   596→    total = await session.scalar(select(func.count()).select_from(stmt.subquery()))
   597→    rows = list(
   598→        (
   599→            await session.scalars(
   600→                stmt.order_by(ResearchRun.created_at.desc(), ResearchRun.id.desc())
   601→                .offset((page - 1) * page_size)
   602→                .limit(page_size)
   603→            )
   604→        ).all()
   605→    )
   606→    total_int = int(total or 0)
   607→    pages = (total_int + page_size - 1) // page_size if page_size else 1
   608→    return (
   609→        [run_list_item(r) for r in rows],
   610→        {"page": page, "page_size": page_size, "total": total_int, "pages": max(1, pages)},
   611→    )

   661→async def list_candidates(
   662→    session: AsyncSession,
   663→    owner_id: str,
   664→    run_id: uuid.UUID,
   665→    *,
   666→    status: str | None = None,
   667→    sort: str = "rank",
   668→    page: int = 1,
   669→    page_size: int = 25,
   670→) -> tuple[list[dict[str, Any]], dict[str, int]]:
   671→    await get_run(session, owner_id, run_id)  # ownership gate
   672→    page = max(1, page)
   673→    page_size = min(max(1, page_size), 100)
   674→    stmt = select(ResearchCandidate).where(
   675→        ResearchCandidate.run_id == run_id,
   676→        ResearchCandidate.owner_id == owner_id,
   677→    )
   678→    if status:
   679→        stmt = stmt.where(ResearchCandidate.status == status)
   680→    total = await session.scalar(select(func.count()).select_from(stmt.subquery()))
   681→    if sort == "rank":
   682→        # NULLs last, then ascending rank; unranked candidates by index.
   683→        stmt = stmt.order_by(
   684→            ResearchCandidate.rank.asc().nullslast(),
   685→            ResearchCandidate.candidate_index.asc(),
   686→        )
   687→    else:
   688→        stmt = stmt.order_by(ResearchCandidate.candidate_index.asc())
   689→    rows = list(
   690→        (
   691→            await session.scalars(stmt.offset((page - 1) * page_size).limit(page_size))
   692→        ).all()
   693→    )
   694→    total_int = int(total or 0)
   695→    pages = (total_int + page_size - 1) // page_size if page_size else 1
   696→    return (
   697→        [candidate_summary(r) for r in rows],
   698→        {"page": page, "page_size": page_size, "total": total_int, "pages": max(1, pages)},
   699→    )

   702→async def get_candidate(
   703→    session: AsyncSession, owner_id: str, run_id: uuid.UUID, candidate_index: int
   704→) -> dict[str, Any]:
   705→    await get_run(session, owner_id, run_id)
   706→    stmt = (
   707→        select(ResearchCandidate)
   708→        .where(
   709→            ResearchCandidate.run_id == run_id,
   710→            ResearchCandidate.owner_id == owner_id,
   711→            ResearchCandidate.candidate_index == candidate_index,
   712→        )
   713→        .limit(1)
   714→    )
   715→    row = (await session.scalars(stmt)).first()
   716→    if row is None:
   717→        raise NotFoundError(
   718→            f"Candidate {candidate_index} was not found in run '{run_id}'.",
   719→            code="CANDIDATE_NOT_FOUND",
   720→        )
   721→    return candidate_detail(row)

   829→async def parameter_schema(
   830→    session: AsyncSession,
   831→    owner_id: str,
   832→    strategy_id: uuid.UUID,
   833→    version_number: int,
   834→) -> dict[str, Any]:
   835→    """Optimizable-parameter inventory of an immutable strategy version."""
   836→
   837→    from app.domains.research.params import parameter_inventory
   838→
   839→    version_row = await strategy_service.get_version(
   840→        session, owner_id, strategy_id, version_number
   841→    )
   842→    try:
   843→        rule_set = validate_rule_set(version_row.rule_set)
   844→    except Exception as exc:
   845→        raise ValidationAppError(
   846→            "The stored rule set failed revalidation.", code="RULE_SET_INVALID"
   847→        ) from exc
   848→    return {
   849→        "strategy_id": str(strategy_id),
   850→        "strategy_version": version_row.version,
   851→        "strategy_version_status": version_row.status,
   852→        "mode": rule_set.mode,
   853→        "direction": rule_set.direction,
   854→        "timeframes": list(rule_set.timeframes),
   855→        "parameters": parameter_inventory(rule_set),
   856→        "note": (
   857→            "Only fields the strategy version already declares are "
   858→            "optimizable; the optimizer never introduces new rules or "
   859→            "parameters."
   860→        ),
   861→    }
