     1→     1→"""Robustness scenarios (Phase 8) — built on the SAME validated
     2→     2→simulation services (no perturbed candles, no second engine).
     3→     3→
     4→     4→Scenario catalogue:
     5→     5→
     6→     6→* ``cost_variant``        — re-runs the configured evaluation with EXPLICIT
     7→     7→                            execution-cost overrides (spread/slippage/
     8→     8→                            commission/swap for Forex; payout/fees/tie
     9→     9→                            policy for Binary Options). Baseline deltas
    10→    10→                            show how sensitive results are to friction.
    11→    11→* ``param_neighbor``      — re-runs the SELECTION window for nearby
    12→    12→                            permitted parameter values (one step per
    13→    13→                            dimension) to expose cliff-like optima.
    14→    14→* ``bootstrap``           — OPTIONAL seeded i.i.d. resampling of the
    15→    15→                            recorded trade P&L list (documented limits).
    16→    16→* ``selection_stability`` — walk-forward only: how consistently the
    17→    17→                            optimizer picked the same parameters across
    18→    18→                            training windows.
    19→    19→
    20→    20→Every scenario documents what it changes and what it cannot establish.

    70→    70→def _current_value(rule_set_dict: dict[str, Any], spec: dict[str, Any]) -> Any:
    71→    71→    """The snapshot's current value of a parameter field (for labels and
    72→    72→    original-baseline neighbour construction)."""

    76→    76→    cond = conditions_index(rule_set_dict).get(spec["condition_id"])
    77→    77→    if cond is None:

    83→    83→        return (cond.get("params") or {}).get(field[len("params."):])

   100→   100→def build_scenarios(
   101→   101→    cfg: Any,
   102→   102→    *,
   103→   103→    candidate_rows: list[ResearchCandidate],
   104→   104→    specs: list[dict[str, Any]],
   105→   105→) -> list[dict[str, Any]]:
   106→   106→    """Plan the scenario list (bounded, deterministic order)."""
   107→   107→
   108→   108→    mode = cfg.validation_mode
   109→   109→    scenarios: list[dict[str, Any]] = []
   110→   110→
   111→   111→    # ---- B: execution-cost variants ---------------------------------------
   112→   112→    forex_variants = cfg.forex_cost_variants or []
   113→   113→    binary_variants = cfg.binary_cost_variants or []
   114→   114→    for i, variant in enumerate(forex_variants + binary_variants, start=1):
   115→   115→        is_forex = i <= len(forex_variants)
   116→   116→        changes = {k: v for k, v in variant.items() if v is not None}
   117→   117→        if not changes:
   118→   118→            continue
   119→   119→        scenarios.append(
   120→   120→            {
   121→   121→                "category": "cost_variant",
   122→   122→                "label": (
   123→   123→                    "Cost variant "
   124→   124→                    + f"{i}: " + ", ".join(f"{k}={v}" for k, v in changes.items())
   125→   125→                ),
   126→   126→                "assumptions": {
   127→   127→                    "kind": "execution_cost",
   128→   128→                    "market": "forex" if is_forex else "binary_options",
   129→   129→                    "overrides": changes,
   130→   130→                    "fixed": (
   131→   131→                        "SL/TP, position sizing, pip value and all strategy "
   132→   132→                        "logic stay identical to the baseline."
   133→   133→                        if is_forex
   134→   134→                        else "Expiry, stake and all strategy logic stay "
   135→   135→                        "identical to the baseline."
   136→   136→                    ),
   137→   137→                },
   138→   138→                "note": _COST_NOTE,
   139→   139→                "exec": {
   140→   140→                    "kind": "cost",
   141→   141→                    "forex_override": None if is_forex else None,
   142→   142→                    "variant": variant,
   143→   143→                },
   144→   144→            }
   145→   145→        )
   146→   146→
   147→   147→    # ---- A: parameter neighbours (not in walk-forward mode; WF gets the
   148→   148→    #      selection-stability view instead) ---------------------------------
   149→   149→    if cfg.param_sensitivity and mode in ("in_sample_only", "holdout"):
   150→   150→        baseline = _baseline_of_exec(cfg, candidate_rows)
   151→   151→        if baseline is not None:
   152→   152→            patch = list(baseline.params.get("patch") or [])
   153→   153→            neighbour_specs: list[tuple[dict[str, Any], Any]] = []
   154→   154→            if patch:
   155→   155→                for nb in neighbour_patches(patch, specs, max_neighbours=16):
   156→   156→                    changed = [
   157→   157→                        f"{e['condition_id']}.{e['field']}={e['value']!r}"
   158→   158→                        for e in nb
   159→   159→                        if e not in patch
   160→   160→                    ]
   161→   161→                    neighbour_specs.append(
   162→   162→                        (nb, "Neighbour: " + ", ".join(changed))
   163→   163→                    )
   164→   164→            else:
   165→   165→                # Baseline is the ORIGINAL parameterization: build
   166→   166→                # one-dimension variants from every allowed value that
   167→   167→                # differs from the snapshot's current value.
   168→   168→                for spec in specs:
   169→   169→                    current = _current_value(cfg.rule_set, spec)
   170→   170→                    for value in spec["values"]:
   171→   171→                        if value == current:
   172→   172→                            continue
   173→   173→                        single = [
   174→   174→                            {
   175→   175→                                "condition_id": spec["condition_id"],
   176→   176→                                "field": spec["field"],
   177→   177→                                "value": value,
   178→   178→                            }
   179→   179→                        ]
   180→   180→                        if len(neighbour_specs) >= 16:
   181→   181→                            break
   182→   182→                        neighbour_specs.append(
   183→   183→                            (
   184→   184→                                single,
   185→   185→                                f"From original: {spec['condition_id']}."
   186→   186→                                f"{spec['field']}={value!r}",
   187→   187→                            )
   188→   188→                        )
   189→   189→            for nb, label in neighbour_specs:
   190→   190→                scenarios.append(
   191→   191→                    {
   192→   192→                        "category": "param_neighbor",
   193→   193→                        "label": label,
   194→   194→                        "assumptions": {
   195→   195→                            "kind": "param_neighbour",
   196→   196→                            "baseline_candidate_index": baseline.candidate_index,
   197→   197→                            "neighbour_patch": nb,
   198→   198→                            "selection_window_only": True,
   199→   199→                        },
   200→   200→                        "note": _NEIGHBOUR_NOTE,
   201→   201→                        "exec": {"kind": "neighbour", "patch": nb},
   202→   202→                    }
   203→   203→                )
   204→   204→
   205→   205→    # ---- F: optional seeded bootstrap ---------------------------------------
   206→   206→    samples = int(cfg.limits.get("bootstrap_samples", 0))
   207→   207→    if cfg.run_bootstrap and samples > 0:
   208→   208→        scenarios.append(
   209→   209→            {
   210→   210→                "category": "bootstrap",
   211→   211→                "label": f"Bootstrap stress test ({samples} resamples, seed {cfg.seed})",
   212→   212→                "assumptions": {
   213→   213→                    "kind": "iid_bootstrap",
   214→   214→                    "samples": samples,
   215→   215→                    "seed": cfg.seed,
   216→   216→                    "input": "recorded simulated trade P&L (win/loss) of the "
   217→   217→                    "baseline candidate; no prices are generated",
   218→   218→                },
   219→   219→                "note": _BOOTSTRAP_NOTE,
   220→   220→                "exec": {"kind": "bootstrap"},
   221→   221→            }
   222→   222→        )
   223→   223→
   224→   224→    # ---- WF selection stability ---------------------------------------------
   225→   225→    if mode == "walk_forward":
   226→   226→        scenarios.append(
   227→   227→            {
   228→   228→                "category": "selection_stability",
   229→   229→                "label": "Walk-forward selection stability",
   230→   230→                "assumptions": {
   231→   231→                    "kind": "selection_stability",
   232→   232→                    "source": "per-window selected parameters of this run",
   233→   233→                },
   234→   234→                "note": _STABILITY_NOTE,
   235→   235→                "exec": {"kind": "stability"},
   236→   236→            }
   237→   237→        )
   238→   238→    return scenarios

   241→   241→def _baseline_of_exec(
   242→   242→    cfg: Any, candidate_rows: list[ResearchCandidate]
   243→   243→) -> ResearchCandidate | None:
   244→   244→    ranked = [r for r in candidate_rows if r.rank is not None]
   245→   245→    if ranked:
   246→   246→        return min(ranked, key=lambda r: r.rank)
   247→   247→    if candidate_rows:
   248→   248→        return candidate_rows[0]
   249→   249→    return None

   253→   253→    executor: Any,
   254→   254→    cfg: Any,
   255→   255→    scenario: dict[str, Any],

   257→   257→    end_epoch: int,
   258→   258→    interval: int,

   260→   260→    """Execute one planned scenario; returns {metrics, delta, status, error}."""

   262→   262→    kind = scenario["exec"]["kind"]
   263→   263→    mode = cfg.validation_mode
   264→   264→    if kind == "cost":
   265→   265→        return await _run_cost(executor, cfg, scenario, start_epoch, end_epoch, interval, mode)

   272→   272→    return {
   273→   273→        "metrics": None,
   274→   274→        "delta": None,
   275→   275→        "status": "failed",
   276→   276→        "error": {"code": "UNKNOWN_SCENARIO", "message": f"Unknown scenario kind '{kind}'."},
   277→   277→    }

   283→   283→async def _run_cost(
   284→   284→    executor: Any,
   285→   285→    cfg: Any,
   286→   286→    scenario: dict[str, Any],
   287→   287→    start_epoch: int,
   288→   288→    end_epoch: int,
   289→   289→    interval: int,
   290→   290→    mode: str,
   291→   291→) -> dict[str, Any]:
   292→   292→    variant = scenario["exec"]["variant"]
   293→   293→    forex_override = variant if cfg.mode == "forex" else None
   294→   294→    binary_override = variant if cfg.mode == "binary_options" else None
   295→   295→    baseline = _baseline_of(executor, mode)
   296→   296→    if baseline is None:
   297→   297→        return _failed("NO_BASELINE", "No baseline candidate exists for this run.")
   298→   298→
   299→   299→    if mode == "walk_forward":
   300→   300→        aggregate = await _wf_aggregate_with_overrides(
   301→   301→            executor, cfg, forex_override, binary_override
   302→   302→        )
   303→   303→        baseline_block = await _wf_baseline_aggregate(executor)
   304→   304→        return {
   305→   305→            "metrics": {"out_of_sample_aggregate": aggregate},
   306→   306→            "delta": _delta_block(baseline_block, aggregate),
   307→   307→            "status": "completed",
   308→   308→        }
   309→   309→
   310→   310→    walk_start, walk_end, oos_start = _selection_window(
   311→   311→        cfg, mode, start_epoch, end_epoch, interval
   312→   312→    )
   313→   313→    is_info = executor._eval_detached(
   314→   314→        cfg,
   315→   315→        baseline.rule_set,
   316→   316→        start_epoch=walk_start,
   317→   317→        end_epoch=walk_end,
   318→   318→        forex_override=forex_override,
   319→   319→        binary_override=binary_override,
   320→   320→    )
   321→   321→    metrics: dict[str, Any] = {"selection_window": is_info["digest"]}
   322→   322→    baseline_digest = baseline.in_sample or {}
   323→   323→    if oos_start is not None:
   324→   324→        oos_info = executor._eval_detached(
   325→   325→            cfg,
   326→   326→            baseline.rule_set,
   327→   327→            start_epoch=oos_start,
   328→   328→            end_epoch=end_epoch,
   329→   329→            forex_override=forex_override,
   330→   330→            binary_override=binary_override,
   331→   331→        )
   332→   332→        metrics["out_of_sample"] = oos_info["digest"]
   333→   333→        baseline_digest = baseline.out_of_sample or {}
   334→   334→    return {
   335→   335→        "metrics": metrics,
   336→   336→        "delta": _delta_block(baseline_digest, metrics.get("out_of_sample") or is_info["digest"]),
   337→   337→        "status": "completed",
   338→   338→    }

   344→   344→async def _run_neighbour(

   349→   349→    end_epoch: int,
   350→   350→    mode: str,

   352→   352→    baseline = _baseline_of(executor, mode)
   353→   353→    if baseline is None:

   355→   355→    nb_patch = scenario["exec"]["patch"]

   360→   360→            rule_set_dict = row.rule_set

   369→   369→        cfg, rule_set_dict, start_epoch=walk_start, end_epoch=walk_end
   370→   370→    )
   371→   371→    baseline_digest = baseline.in_sample or {}
   372→   372→    return {
   373→   373→        "metrics": {"selection_window": info["digest"]},
   374→   374→        "delta": _delta_block(baseline_digest, info["digest"]),
   375→   375→        "status": "completed",
   376→   376→    }

   382→   382→def _decided_pnls(output: Any) -> list[float]:
   383→   383→    return [
   384→   384→        float(r["pnl"])
   385→   385→        for r in getattr(output, "records", [])
   386→   386→        if r.get("kind") == "trade"
   387→   387→        and r.get("outcome") in ("win", "loss")
   388→   388→        and r.get("pnl") is not None
   389→   389→    ]

   392→   392→async def _run_bootstrap(
   393→   393→    executor: Any,
   394→   394→    cfg: Any,
   395→   395→    scenario: dict[str, Any],
   396→   396→    start_epoch: int,
   397→   397→    end_epoch: int,
   398→   398→    interval: int,
   399→   399→    mode: str,
   400→   400→) -> dict[str, Any]:
   401→   401→    samples = int(cfg.limits.get("bootstrap_samples", 0))
   402→   402→    seed = int(cfg.seed)
   403→   403→    baseline = _baseline_of(executor, mode)
   404→   404→    if baseline is None:
   405→   405→        return _failed("NO_BASELINE", "No baseline candidate exists for this run.")
   406→   406→    metrics: dict[str, Any] = {}
   407→   407→    if mode == "walk_forward":
   408→   408→        pnls = await _wf_oos_pnls(executor, cfg)
   409→   409→        metrics["out_of_sample"] = bootstrap_stats(pnls, samples=samples, seed=seed)
   410→   410→    else:
   411→   411→        walk_start, walk_end, oos_start = _selection_window(
   412→   412→            cfg, mode, start_epoch, end_epoch, executor._interval
   413→   413→        )
   414→   414→        is_out = executor._eval_detached(
   415→   415→            cfg, baseline.rule_set, start_epoch=walk_start, end_epoch=walk_end
   416→   416→        )
   417→   417→        metrics["selection_window"] = bootstrap_stats(
   418→   418→            _decided_pnls(is_out["output"]), samples=samples, seed=seed
   419→   419→        )
   420→   420→        if oos_start is not None:
   421→   421→            oos_out = executor._eval_detached(
   422→   422→                cfg, baseline.rule_set, start_epoch=oos_start, end_epoch=end_epoch
   423→   423→            )
   424→   424→            metrics["out_of_sample"] = bootstrap_stats(
   425→   425→                _decided_pnls(oos_out["output"]), samples=samples, seed=seed
   426→   426→            )
   427→   427→    return {"metrics": metrics, "delta": None, "status": "completed"}

   432→   432→
   433→   433→async def _run_stability(executor: Any, cfg: Any) -> dict[str, Any]:
   434→   434→    windows = await _windows(executor)
   435→   435→    completed = [
   436→   436→        w
   437→   437→        for w in windows
   438→   438→        if w.status == "completed" and w.selected_candidate_index is not None
   439→   439→    ]
   440→   440→    frequency: dict[str, int] = {}
   441→   441→    for w in completed:
   442→   442→        label = (w.selected_params or {}).get("label") or "unknown"
   443→   443→        frequency[label] = frequency.get(label, 0) + 1
   444→   444→    unique = len(frequency)
   445→   445→    most = max(frequency.items(), key=lambda kv: kv[1]) if frequency else None
   446→   446→    return {
   447→   447→        "metrics": {
   448→   448→            "windows_with_selection": len(completed),
   449→   449→            "windows_total": len(windows),
   450→   450→            "unique_selections": unique,
   451→   451→            "most_selected": {"label": most[0], "count": most[1]} if most else None,
   452→   452→            "frequency": frequency,
   453→   453→            "stability_ratio": (
   454→   454→                round(most[1] / len(completed), 4)
   455→   455→                if most and completed
   456→   456→                else None
   457→   457→            ),
   458→   458→        },
   459→   459→        "delta": None,
   460→   460→        "status": "completed",
   461→   461→    }

   467→   467→async def _windows(executor: Any) -> list[ResearchWindow]:
   468→   468→    result = await executor.session.scalars(
   469→   469→        select(ResearchWindow)
   470→   470→        .where(ResearchWindow.run_id == executor.run.id)
   471→   471→        .order_by(ResearchWindow.window_index.asc())
   472→   472→    )
   473→   473→    return list(result.all())

   479→   479→    forex_override: dict[str, Any] | None,
   480→   480→    binary_override: dict[str, Any] | None,
   481→   481→) -> dict[str, Any] | None:
   482→   482→    """Re-run the FULL walk-forward procedure with variant costs (no writes)."""
   483→   483→
   484→   484→    mode_cfg = cfg
   485→   485→    start_epoch = executor._start_epoch
   486→   486→    end_epoch = executor._end_epoch
   487→   487→    interval = executor._interval
   488→   488→    wf = dict(cfg.walk_forward or {})
   489→   489→    windows, _warnings = generate_windows(
   490→   490→        start_epoch=start_epoch,
   491→   491→        end_epoch=end_epoch,
   492→   492→        interval_seconds=interval,
   493→   493→        train_candles=wf["train_candles"],
   494→   494→        test_candles=wf["test_candles"],
   495→   495→        step_candles=wf["step_candles"],
   496→   496→        window_mode=wf["window_mode"],
   497→   497→        max_windows=int(cfg.limits.get("max_windows", 40)),
   498→   498→    )
   499→   499→    digests: list[dict[str, Any] | None] = []
   500→   500→    for spec in windows:
   501→   501→        train_count = executor._primary_candle_count(spec.train_start, spec.train_end)
   502→   502→        test_count = executor._primary_candle_count(spec.test_start, spec.test_end)
   503→   503→        if train_count < wf["train_candles"] or test_count == 0:
   504→   504→            digests.append(None)
   505→   505→            continue
   506→   506→        scored = []
   507→   507→        train_info: dict[int, dict[str, Any]] = {}
   508→   508→        for row in executor.candidate_rows:
   509→   509→            info = executor._eval_detached(
   510→   510→                mode_cfg,
   511→   511→                row.rule_set,
   512→   512→                start_epoch=spec.train_start,
   513→   513→                end_epoch=spec.train_end,
   514→   514→            )
   515→   515→            train_info[row.candidate_index] = info
   516→   516→            scored.append(
   517→   517→                {
   518→   518→                    "candidate_index": row.candidate_index,
   519→   519→                    "score": info["score"],
   520→   520→                    "constraint_violations": info["violations"],
   521→   521→                }
   522→   522→            )
   523→   523→        from app.domains.research.objectives import rank_scored, select_candidate as _sel
   524→   524→
   525→   525→        rank_scored(scored, cfg.objective_direction)
   526→   526→        selected = _sel(scored)
   527→   527→        if selected is None:
   528→   528→            digests.append(None)
   529→   529→            continue
   530→   530→        selected_row = executor.candidate_rows[selected["candidate_index"]]
   531→   531→        test_info = executor._eval_detached(
   532→   532→            mode_cfg,
   533→   533→            selected_row.rule_set,
   534→   534→            start_epoch=spec.test_start,
   535→   535→            end_epoch=spec.test_end,
   536→   536→            forex_override=forex_override,
   537→   537→            binary_override=binary_override,
   538→   538→        )
   539→   539→        digests.append(test_info["digest"])
   540→   540→    return aggregate_oos(digests)

   553→   553→async def _wf_oos_pnls(executor: Any, cfg: Any) -> list[float]:
   554→   554→    """Decided-trade P&Ls of the run's OWN out-of-sample test windows."""
   555→   555→
   556→   556→    windows = await _windows(executor)
   557→   557→    pnls: list[float] = []
   558→   558→    for w in windows:
   559→   559→        if w.status != "completed" or w.selected_candidate_index is None:
   560→   560→            continue
   561→   561→        # Re-walk the frozen winner on its test window to recover records.
   562→   562→        row_index = w.selected_candidate_index
   563→   563→        row = executor.candidate_rows[row_index]
   564→   564→        info = executor._eval_detached(
   565→   565→            cfg,
   566→   566→            row.rule_set,

   568→   568→            end_epoch=int(w.test_end.timestamp()),
   569→   569→        )
   570→   570→        pnls.extend(_decided_pnls(info["output"]))

   584→   584→def _delta_block(baseline_digest: dict[str, Any] | None, scenario_digest: dict[str, Any] | None) -> dict[str, Any] | None:
   585→   585→    base = _digest_trades(baseline_digest)
   586→   586→    scen = _digest_trades(scenario_digest)
   587→   587→    if not base and not scen:
   588→   588→        return None
   589→   589→    delta: dict[str, Any] = {}
   590→   590→    for key in _DELTA_KEYS:
   591→   591→        b = base.get(key)
   592→   592→        s = scen.get(key)
   593→   593→        if b is None or s is None:
   594→   594→            delta[key] = {"baseline": b, "scenario": s, "change": None}
   595→   595→        elif key in ("net_pnl", "expectancy"):
   596→   596→            delta[key] = {
   597→   597→                "baseline": b,
   598→   598→                "scenario": s,
   599→   599→                "change": round(float(s) - float(b), 8),
   600→   600→            }
   601→   601→        elif key in ("win_rate",):
   602→   602→            delta[key] = {
   603→   603→                "baseline": b,
   604→   604→                "scenario": s,
   605→   605→                "change_pp": round((float(s) - float(b)) * 100.0, 4),
   606→   606→            }
   607→   607→        else:
   608→   608→            delta[key] = {
   609→   609→                "baseline": b,
   610→   610→                "scenario": s,
   611→   611→                "change": round(float(s) - float(b), 8),
   612→   612→            }
   613→   613→    return delta

   625→   625→def stability_views(cfg: Any, executor: Any, mode: str) -> dict[str, Any]:
   626→   626→    """Derived C/D/E views computed from ALREADY-persisted digests."""
   627→   627→
   628→   628→    baseline = _baseline_of_exec(cfg, executor.candidate_rows)
   629→   629→    views: dict[str, Any] = {}
   630→   630→
   631→   631→    # C: time-period stability (from monthly breakdowns / per-window OOS).
   632→   632→    time_view: dict[str, Any] = {"note": None}
   633→   633→    if baseline is not None and mode != "walk_forward":
   634→   634→        in_digest = baseline.in_sample or {}
   635→   635→        out_digest = baseline.out_of_sample
   636→   636→        time_view = {
   637→   637→            "selection_window_by_month": (_digest_trades(in_digest).get("breakdowns") or {}).get("by_month"),
   638→   638→            "out_of_sample_by_month": (
   639→   639→                (_digest_trades(out_digest).get("breakdowns") or {}).get("by_month")
   640→   640→                if out_digest
   641→   641→                else None
   642→   642→            ),
   643→   643→            "note": (
   644→   644→                "Monthly breakdowns of the baseline candidate. Compare "
   645→   645→                "in-sample vs out-of-sample months; a strategy that only "
   646→   646→                "works in some months is period-dependent."
   647→   647→            ),
   648→   648→        }
   649→   649→    elif mode == "walk_forward":
   650→   650→        time_view = {
   651→   651→            "note": (
   652→   652→                "Chronological out-of-sample results per walk-forward "
   653→   653→                "window (see the walk-forward view for full details)."
   654→   654→            ),
   655→   655→            "see": "/walk-forward endpoint (per-window test metrics)",
   656→   656→        }
   657→   657→    views["time_periods"] = time_view
   658→   658→
   659→   659→    # D: market-regime stability (only from the engine's own classifier).
   660→   660→    regime_view: dict[str, Any] = {"note": None}
   661→   661→    if baseline is not None:
   662→   662→        in_digest = baseline.in_sample or {}
   663→   663→        out_digest = baseline.out_of_sample
   664→   664→        regime_view = {
   665→   665→            "selection_window_by_regime": (_digest_trades(in_digest).get("breakdowns") or {}).get("by_regime"),
   666→   666→            "out_of_sample_by_regime": (
   667→   667→                (_digest_trades(out_digest).get("breakdowns") or {}).get("by_regime")
   668→   668→                if out_digest
   669→   669→                else None
   670→   670→            ),
   671→   671→            "note": (
   672→   672→                "Regime labels come from the platform's deterministic "
   673→   673→                "classifier over the same stored candles. Regime buckets "
   674→   674→                "with very few trades cannot support conclusions."
   675→   675→            ),
   676→   676→        }
   677→   677→    views["market_regimes"] = regime_view
   678→   678→
   679→   679→    # E: data-quality sensitivity.
   680→   680→    dq = executor.data_quality or {}
   681→   681→    warning_codes: list[str] = []
   682→   682→    for tf in dq.get("timeframes", []) or []:
   683→   683→        for w in tf.get("warnings", []) or []:
   684→   684→            code = w.get("code")
   685→   685→            if code and code not in warning_codes:
   686→   686→                warning_codes.append(code)
   687→   687→    views["data_quality"] = {
   688→   688→        "overall_status": dq.get("overall_status"),
   689→   689→        "warning_codes": warning_codes,
   690→   690→        "note": (
   691→   691→            "Results from periods containing data gaps inherit that "
   692→   692→            "uncertainty; the platform never fills gaps. Treat metrics "
   693→   693→            "from low-coverage periods with suspicion."
   694→   694→            if warning_codes
   695→   695→            else "No data-quality warnings were raised for the loaded "
   696→   696→            "history used by this run."
   697→   697→        ),
   698→   698→    }
   699→   699→    return views
