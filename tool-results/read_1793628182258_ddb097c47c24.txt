     1→     1→"""In-process research runner (asyncio) — Phase 8.
     2→     2→
     3→     3→Mirrors the Phase 6 ``BacktestRunner`` architecture on purpose: a
     4→     4→single-process FastAPI app, runs executed as ``asyncio.Task`` objects
     5→     5→guarded by a semaphore, COOPERATIVE cancellation checked at every
     6→     6→engine step, one private DB session per execution, and honest startup
     7→     7→recovery (interrupted runs are marked failed, never faked completed).
     8→     8→No external task queue is introduced.
     9→     9→
    10→    10→Leakage control (enforced here, tested in the suite):
    11→    11→
    12→    12→* Candles are loaded ONCE for the whole configured range (plus warmup)
    13→    13→  and every candidate/window reuses the exact same series — candidates
    14→    14→  can never see different data.
    15→    15→* Each walk is executed against a series CLIPPED at the walk's phase
    16→    16→  end: candles closing after the phase end are removed, so neither the
    17→    17→  evaluator nor the exit resolver can touch future data. Trades that
    18→    18→  cannot settle inside their phase become 'open'/'unresolved' and are
    19→    19→  excluded from decided-trade statistics (documented purge policy).
    20→    20→* Walk-forward candidate selection consumes ONLY the training-window
    21→    21→  digests; the test window is walked once with the FROZEN winner.
    22→    22→* The out-of-sample period of a holdout run is walked exactly once,
    23→    23→  for the candidate selected on in-sample data.
    24→    24→"""

    92→    92→def clip_series(series: _TimeframeSeries, until_epoch: int) -> _TimeframeSeries:
    93→    93→    """Series restricted to candles that CLOSE at or before ``until_epoch``.
    94→    94→
    95→    95→    This is the purge boundary: an exit resolver working on this series
    96→    96→    physically cannot consult data beyond the phase end.
    97→    97→    """
    98→    98→
    99→    99→    rows = [
   100→   100→        r
   101→   101→        for r in series.candles
   102→   102→        if int(r["time"]) + series.interval <= until_epoch
   103→   103→    ]
   104→   104→    return _TimeframeSeries(series.timeframe, rows)

   107→   107→class ResearchRunner:
   108→   108→    """Owns research execution tasks + cancel events (in-process)."""
   109→   109→
   110→   110→    def __init__(self, max_concurrent: int = 1) -> None:
   111→   111→        self._semaphore = asyncio.Semaphore(max(1, max_concurrent))
   112→   112→        self._tasks: dict[uuid.UUID, asyncio.Task] = {}
   113→   113→        self._cancel_events: dict[uuid.UUID, asyncio.Event] = {}
   114→   114→
   115→   115→    # ---- submission / cancellation ----------------------------------------
   116→   116→
   117→   117→    def submit(self, run_id: uuid.UUID) -> None:
   118→   118→        if run_id in self._tasks:
   119→   119→            return
   120→   120→        event = asyncio.Event()
   121→   121→        self._cancel_events[run_id] = event
   122→   122→        task = asyncio.create_task(self._execute(run_id, event))
   123→   123→
   124→   124→        def _done(t: asyncio.Task) -> None:
   125→   125→            self._cleanup(run_id)
   126→   126→            # A task that dies OUTSIDE its internal try (e.g. loop-binding
   127→   127→            # or session-factory errors) would leave the row queued forever
   128→   128→            # — log it loudly; startup recovery marks such rows failed.
   129→   129→            if not t.cancelled() and t.exception() is not None:
   130→   130→                logger.error(
   131→   131→                    "research_task_crashed",
   132→   132→                    extra={
   133→   133→                        "run_id": str(run_id),
   134→   134→                        "error": f"{type(t.exception()).__name__}: {t.exception()}",
   135→   135→                    },
   136→   136→                )
   137→   137→
   138→   138→        task.add_done_callback(_done)
   139→   139→
   140→   140→    def cancel(self, run_id: uuid.UUID) -> bool:
   141→   141→        event = self._cancel_events.get(run_id)
   142→   142→        if event is None:
   143→   143→            return False
   144→   144→        event.set()
   145→   145→        return True
   146→   146→
   147→   147→    def is_tracked(self, run_id: uuid.UUID) -> bool:
   148→   148→        return run_id in self._tasks
   149→   149→
   150→   150→    def _cleanup(self, run_id: uuid.UUID) -> None:
   151→   151→        self._tasks.pop(run_id, None)
   152→   152→        self._cancel_events.pop(run_id, None)
   153→   153→
   154→   154→    # ---- execution ---------------------------------------------------------
   155→   155→
   156→   156→    async def _execute(self, run_id: uuid.UUID, cancel_event: asyncio.Event) -> None:
   157→   157→        from app.core.config import get_settings
   158→   158→        from app.domains.research import service as research_service
   159→   159→
   160→   160→        settings = get_settings()
   161→   161→        session_factory = get_session_factory()
   162→   162→        async with self._semaphore:
   163→   163→            async with session_factory() as session:
   164→   164→                try:
   165→   165→                    run = await research_service.mark_running(session, run_id)
   166→   166→                    await session.commit()
   167→   167→                    executor = _RunExecutor(session, settings, run, cancel_event)
   168→   168→                    await executor.execute()
   169→   169→                    await session.commit()
   170→   170→                    logger.info(
   171→   171→                        "research_completed",
   172→   172→                        extra={"run_id": str(run_id), "walks": executor.walks_executed},
   173→   173→                    )
   174→   174→                except CancelledError:
   175→   175→                    await session.rollback()
   176→   176→                    await research_service.mark_cancelled(session, run_id)
   177→   177→                    await session.commit()
   178→   178→                    logger.info("research_cancelled", extra={"run_id": str(run_id)})
   179→   179→                except asyncio.CancelledError:
   180→   180→                    await session.rollback()
   181→   181→                    await research_service.mark_failed(
   182→   182→                        session,
   183→   183→                        run_id,
   184→   184→                        {
   185→   185→                            "code": "RUN_CANCELLED_PROCESS",
   186→   186→                            "message": "The run was terminated because the "
   187→   187→                            "backend process is shutting down.",
   188→   188→                            "details": [],
   189→   189→                        },
   190→   190→                    )
   191→   191→                    await session.commit()
   192→   192→                    raise
   193→   193→                except Exception as exc:  # noqa: BLE001 - every failure recorded
   194→   194→                    await session.rollback()
   195→   195→                    await research_service.mark_failed(
   196→   196→                        session,
   197→   197→                        run_id,
   198→   198→                        {
   199→   199→                            "code": getattr(exc, "code", None) or "RESEARCH_FAILED",
   200→   200→                            "message": str(exc),
   201→   201→                            "details": getattr(exc, "details", []) or [],
   202→   202→                        },
   203→   203→                    )
   204→   204→                    await session.commit()
   205→   205→                    import traceback as _tb
   206→   206→
   207→   207→                    logger.warning(
   208→   208→                        "research_failed",
   209→   209→                        extra={
   210→   210→                            "run_id": str(run_id),
   211→   211→                            "error": f"{type(exc).__name__}: {exc}",
   212→   212→                            "traceback": _tb.format_exc(),
   213→   213→                        },
   214→   214→                    )

   271→   271→    def _clipped(self, until_epoch: int) -> dict[str, _TimeframeSeries]:
   272→   272→        """LRU-cached series clipped at a phase boundary (purge policy)."""
   273→   273→
   274→   274→        cached = self._clip_cache.get(until_epoch)
   275→   275→        if cached is not None:
   276→   276→            self._clip_cache.move_to_end(until_epoch)
   277→   277→            return cached
   278→   278→        clipped = {
   279→   279→            tf: clip_series(s, until_epoch) for tf, s in self.series_by_tf.items()
   280→   280→        }
   281→   281→        self._clip_cache[until_epoch] = clipped
   282→   282→        while len(self._clip_cache) > self._CLIP_CACHE_MAX:
   283→   283→            self._clip_cache.popitem(last=False)
   284→   284→        return clipped

   286→   286→    def _primary_candle_count(self, start_epoch: int, end_epoch: int) -> int:
   287→   287→        """Actual stored closed candles in [start, end) on the primary TF.
   288→   288→
   289→   289→        Used for honest window data probes WITHOUT burning engine walks.
   290→   290→        """
   291→   291→
   292→   292→        from bisect import bisect_left, bisect_right
   293→   293→
   294→   294→        cfg = self._config()
   295→   295→        series = self.series_by_tf.get(cfg.timeframe)
   296→   296→        if series is None or not series.candles:
   297→   297→            return 0
   298→   298→        open_times = [int(r["time"]) for r in series.candles]
   299→   299→        # Mirrors run_walk's window selection against the CLIPPED series:
   300→   300→        # candles are walkable when start <= open_time < end (candles
   301→   301→        # opening at `end` close after the phase boundary and are purged).
   302→   302→        lo = bisect_left(open_times, start_epoch)
   303→   303→        hi = bisect_left(open_times, end_epoch)
   304→   304→        return max(0, hi - lo)

   339→   339→    def _eval_detached(
   340→   340→        self,
   341→   341→        cfg: ResearchConfigPayload,
   342→   342→        rule_set_dict: dict[str, Any],
   343→   343→        *,
   344→   344→        start_epoch: int,
   345→   345→        end_epoch: int,
   346→   346→        forex_override: dict[str, Any] | None = None,
   347→   347→        binary_override: dict[str, Any] | None = None,
   348→   348→    ) -> dict[str, Any]:
   349→   349→        """One engine walk WITHOUT touching candidate rows.
   350→   350→
   351→   351→        Raises whatever the engine raises (incl. CancelledError); the
   352→   352→        caller decides how failures are recorded.
   353→   353→        """
   354→   354→
   355→   355→        if self.walks_executed >= self.walk_budget:
   356→   356→            raise ResearchBudgetError(
   357→   357→                "The run reached its engine-walk budget before finishing; "
   358→   358→                "this is a bounded-computation guard (the configuration "
   359→   359→                "should have been rejected at creation)."
   360→   360→            )
   361→   361→        self._check_cancel()
   362→   362→        validated = validate_rule_set(rule_set_dict)
   363→   363→        forex, binary = self._apply_execution_variant(
   364→   364→            cfg, forex_override, binary_override
   365→   365→        )
   366→   366→        config = self._walk_config(
   367→   367→            cfg,
   368→   368→            validated.to_dict(),
   369→   369→            start_epoch=start_epoch,
   370→   370→            end_epoch=end_epoch,
   371→   371→            forex_execution=forex,
   372→   372→            binary=binary,
   373→   373→        )
   374→   374→        self.walks_executed += 1
   375→   375→        output = run_walk(
   376→   376→            config=config,
   377→   377→            rule_set=validated,
   378→   378→            series_by_tf=self._clipped(end_epoch),
   379→   379→            cancel_check=self._check_cancel,
   380→   380→        )
   381→   381→        if self.data_quality is None and output.data_quality:
   382→   382→            self.data_quality = {
   383→   383→                **output.data_quality,
   384→   384→                "note": (
   385→   385→                    "Data-quality assessment of the loaded history; the "
   386→   386→                    "same stored series is shared by every candidate and "
   387→   387→                    "window of this run."
   388→   388→                ),
   389→   389→            }
   390→   390→        return {
   391→   391→            "score": metric_value(output.metrics, cfg.objective["metric"]),
   392→   392→            "violations": evaluate_constraints(
   393→   393→                output.metrics, cfg.objective_constraints
   394→   394→            ),
   395→   395→            "digest": metrics_digest(output.metrics),
   396→   396→            "output": output,
   397→   397→        }

   399→   399→    def _apply_execution_variant(
   400→   400→        self,
   401→   401→        cfg: ResearchConfigPayload,
   402→   402→        forex_override: dict[str, Any] | None,
   403→   403→        binary_override: dict[str, Any] | None,
   404→   404→    ) -> tuple[dict[str, Any] | None, dict[str, Any] | None]:
   405→   405→        forex = dict(cfg.forex_execution) if cfg.forex_execution else None
   406→   406→        binary = dict(cfg.binary) if cfg.binary else None
   407→   407→        if forex is not None and forex_override:
   408→   408→            for key, value in forex_override.items():
   409→   409→                if value is not None:
   410→   410→                    forex[key] = value
   411→   411→        if binary is not None and binary_override:
   412→   412→            for key, value in binary_override.items():
   413→   413→                if value is not None:
   414→   414→                    binary[key] = value
   415→   415→        return forex, binary

   417→   417→    async def _evaluate_candidate_on_window(
   418→   418→        self,
   419→   419→        cfg: ResearchConfigPayload,
   420→   420→        row: ResearchCandidate,
   421→   421→        *,
   422→   422→        start_epoch: int,
   423→   423→        end_epoch: int,
   424→   424→    ) -> dict[str, Any]:
   425→   425→        """Evaluate one candidate on one window; update its row honestly.
   426→   426→
   427→   427→        Failed/invalid candidates are RECORDED (status + error), never
   428→   428→        silently discarded. Cancellation always propagates.
   429→   429→        """
   430→   430→
   431→   431→        try:
   432→   432→            info = self._eval_detached(
   433→   433→                cfg, row.rule_set, start_epoch=start_epoch, end_epoch=end_epoch
   434→   434→            )
   435→   435→        except CancelledError:
   436→   436→            raise
   437→   437→        except ResearchBudgetError as exc:
   438→   438→            row.status = "failed"
   439→   439→            row.error = {"code": exc.code, "message": str(exc)}
   440→   440→            self.progress["candidates_failed"] += 1
   441→   441→            return {"score": None, "violations": [], "digest": {}}
   442→   442→        except Exception as exc:  # noqa: BLE001 - record, never discard
   443→   443→            row.status = "failed"
   444→   444→            row.error = {
   445→   445→                "code": getattr(exc, "code", None) or "CANDIDATE_FAILED",
   446→   446→                "message": f"{type(exc).__name__}: {exc}",
   447→   447→            }
   448→   448→            self.progress["candidates_failed"] += 1
   449→   449→            return {"score": None, "violations": [], "digest": {}}
   450→   450→        row.status = "evaluated"
   451→   451→        row.score = info["score"]
   452→   452→        row.constraint_violations = info["violations"] or None
   453→   453→        row.selectable = info["score"] is not None and not info["violations"]
   454→   454→        return info

   476→   476→            max_candles=cfg.max_candles,
   477→   477→        )
   478→   478→        self.series_by_tf = {
   479→   479→            tf: _TimeframeSeries(tf, rows) for tf, rows in rows_by_tf.items()
   480→   480→        }
   481→   481→
   482→   482→        start_epoch = _epoch(cfg.start)
   483→   483→        end_epoch = _epoch(cfg.end)
   484→   484→        interval = timeframe_seconds(cfg.timeframe)
   485→   485→        # Shared state for the robustness module (documented attributes).
   486→   486→        self._start_epoch = start_epoch
   487→   487→        self._end_epoch = end_epoch
   488→   488→        self._interval = interval
   489→   489→
   490→   490→        # Candidate 0 = original parameterization; 1..N = explicit grid.
   491→   491→        patches: list[list[dict[str, Any]] | None] = [None]
   492→   492→        patches.extend(enumerate_patches(cfg.parameters))
   493→   493→        self.progress["candidates_total"] = len(patches)
   494→   494→
   495→   495→        self.candidate_rows = []
   496→   496→        for index, patch in enumerate(patches):
   497→   497→            rule_set_dict = (
   498→   498→                dict(cfg.rule_set)
   499→   499→                if patch is None
   500→   500→                else params_mod.apply_patch(cfg.rule_set, patch)
   501→   501→            )
   502→   502→            row = ResearchCandidate(
   503→   503→                run_id=self.run.id,
   504→   504→                owner_id=self.run.owner_id,
   505→   505→                candidate_index=index,
   506→   506→                status="pending",
   507→   507→                params={
   508→   508→                    "patch": patch or [],
   509→   509→                    "label": params_mod.patch_label(patch or [], cfg.parameter_labels),
   510→   510→                },
   511→   511→                rule_set=rule_set_dict,
   512→   512→            )
   513→   513→            self.session.add(row)
   514→   514→            self.candidate_rows.append(row)
   515→   515→        await self.session.flush()
   516→   516→        await self._commit_progress(phase="candidates")
   517→   517→
   518→   518→        specs = cfg.parameters
   519→   519→        mode = cfg.validation_mode
   520→   520→        if mode == "walk_forward":
   521→   521→            await self._phase_walk_forward(cfg, specs, interval)
   522→   522→        elif mode == "holdout":
   523→   523→            await self._phase_holdout(cfg, specs, start_epoch, end_epoch, interval)
   524→   524→        else:
   525→   525→            await self._phase_in_sample_only(cfg, specs, start_epoch, end_epoch)
   526→   526→
   527→   527→        # ---- robustness ----------------------------------------------------
   528→   528→        if not self.cancel_event.is_set():
   529→   529→            await self._phase_robustness(cfg, specs, start_epoch, end_epoch, interval)
   530→   530→
   531→   531→        # ---- aggregate ------------------------------------------------------
   532→   532→        aggregate = await self._build_aggregate(cfg, mode)
   533→   533→        self.run.aggregate = aggregate
   534→   534→        # Keep the creation-time coverage block; phase data quality wins on
   535→   535→        # shared keys (it reflects the actual walks).
   536→   536→        self.run.data_quality = {
   537→   537→            **(self.data_quality or {}),
   538→   538→            **(self.run.data_quality or {}),
   539→   539→        }
   540→   540→        if self.warnings:
   541→   541→            merged = list(cfg.warnings) + list(self.warnings)
   542→   542→            self.run.config = {**cfg.model_dump(), "warnings": merged}
   543→   543→        await self._commit_progress(phase="completed")
   544→   544→        await research_service.mark_completed(self.session, self.run)

   548→   548→    async def _phase_in_sample_only(
   549→   549→        self,
   550→   550→        cfg: ResearchConfigPayload,
   551→   551→        specs: list[dict[str, Any]],
   552→   552→        start_epoch: int,
   553→   553→        end_epoch: int,
   554→   554→    ) -> None:
   555→   555→        """Every candidate evaluated on the FULL window; ranking is
   556→   556→        explicitly an in-sample fit (no out-of-sample claims)."""
   557→   557→
   558→   558→        self.warnings.append(
   559→   559→            {
   560→   560→                "code": "IN_SAMPLE_ONLY",
   561→   561→                "message": "No out-of-sample evaluation was configured: "
   562→   562→                "every ranking in this run is an in-sample fit and must "
   563→   563→                "not be read as expected performance.",
   564→   564→            }
   565→   565→        )
   566→   566→        for row in self.candidate_rows:
   567→   567→            self._check_cancel()
   568→   568→            info = await self._evaluate_candidate_on_window(
   569→   569→                cfg, row, start_epoch=start_epoch, end_epoch=end_epoch
   570→   570→            )
   571→   571→            row.in_sample = info.get("digest") or None
   572→   572→            if row.status == "evaluated":
   573→   573→                self.progress["candidates_evaluated"] += 1
   574→   574→            await self._commit_progress(phase="candidates")
   575→   575→
   576→   576→        scored = [self._candidate_score_view(r) for r in self.candidate_rows]
   577→   577→        rank_scored(scored, cfg.objective_direction)
   578→   578→        self._apply_ranks(scored)
   579→   579→        await self.session.flush()
   580→   580→        await self._commit_progress(phase="candidates_ranked")

   582→   582→    async def _phase_holdout(
   583→   583→        self,
   584→   584→        cfg: ResearchConfigPayload,
   585→   585→        specs: list[dict[str, Any]],
   586→   586→        start_epoch: int,
   587→   587→        end_epoch: int,
   588→   588→        interval: int,
   589→   589→    ) -> None:
   590→   590→        """Selection on the in-sample window; OOS walked ONCE for the
   591→   591→        in-sample winner with FROZEN parameters."""
   592→   592→
   593→   593→        is_end = _epoch(str(cfg.in_sample_end))
   594→   594→        # Documented boundary exclusion: the OOS walk starts one primary
   595→   595→        # interval after the in-sample end, so no candle is evaluated in
   596→   596→        # both periods.
   597→   597→        oos_start = is_end + interval
   598→   598→        for row in self.candidate_rows:
   599→   599→            self._check_cancel()
   600→   600→            info = await self._evaluate_candidate_on_window(
   601→   601→                cfg, row, start_epoch=start_epoch, end_epoch=is_end
   602→   602→            )
   603→   603→            row.in_sample = info.get("digest") or None
   604→   604→            if row.status == "evaluated":
   605→   605→                self.progress["candidates_evaluated"] += 1
   606→   606→            await self._commit_progress(phase="candidates_in_sample")
   607→   607→
   608→   608→        scored = [self._candidate_score_view(r) for r in self.candidate_rows]
   609→   609→        rank_scored(scored, cfg.objective_direction)
   610→   610→        self._apply_ranks(scored)
   611→   611→        await self.session.flush()
   612→   612→
   613→   613→        selected = select_candidate(scored)
   614→   614→        if selected is None:
   615→   615→            self.warnings.append(
   616→   616→                {
   617→   617→                    "code": "NO_SELECTABLE_CANDIDATE",
   618→   618→                    "message": "No candidate produced a rankable objective "
   619→   619→                    "value within the configured constraints on the "
   620→   620→                    "in-sample window; the out-of-sample phase was not "
   621→   621→                    "executed.",
   622→   622→                }
   623→   623→            )
   624→   624→            await self._commit_progress(phase="candidates_ranked")
   625→   625→            return
   626→   626→        selected_row = self.candidate_rows[selected["candidate_index"]]
   627→   627→        self._check_cancel()
   628→   628→        try:
   629→   629→            oos_info = self._eval_detached(
   630→   630→                cfg,
   631→   631→                selected_row.rule_set,
   632→   632→                start_epoch=oos_start,
   633→   633→                end_epoch=end_epoch,
   634→   634→            )
   635→   635→        except CancelledError:
   636→   636→            raise
   637→   637→        except Exception as exc:  # noqa: BLE001
   638→   638→            self.warnings.append(
   639→   639→                {
   640→   640→                    "code": "OUT_OF_SAMPLE_FAILED",
   641→   641→                    "message": f"The out-of-sample walk failed: {exc}",
   642→   642→                }
   643→   643→            )
   644→   644→            await self._commit_progress(phase="out_of_sample")
   645→   645→            return
   646→   646→        selected_row.out_of_sample = oos_info.get("digest") or None
   647→   647→        await self.session.flush()
   648→   648→        await self._commit_progress(phase="out_of_sample")

   650→   650→    def _probe_window(self, spec: Any, wf: dict[str, Any]) -> str | None:
   651→   651→        """Honest data probe for a walk-forward window (no engine walk).
   652→   652→
   653→   653→        Returns a skip reason, or None when the window has enough stored
   654→   654→        history per the configured minimum requirements.
   655→   655→        """
   656→   656→
   657→   657→        train_count = self._primary_candle_count(spec.train_start, spec.train_end)
   658→   658→        if train_count < wf["train_candles"]:
   659→   659→            return (
   660→   660→                f"INSUFFICIENT_WINDOW_DATA: the training window holds "
   661→   661→                f"{train_count} closed {self._config().timeframe} candles, "
   662→   662→                f"below the configured minimum of {wf['train_candles']}."
   663→   663→            )
   664→   664→        test_count = self._primary_candle_count(spec.test_start, spec.test_end)
   665→   665→        if test_count == 0:
   666→   666→            return (
   667→   667→                "INSUFFICIENT_WINDOW_DATA: the test window holds no closed "
   668→   668→                f"{self._config().timeframe} candles at all."
   669→   669→            )
   670→   670→        return None

   672→   672→    async def _phase_walk_forward(
   673→   673→        self,
   674→   674→        cfg: ResearchConfigPayload,
   675→   675→        specs: list[dict[str, Any]],
   676→   676→        interval: int,
   677→   677→    ) -> None:
   678→   678→        """Chronological windows: select on train, freeze, test OOS."""
   679→   679→
   680→   680→        wf = dict(cfg.walk_forward or {})
   681→   681→        start_epoch = _epoch(cfg.start)
   682→   682→        end_epoch = _epoch(cfg.end)
   683→   683→        windows, warnings = generate_windows(
   684→   684→            start_epoch=start_epoch,
   685→   685→            end_epoch=end_epoch,
   686→   686→            interval_seconds=interval,
   687→   687→            train_candles=wf["train_candles"],
   688→   688→            test_candles=wf["test_candles"],
   689→   689→            step_candles=wf["step_candles"],
   690→   690→            window_mode=wf["window_mode"],
   691→   691→            max_windows=int(cfg.limits.get("max_windows", 40)),
   692→   692→        )
   693→   693→        for message in warnings:
   694→   694→            code, _, rest = message.partition(":")
   695→   695→            self.warnings.append({"code": code, "message": rest.strip() or message})
   696→   696→        self.progress["windows_total"] = len(windows)
   697→   697→
   698→   698→        wf_rows: list[ResearchWindow] = []
   699→   699→        for spec in windows:
   700→   700→            wf_rows.append(
   701→   701→                ResearchWindow(
   702→   702→                    run_id=self.run.id,
   703→   703→                    owner_id=self.run.owner_id,
   704→   704→                    window_index=spec.index,
   705→   705→                    status="completed",
   706→   706→                    train_start=datetime.fromtimestamp(spec.train_start, tz=timezone.utc),
   707→   707→                    train_end=datetime.fromtimestamp(spec.train_end, tz=timezone.utc),
   708→   708→                    test_start=datetime.fromtimestamp(spec.test_start, tz=timezone.utc),
   709→   709→                    test_end=datetime.fromtimestamp(spec.test_end, tz=timezone.utc),
   710→   710→                )
   711→   711→            )
   712→   712→        self.session.add_all(wf_rows)
   713→   713→        await self.session.flush()
   714→   714→
   715→   715→        for wf_row, spec in zip(wf_rows, windows):
   716→   716→            self._check_cancel()
   717→   717→            skip_reason = self._probe_window(spec, wf)
   718→   718→            if skip_reason is not None:
   719→   719→                wf_row.status = "skipped"
   720→   720→                wf_row.error = {
   721→   721→                    "code": "INSUFFICIENT_WINDOW_DATA",
   722→   722→                    "message": skip_reason,
   723→   723→                }
   724→   724→                wf_row.note = (
   725→   725→                    "The run continues with later windows; aggregated "
   726→   726→                    "out-of-sample results exclude skipped windows and "
   727→   727→                    "the run is marked incomplete in its aggregate."
   728→   728→                )
   729→   729→                self.progress["windows_skipped"] += 1
   730→   730→                await self._commit_progress(phase="walk_forward")
   731→   731→                continue
   732→   732→
   733→   733→            # ---- selection: TRAINING WINDOW ONLY --------------------------
   734→   734→            train_evaluations: list[dict[str, Any]] = []
   735→   735→            scored: list[dict[str, Any]] = []
   736→   736→            train_info_by_index: dict[int, dict[str, Any]] = {}
   737→   737→            for row in self.candidate_rows:
   738→   738→                self._check_cancel()
   739→   739→                try:
   740→   740→                    info = self._eval_detached(
   741→   741→                        cfg,
   742→   742→                        row.rule_set,
   743→   743→                        start_epoch=spec.train_start,
   744→   744→                        end_epoch=spec.train_end,
   745→   745→                    )
   746→   746→                except CancelledError:
   747→   747→                    raise
   748→   748→                except ResearchBudgetError as exc:
   749→   749→                    info = {"score": None, "violations": [], "digest": {}}
   750→   750→                    row.status = "failed"
   751→   751→                    row.error = {"code": exc.code, "message": str(exc)}
   752→   752→                    self.progress["candidates_failed"] += 1
   753→   753→                except Exception as exc:  # noqa: BLE001 - record, never discard
   754→   754→                    if self.cancel_event.is_set():
   755→   755→                        raise CancelledError() from exc
   756→   756→                    info = {"score": None, "violations": [], "digest": {}}
   757→   757→                    row.status = "failed"
   758→   758→                    row.error = {
   759→   759→                        "code": getattr(exc, "code", None) or "CANDIDATE_FAILED",
   760→   760→                        "message": f"{type(exc).__name__}: {exc}",
   761→   761→                    }
   762→   762→                    self.progress["candidates_failed"] += 1
   763→   763→                decided = ((info.get("digest") or {}).get("trades") or {}).get(
   764→   764→                    "decided_trades"
   765→   765→                )
   766→   766→                train_evaluations.append(
   767→   767→                    {
   768→   768→                        "candidate_index": row.candidate_index,
   769→   769→                        "label": row.params.get("label"),
   770→   770→                        "score": info["score"],
   771→   771→                        "decided_trades": decided,
   772→   772→                        "constraint_violations": info["violations"],
   773→   773→                    }
   774→   774→                )
   775→   775→                scored.append(
   776→   776→                    {
   777→   777→                        "candidate_index": row.candidate_index,
   778→   778→                        "score": info["score"],
   779→   779→                        "constraint_violations": info["violations"],
   780→   780→                    }
   781→   781→                )
   782→   782→                train_info_by_index[row.candidate_index] = info
   783→   783→                if row.status == "evaluated":
   784→   784→                    self.progress["candidates_evaluated"] += 1
   785→   785→
   786→   786→            rank_scored(scored, cfg.objective_direction)
   787→   787→            selected = select_candidate(scored)
   788→   788→            wf_row.train_evaluations = train_evaluations
   789→   789→            if selected is None:
   790→   790→                wf_row.note = (
   791→   791→                    "No candidate produced a rankable objective value "
   792→   792→                    "within the configured constraints on this training "
   793→   793→                    "window; the test window was not walked."
   794→   794→                )
   795→   795→                wf_row.status = "completed"
   796→   796→                self.progress["windows_completed"] += 1
   797→   797→                await self._commit_progress(phase="walk_forward")
   798→   798→                continue
   799→   799→
   800→   800→            selected_row = self.candidate_rows[selected["candidate_index"]]
   801→   801→            wf_row.selected_candidate_index = selected_row.candidate_index
   802→   802→            wf_row.selected_params = selected_row.params
   803→   803→            wf_row.selection_score = selected["score"]
   804→   804→
   805→   805→            # ---- testing: FROZEN winner on the held-out window -------------
   806→   806→            self._check_cancel()
   807→   807→            try:
   808→   808→                test_info = self._eval_detached(
   809→   809→                    cfg,
   810→   810→                    selected_row.rule_set,
   811→   811→                    start_epoch=spec.test_start,
   812→   812→                    end_epoch=spec.test_end,
   813→   813→                )
   814→   814→            except CancelledError:
   815→   815→                raise
   816→   816→            except ResearchBudgetError as exc:
   817→   817→                wf_row.status = "failed"
   818→   818→                wf_row.error = {"code": exc.code, "message": str(exc)}
   819→   819→                self.progress["windows_completed"] += 1
   820→   820→                await self._commit_progress(phase="walk_forward")
   821→   821→                continue
   822→   822→            except Exception as exc:  # noqa: BLE001
   823→   823→                if self.cancel_event.is_set():
   824→   824→                    raise CancelledError() from exc
   825→   825→                wf_row.status = "failed"
   826→   826→                wf_row.error = {
   827→   827→                    "code": "TEST_WALK_FAILED",
   828→   828→                    "message": f"{type(exc).__name__}: {exc}",
   829→   829→                }
   830→   830→                self.progress["windows_completed"] += 1
   831→   831→                await self._commit_progress(phase="walk_forward")
   832→   832→                continue
   833→   833→            wf_row.test_metrics = test_info.get("digest") or None
   834→   834→            selected_row.times_selected += 1
   835→   835→            wf_row.status = "completed"
   836→   836→            self.progress["windows_completed"] += 1
   837→   837→            await self._commit_progress(phase="walk_forward")
   838→   838→
   839→   839→        # Candidate-level walk-forward summary (avg TRAINING score; test
   840→   840→        # evidence lives on the windows — the candidate row keeps only
   841→   841→        # selection-window information by construction).
   842→   842→        for row in self.candidate_rows:
   843→   843→            scores: list[float] = []
   844→   844→            for wf_row in wf_rows:
   845→   845→                for entry in wf_row.train_evaluations or []:
   846→   846→                    if (
   847→   847→                        entry["candidate_index"] == row.candidate_index
   848→   848→                        and entry["score"] is not None
   849→   849→                    ):
   850→   850→                        scores.append(float(entry["score"]))
   851→   851→            if scores:
   852→   852→                row.score = round(sum(scores) / len(scores), 8)
   853→   853→                row.in_sample = {
   854→   854→                    "walk_forward_summary": {
   855→   855→                        "windows_evaluated": len(scores),
   856→   856→                        "avg_train_score": row.score,
   857→   857→                        "times_selected": row.times_selected,
   858→   858→                    }
   859→   859→                }
   860→   860→            row.rank = None
   861→   861→            row.rank_note = (
   862→   862→                "Walk-forward run: selection happens per training window "
   863→   863→                "(see the walk-forward view). rank is intentionally NULL — "
   864→   864→                "there is no single global ranking in this mode."
   865→   865→            )
   866→   866→        await self.session.flush()
   867→   867→        await self._commit_progress(phase="walk_forward")

   924→   924→    def _candidate_score_view(self, row: ResearchCandidate) -> dict[str, Any]:

   945→   945→            return None
   946→   946→        return min(ranked, key=lambda r: r.rank)

   952→   952→            self.candidate_rows,
   953→   953→            key=lambda r: (r.times_selected, -r.candidate_index),

  1056→  1056→def get_runner() -> ResearchRunner:
  1057→  1057→    """Process-wide runner, scoped to the RUNNING event loop.
  1058→  1058→
  1059→  1059→    asyncio primitives bind to the loop on first contended use; a module
  1060→  1060→    singleton reused across loops (tests create one loop per test) would
  1061→  1061→    crash with "bound to a different event loop" and silently strand
  1062→  1062→    queued runs. Recreating the runner per loop is cheap and safe: cancel
  1063→  1063→    events for finished runs are irrelevant, and startup recovery already
  1064→  1064→    handles rows stranded by restarts.
  1065→  1065→    """
  1066→  1066→
  1067→  1067→    global _runner, _runner_loop
  1068→  1068→    loop = asyncio.get_running_loop()
  1069→  1069→    if _runner is None or _runner_loop is not loop:
  1070→  1070→        from app.core.config import get_settings
  1071→  1071→
  1072→  1072→        _runner = ResearchRunner(
  1073→  1073→            max_concurrent=get_settings().research_max_concurrent
  1074→  1074→        )
  1075→  1075→        _runner_loop = loop
  1076→  1076→    return _runner
  1077→  1077→
  1078→  1078→
  1079→  1079→def reset_runner() -> None:
  1080→  1080→    """Test hook: drop the singleton (settings may have changed)."""
  1081→  1081→
  1082→  1082→    global _runner, _runner_loop
  1083→  1083→    _runner = None
  1084→  1084→    _runner_loop = None
