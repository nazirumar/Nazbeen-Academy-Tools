     3→/**
     4→ * Strategy Builder (Phase 3) — professional editor for user-defined
     5→ * strategies.
     6→ *
     7→ * Design rules:
     8→ * - Everything is persisted through the backend; nothing shown as "saved"
     9→ *   is only local state. Unsaved edits are guarded (beforeunload + in-app
    10→ *   confirmation) and never lost silently.
    11→ * - Rule types come from GET /api/v1/rule-types. Planned (non-operational)
    12→ *   types are visible but disabled and clearly badged — the UI never
    13→ *   pretends an unsupported rule works.
    14→ * - Parameter forms are rendered from the registry's params_schema, so new
    15→ *   rule types appear in the editor without frontend changes.
    16→ * - Validation errors from the backend are rendered next to the offending
    17→ *   field via their precise paths.
    18→ * - Publishing shows a preview of the COMPLETE definition first and stores
    19→ *   an immutable version; previous published versions become superseded.
    20→ */

    99→function errorText(e: unknown): string {
   100→  if (e instanceof ApiError) {
   101→    const detail = e.details.length
   102→      ? ` (${e.details.map((d) => `${d.path || ''} ${d.message ?? ''}`).join('; ')})`
   103→      : ''
   104→    return `${e.message}${detail}${e.requestId ? ` [${e.requestId}]` : ''}`
   105→  }
   106→  return e instanceof Error ? e.message : 'Unexpected error'
   107→}

   109→let conditionCounter = 0
   110→function nextConditionId(): string {
   111→  conditionCounter += 1
   112→  return `rule_${conditionCounter}`

   115→function emptyGroup(logic: 'AND' | 'OR' = 'AND'): RuleGroup {
   116→  return { logic, conditions: [], groups: [] }

   119→function defaultRuleSet(mode: PlatformMode, direction: StrategyDirection): RuleSet {
   120→  return {
   121→    schema_version: '2.0',
   122→    mode,
   123→    timeframes: mode === 'binary_options' ? ['M1'] : ['H1'],
   124→    direction,
   125→    conditions: emptyGroup('AND'),
   126→    confirmation: null,
   127→    invalidation: null,
   128→    symbols: [],
   129→    market_regime: 'any',
   130→    evaluation: { on_close_only: true },
   131→  }
   132→}

   134→/** Errors whose path belongs to a subtree (used for inline placement). */
   135→function errorsForPrefix(errors: RuleSetValidation['errors'], prefix: string) {
   136→  return errors.filter((e) => e.path === prefix || e.path.startsWith(`${prefix}.`))
   137→}

   148→export function StrategyBuilder({
   149→  strategyId,
   150→  mode,
   151→  categories,
   152→  onCategoriesChanged,
   153→  onClose,
   154→  onSaved,
   155→}: StrategyBuilderProps) {
   156→  const [loading, setLoading] = useState(true)
   157→  const [strategy, setStrategy] = useState<StrategyDetail | null>(null)
   158→  const [versions, setVersions] = useState<StrategyVersion[]>([])
   159→  const [ruleTypes, setRuleTypes] = useState<RuleTypeSpec[]>([])
   160→
   161→  // Editable state -----------------------------------------------------------
   162→  const [name, setName] = useState('')
   163→  const [description, setDescription] = useState('')
   164→  const [direction, setDirection] = useState<StrategyDirection>('bullish')
   165→  const [ruleSet, setRuleSet] = useState<RuleSet | null>(null)
   166→  const [changelog, setChangelog] = useState('')
   167→  const [errors, setErrors] = useState<RuleSetValidation['errors']>([])
   168→
   169→  // Workflow state -----------------------------------------------------------
   170→  const [busy, setBusy] = useState<string | null>(null)
   171→  const [jsonMode, setJsonMode] = useState<'visual' | 'json'>('visual')
   172→  const [jsonText, setJsonText] = useState('')
   173→  const [previewOpen, setPreviewOpen] = useState(false)
   174→  const [leaveConfirmOpen, setLeaveConfirmOpen] = useState(false)
   175→  const baselineRef = useRef<string>('')
   176→
   177→  const dirty = useMemo(() => {
   178→    if (!ruleSet) return false
   179→    return (
   180→      JSON.stringify({
   181→        name,
   182→        description,
   183→        direction,
   184→        categoryId: strategy?.category_id ?? null,
   185→        ruleSet,
   186→      }) !== baselineRef.current
   187→    )
   188→  }, [name, description, direction, ruleSet, strategy?.category_id])
   189→
   190→  // ---- unsaved-changes guard ------------------------------------------------
   191→  useEffect(() => {
   192→    if (!dirty) return
   193→    const handler = (e: BeforeUnloadEvent) => {
   194→      e.preventDefault()
   195→      e.returnValue = ''
   196→    }
   197→    window.addEventListener('beforeunload', handler)
   198→    return () => window.removeEventListener('beforeunload', handler)
   199→  }, [dirty])
   200→
   201→  const requestClose = useCallback(() => {
   202→    if (dirty) setLeaveConfirmOpen(true)
   203→    else onClose()
   204→  }, [dirty, onClose])
   205→
   206→  // ---- data loading ----------------------------------------------------------
   207→  useEffect(() => {
   208→    let cancelled = false
   209→    ;(async () => {
   210→      setLoading(true)
   211→      try {
   212→        const [detail, vers, types] = await Promise.all([
   213→          api.get<StrategyDetail>(`/api/v1/strategies/${strategyId}`),
   214→          api.get<StrategyVersion[]>(`/api/v1/strategies/${strategyId}/versions`),
   215→          api.get<{ rule_types: RuleTypeSpec[] }>('/api/v1/rule-types'),
   216→        ])
   217→        if (cancelled) return
   218→        setStrategy(detail)
   219→        setVersions(vers)
   220→        setRuleTypes(types.rule_types)
   221→        setName(detail.name)
   222→        setDescription(detail.description ?? '')
   223→        setDirection(detail.direction)
   224→        const rs: RuleSet = detail.rule_set
   225→          ? {
   226→              ...defaultRuleSet(detail.mode, detail.direction),
   227→              ...detail.rule_set,
   228→            }
   229→          : defaultRuleSet(detail.mode, detail.direction)
   230→        setRuleSet(rs)
   231→        setJsonText(JSON.stringify(rs, null, 2))
   232→        baselineRef.current = JSON.stringify({
   233→          name: detail.name,
   234→          description: detail.description ?? '',
   235→          direction: detail.direction,
   236→          categoryId: detail.category_id,
   237→          ruleSet: rs,
   238→        })
   239→      } catch (e) {
   240→        toast({ title: 'Could not load strategy', description: errorText(e), variant: 'destructive' })
   241→        onClose()
   242→      } finally {
   243→        if (!cancelled) setLoading(false)
   244→      }
   245→    })()
   246→    return () => {
   247→      cancelled = true
   248→    }
   249→     
   250→  }, [strategyId])
   251→
   252→  const typeBySlug = useMemo(() => {
   253→    const map = new Map<string, RuleTypeSpec>()
   254→    for (const t of ruleTypes) map.set(t.type, t)
   255→    return map
   256→  }, [ruleTypes])
   257→
   258→  const availableTypes = useMemo(
   259→    () => ruleTypes.filter((t) => t.available),
   260→    [ruleTypes],
   261→  )
   262→  const plannedTypes = useMemo(
   263→    () => ruleTypes.filter((t) => !t.available),
   264→    [ruleTypes],
   265→  )

   269→  function updateRuleSet(updater: (rs: RuleSet) => RuleSet) {
   270→    setRuleSet((prev) => {
   271→      if (!prev) return prev
   272→      const next = updater(structuredClone(prev))
   273→      setJsonText(JSON.stringify(next, null, 2))
   274→      return next
   275→    })
   276→  }
   277→
   278→  function setGroupAt(
   279→    root: RuleSet,
   280→    path: string,
   281→    updater: (g: RuleGroup) => RuleGroup,
   282→  ): RuleSet {
   283→    // path examples: ".conditions", ".confirmation", ".invalidation",
   284→    // ".conditions.groups[0]"
   285→    const clone = structuredClone(root)
   286→    const segments: (string | number)[] = []
   287→    const regex = /\.([a-zA-Z_]+)(?:\[(\d+)\])?/g
   288→    let m: RegExpExecArray | null
   289→    while ((m = regex.exec(path)) !== null) {
   290→      segments.push(m[1])
   291→      if (m[2] !== undefined) segments.push(Number(m[2]))
   292→    }
   293→     
   294→    let node: any = clone
   295→    for (let i = 0; i < segments.length - 1; i++) node = node[segments[i]]
   296→    const last = segments[segments.length - 1]
   297→    node[last] = updater(node[last] as RuleGroup)
   298→    return clone
   299→  }

   301→  const groupedErrors = useCallback(
   302→    (prefix: string) => errorsForPrefix(errors, prefix),
   303→    [errors],
   304→  )

   306→  async function handleValidate() {
   307→    if (!ruleSet) return
   308→    let payload: unknown = ruleSet
   309→    if (jsonMode === 'json') {
   310→      try {
   311→        payload = JSON.parse(jsonText)
   312→      } catch (e) {
   313→        toast({
   314→          title: 'JSON syntax error',
   315→          description: (e as Error).message,
   316→          variant: 'destructive',
   317→        })
   318→        return
   319→      }
   320→    }
   321→    setBusy('validate')
   322→    try {
   323→      const res = await api.post<RuleSetValidation>('/api/v1/rule-sets/validate', {
   324→        rule_set: payload,
   325→      })
   326→      if (res.valid) {
   327→        setErrors([])
   328→        toast({ title: 'Rule set is valid', description: `Schema ${res.normalized?.schema_version ?? '?'}` })
   329→        if (jsonMode === 'json' && res.normalized) {
   330→          setRuleSet(res.normalized)
   331→          setJsonText(JSON.stringify(res.normalized, null, 2))
   332→        }
   333→      } else {
   334→        setErrors(res.errors)
   335→        toast({
   336→          title: `Invalid: ${res.errors.length} error(s)`,
   337→          description: res.errors.slice(0, 3).map((er) => `${er.path}: ${er.message}`).join('\n'),
   338→          variant: 'destructive',
   339→        })
   340→      }
   341→    } catch (e) {
   342→      toast({ title: 'Validation failed', description: errorText(e), variant: 'destructive' })
   343→    } finally {
   344→      setBusy(null)
   345→    }
   346→  }

   348→  async function saveDraft(): Promise<number | null> {
   349→    if (!strategy || !ruleSet) return null
   350→    let payload: RuleSet = ruleSet
   351→    if (jsonMode === 'json') {
   352→      try {
   353→        payload = JSON.parse(jsonText)
   354→      } catch (e) {
   355→        toast({
   356→          title: 'JSON syntax error',
   357→          description: (e as Error).message,
   358→          variant: 'destructive',
   359→        })
   360→        return null
   361→      }
   362→    }
   363→    setBusy('save')
   364→    try {
   365→      // Metadata first (name/description/direction/category may have changed).
   366→      const patchRes = await api.patch<StrategyDetail>(`/api/v1/strategies/${strategy.id}`, {
   367→        name: name.trim() || strategy.name,
   368→        description: description.trim() || null,
   369→        direction,
   370→        category_id: strategy.category_id ?? null,
   371→      })
   372→      const version = await api.post<StrategyVersion>(
   373→        `/api/v1/strategies/${strategy.id}/versions`,
   374→        { rule_set: payload, changelog: changelog.trim() || null },
   375→      )
   376→      setStrategy({ ...patchRes, rule_set: version.rule_set })
   377→      setErrors([])
   378→      setChangelog('')
   379→      const vers = await api.get<StrategyVersion[]>(`/api/v1/strategies/${strategy.id}/versions`)
   380→      setVersions(vers)
   381→      baselineRef.current = JSON.stringify({
   382→        name,
   383→        description,
   384→        direction,
   385→        categoryId: patchRes.category_id,
   386→        ruleSet: payload,
   387→      })
   388→      toast({
   389→        title: `Draft v${version.version} saved`,
   390→        description: 'Previous versions remain untouched.',
   391→      })
   392→      await onSaved()
   393→      return version.version
   394→    } catch (e) {
   395→      const err = e as ApiError
   396→      if (err.details?.length) setErrors(err.details as RuleSetValidation['errors'])
   397→      toast({ title: 'Could not save draft', description: errorText(e), variant: 'destructive' })
   398→      return null
   399→    } finally {
   400→      setBusy(null)
   401→    }
   402→  }
   403→
   404→  async function handlePublish() {
   405→    if (!strategy) return
   406→    setPreviewOpen(true)
   407→  }
   408→
   409→  async function confirmPublish() {
   410→    if (!strategy || !ruleSet) return
   411→    setPreviewOpen(false)
   412→    setBusy('publish')
   413→    try {
   414→      // Publishing persists the current definition as a draft version first.
   415→      const savedVersion = dirty || strategy.current_version === 0 ? await saveDraft() : strategy.current_version
   416→      if (savedVersion === null) return
   417→      const res = await api.post<{ version: StrategyVersion; strategy: StrategyDetail; summary: string[] }>(
   418→        `/api/v1/strategies/${strategy.id}/versions/${savedVersion}/publish`,
   419→      )
   420→      setStrategy({ ...res.strategy, rule_set: res.version.rule_set, summary: res.summary })
   421→      const vers = await api.get<StrategyVersion[]>(`/api/v1/strategies/${strategy.id}/versions`)
   422→      setVersions(vers)
   423→      toast({
   424→        title: `Version ${savedVersion} published`,
   425→        description: 'The definition is now immutable and safe to evaluate later.',
   426→      })
   427→      await onSaved()
   428→    } catch (e) {
   429→      toast({ title: 'Publish failed', description: errorText(e), variant: 'destructive' })
   430→    } finally {
   431→      setBusy(null)
   432→    }
   433→  }
   434→
   435→  async function handleArchive() {
   436→    if (!strategy) return
   437→    setBusy('archive')
   438→    try {
   439→      await api.post(`/api/v1/strategies/${strategy.id}/archive`)
   440→      toast({ title: 'Strategy archived', description: name })
   441→      await onSaved()
   442→      onClose()
   443→    } catch (e) {
   444→      toast({ title: 'Could not archive', description: errorText(e), variant: 'destructive' })
   445→    } finally {
   446→      setBusy(null)
   447→    }
   448→  }
   449→
   450→  async function handleDuplicate() {
   451→    if (!strategy) return
   452→    setBusy('duplicate')
   453→    try {
   454→      await api.post(`/api/v1/strategies/${strategy.id}/duplicate`, {})
   455→      toast({ title: 'Duplicated', description: `Copy of ${name}` })
   456→      await onSaved()
   457→      onClose()
   458→    } catch (e) {
   459→      toast({ title: 'Could not duplicate', description: errorText(e), variant: 'destructive' })
   460→    } finally {
   461→      setBusy(null)
   462→    }
   463→  }
   464→
   465→  // ---- derived -----------------------------------------------------------------
   466→
   467→  const roots = categories.filter((c) => !c.parent_id && !c.is_archived)
   468→  const selectedCategory = categories.find((c) => c.id === strategy?.category_id)
   469→  const selectedRoot = selectedCategory?.parent_id
   470→    ? categories.find((c) => c.id === selectedCategory.parent_id)
   471→    : selectedCategory
   472→  const subcategories = selectedRoot?.children?.filter((c) => !c.is_archived) ?? []
   473→
   474→  if (loading || !strategy || !ruleSet) {
   475→    return (
   476→      <div className="flex items-center gap-2 py-16 text-sm text-muted-foreground" role="status">
   477→        <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> Loading strategy builder…
   478→      </div>
   479→    )
   480→  }
   481→
   482→  return (
   483→    <div className="space-y-4">
   484→      {/* Header ---------------------------------------------------------- */}
   485→      <div className="flex flex-wrap items-center justify-between gap-2">
   486→        <div className="flex min-w-0 items-center gap-2">
   487→          <Button variant="outline" size="sm" onClick={requestClose} aria-label="Back to strategy library">
   488→            <ArrowLeft className="h-4 w-4" aria-hidden /> Library
   489→          </Button>
   490→          <h2 className="truncate text-lg font-semibold">Strategy Builder</h2>
   491→          <Badge variant="secondary" className="font-mono">
   492→            v{strategy.current_version || 0}
   493→          </Badge>
   494→          <StatusBadges strategy={strategy} />
   495→        </div>
   496→        <div className="flex items-center gap-2">
   497→          <Tabs value={jsonMode} onValueChange={(v) => setJsonMode(v as 'visual' | 'json')}>
   498→            <TabsList className="h-8">
   499→              <TabsTrigger value="visual" className="gap-1 px-2 text-xs">
   500→                <Layers className="h-3.5 w-3.5" aria-hidden /> Visual
   501→              </TabsTrigger>
   502→              <TabsTrigger value="json" className="gap-1 px-2 text-xs">
   503→                <Braces className="h-3.5 w-3.5" aria-hidden /> JSON
   504→              </TabsTrigger>
   505→            </TabsList>
   506→          </Tabs>
   507→        </div>
   508→      </div>
   509→
   510→      {dirty && (
   511→        <div className="rounded-md border border-amber-500/40 bg-amber-500/5 px-3 py-2 text-xs text-amber-700 dark:text-amber-400">
   512→          You have unsaved changes. Save a draft version to keep them.
   513→        </div>
   514→      )}
   515→
   516→      {/* Identity ---------------------------------------------------------- */}
   517→      <Card>
   518→        <CardHeader className="pb-3">
   519→          <CardTitle className="text-base">Strategy</CardTitle>
   520→          <CardDescription>
   521→            {MODE_LABELS[mode]} mode · direction, organisation and documentation
   522→          </CardDescription>
   523→        </CardHeader>
   524→        <CardContent className="grid gap-4 md:grid-cols-2">
   525→          <div className="space-y-1.5">
   526→            <Label htmlFor="sb-name">Name</Label>
   527→            <Input
   528→              id="sb-name"
   529→              value={name}
   530→              onChange={(e) => setName(e.target.value)}
   531→              placeholder="e.g. M1 pullback continuation"
   532→            />
   533→          </div>
   534→          <div className="space-y-1.5">
   535→            <Label htmlFor="sb-direction">Direction</Label>
   536→            <Select
   537→              value={direction}

   540→              <SelectTrigger id="sb-direction" aria-label="Direction">
   541→                <SelectValue />
   542→              </SelectTrigger>
   543→              <SelectContent>
   544→                <SelectItem value="bullish">Bullish</SelectItem>
   545→                <SelectItem value="bearish">Bearish</SelectItem>
   546→                <SelectItem value="both">Both directions</SelectItem>
   547→              </SelectContent>
   548→            </Select>
   549→          </div>
   550→          <div className="space-y-1.5">
   551→            <Label>Category</Label>
   552→            <CategorySelect
   553→              categories={roots}
   554→              value={strategy.category_id}
   555→              mode={mode}
   556→              onChanged={async () => {
   557→                await onCategoriesChanged()
   558→              }}
   559→              onSelected={(categoryId) => {
   560→                setStrategy((prev) => (prev ? { ...prev, category_id: categoryId } : prev))
   561→              }}
   562→            />
   563→          </div>
   564→          <div className="space-y-1.5">

   571→              onChanged={onCategoriesChanged}
   572→              onSelected={(categoryId) => {

   578→            <Label htmlFor="sb-desc">Description / intent</Label>
   579→            <Textarea
   580→              id="sb-desc"

   586→          </div>
   587→        </CardContent>
   588→      </Card>

   590→      {/* Scope ------------------------------------------------------------- */}
   591→      <Card>
   592→        <CardHeader className="pb-3">

   900→function SymbolsInput({
   901→  symbols,
   902→  onChange,
   903→  errors,
   904→}: {
   905→  symbols: string[]
   906→  onChange: (symbols: string[]) => void
   907→  errors: RuleSetValidation['errors']
   908→}) {
   909→  const [draft, setDraft] = useState('')
   910→  function add() {
   911→    const value = draft.trim().toUpperCase()
   912→    if (!value || symbols.includes(value)) return
   913→    onChange([...symbols, value])
   914→    setDraft('')
   915→  }
   916→  return (
   917→    <div className="space-y-1.5">
   918→      <Label htmlFor="sb-symbols">Symbol scope (optional)</Label>
   919→      <div className="flex gap-2">
   920→        <Input
   921→          id="sb-symbols"
   922→          value={draft}
   923→          onChange={(e) => setDraft(e.target.value)}
   924→          onKeyDown={(e) => {
   925→            if (e.key === 'Enter') {
   926→              e.preventDefault()
   927→              add()
   928→            }
   929→          }}
   930→          placeholder="e.g. EURUSD (empty = any symbol)"
   931→        />
   932→        <Button size="icon" variant="outline" aria-label="Add symbol" onClick={add}>
   933→          <Plus className="h-4 w-4" aria-hidden />
   934→        </Button>
   935→      </div>
   936→      {symbols.length > 0 && (
   937→        <div className="flex flex-wrap gap-1.5">
   938→          {symbols.map((s) => (
   939→            <Badge key={s} variant="secondary" className="gap-1 font-mono">
   940→              {s}
   941→              <button
   942→                type="button"
   943→                aria-label={`Remove symbol ${s}`}
   944→                onClick={() => onChange(symbols.filter((x) => x !== s))}
   945→                className="ml-0.5 rounded-full hover:bg-muted"
   946→              >
   947→                <X className="h-3 w-3" aria-hidden />
   948→              </button>
   949→            </Badge>
   950→          ))}
   951→        </div>
   952→      )}
   953→      {errors.map((e, i) => (
   954→        <p key={i} className="text-xs text-destructive">{e.message}</p>
   955→      ))}
   956→    </div>
   957→  )

  1173→function VersionHistory({ versions }: { versions: StrategyVersion[] }) {

  1176→    s === 'published'
  1177→      ? 'border-emerald-600/40 text-emerald-700 dark:text-emerald-400'

  1232→function RuleSectionCard(props: RuleSectionProps) {

  1234→

  1236→    .replace(/^\./, '')
  1237→    .split('.')
  1238→     
  1239→    .reduce<any>((node, key) => {
  1240→      const m = key.match(/^([a-zA-Z_]+)(?:\[(\d+)\])?$/)
  1241→      if (!m || node === undefined) return node
  1242→      let next = node?.[m[1]]
  1243→      if (m[2] !== undefined) next = next?.[Number(m[2])]
  1244→      return next
  1245→    }, ruleSet) as RuleGroup | null
  1246→
  1247→  return (
  1248→    <Card>
  1249→      <CardHeader className="pb-3">
  1250→        <div className="flex flex-wrap items-center justify-between gap-2">
  1251→          <div>
  1252→            <CardTitle className="text-base">{props.title}</CardTitle>
  1253→            <CardDescription>{props.description}</CardDescription>
  1254→          </div>
  1255→          {clearable && group && (
  1256→            <Button
  1257→              size="sm"
  1258→              variant="ghost"
  1259→              onClick={() => setRuleSet((rs) => setGroupAt(rs, path, () => null as unknown as RuleGroup))}
  1260→              aria-label={`Remove ${props.title.toLowerCase()} rules`}
  1261→            >
  1262→              <Trash2 className="h-3.5 w-3.5" aria-hidden /> Remove
  1263→            </Button>
  1264→          )}
  1265→        </div>
  1266→      </CardHeader>
  1267→      <CardContent>
  1268→        {group ? (
  1269→          <GroupEditor {...props} group={group} />
  1270→        ) : (
  1271→          <Button
  1272→            size="sm"
  1273→            variant="outline"
  1274→            onClick={() =>
  1275→              setRuleSet((rs) =>
  1276→                setGroupAt(rs, path, () => emptyGroup('AND') as unknown as RuleGroup),
  1277→              )
  1278→            }
  1279→          >
  1280→            <Plus className="h-4 w-4" aria-hidden /> Add rules
  1281→          </Button>
  1282→        )}
  1283→      </CardContent>
  1284→    </Card>
  1285→  )

  1288→function GroupEditor({
  1289→  group,
  1290→  path,
  1291→  ruleSet,
  1292→  setRuleSet,
  1293→  setGroupAt,
  1294→  timeframes,
  1295→  availableTypes,
  1296→  plannedTypes,
  1297→  typeBySlug,
  1298→  errors,
  1299→  depth = 0,
  1300→}: RuleSectionProps & { group: RuleGroup; depth?: number }) {
  1301→  const prefix = path
  1302→  const groupErrors = errorsForPrefix(errors, prefix).filter((e) => e.path === prefix)
  1303→
  1304→  function patchGroup(updater: (g: RuleGroup) => RuleGroup) {
  1305→    setRuleSet((rs) => setGroupAt(rs, path, updater))
  1306→  }
  1307→
  1308→  function addCondition() {
  1309→    patchGroup((g) => ({
  1310→      ...g,
  1311→      conditions: [
  1312→        ...g.conditions,
  1313→        {
  1314→          id: nextConditionId(),
  1315→          type: availableTypes[0]?.type ?? 'price_level',
  1316→          timeframe: timeframes[0] ?? 'H1',
  1317→          lookback: availableTypes[0]?.default_lookback ?? 20,
  1318→          params: {},
  1319→        },
  1320→      ],
  1321→    }))
  1322→  }
  1323→
  1324→  return (
  1325→    <div className="space-y-3 rounded-md border p-3" data-testid={`group${path}`}>
  1326→      {/* Logic toggle */}
  1327→      <div className="flex flex-wrap items-center justify-between gap-2">
  1328→        <div
  1329→          className="flex overflow-hidden rounded-md border"
  1330→          role="group"
  1331→          aria-label="Group logic"
  1332→        >
  1333→          {(['AND', 'OR'] as const).map((logic) => (
  1334→            <button
  1335→              key={logic}
  1336→              type="button"
  1337→              aria-pressed={group.logic === logic}
  1338→              onClick={() => patchGroup((g) => ({ ...g, logic }))}
  1339→              className={`px-3 py-1 text-xs font-semibold transition-colors ${
  1340→                group.logic === logic
  1341→                  ? 'bg-primary text-primary-foreground'
  1342→                  : 'bg-background text-muted-foreground hover:bg-muted'
  1343→              }`}
  1344→            >
  1345→              {logic}
  1346→            </button>
  1347→          ))}
  1348→        </div>
  1349→        <div className="flex items-center gap-1">
  1350→          <Button size="sm" variant="outline" onClick={addCondition}>
  1351→            <Plus className="h-3.5 w-3.5" aria-hidden /> Rule
  1352→          </Button>
  1353→          {depth < 3 && (
  1354→            <Button
  1355→              size="sm"
  1356→              variant="outline"
  1357→              onClick={() => patchGroup((g) => ({ ...g, groups: [...g.groups, emptyGroup('OR')] }))}
  1358→            >
  1359→              <Plus className="h-3.5 w-3.5" aria-hidden /> Group
  1360→            </Button>
  1361→          )}
  1362→        </div>
  1363→      </div>
  1364→
  1365→      {groupErrors.map((e, i) => (
  1366→        <p key={i} className="text-xs text-destructive">{e.message}</p>
  1367→      ))}
  1368→
  1369→      {/* Conditions */}
  1370→      <div className="space-y-2">
  1371→        {group.conditions.map((cond, index) => {
  1372→          const condPath = `${prefix}.conditions[${index}]`
  1373→          return (
  1374→            <ConditionCard
  1375→              key={cond.id}
  1376→              condition={cond}
  1377→              path={condPath}
  1378→              index={index}
  1379→              total={group.conditions.length}
  1380→              timeframes={timeframes}
  1381→              availableTypes={availableTypes}
  1382→              plannedTypes={plannedTypes}
  1383→              typeBySlug={typeBySlug}
  1384→              errors={errors}
  1385→              onChange={(next) =>
  1386→                patchGroup((g) => ({
  1387→                  ...g,
  1388→                  conditions: g.conditions.map((c, i) => (i === index ? next : c)),
  1389→                }))
  1390→              }
  1391→              onRemove={() =>
  1392→                patchGroup((g) => ({
  1393→                  ...g,
  1394→                  conditions: g.conditions.filter((_, i) => i !== index),
  1395→                }))
  1396→              }
  1397→              onMove={(dir) =>
  1398→                patchGroup((g) => {
  1399→                  const next = [...g.conditions]
  1400→                  const target = index + dir
  1401→                  if (target < 0 || target >= next.length) return g
  1402→                  ;[next[index], next[target]] = [next[target], next[index]]
  1403→                  return { ...g, conditions: next }
  1404→                })
  1405→              }
  1406→            />
  1407→          )
  1408→        })}
  1409→      </div>
  1410→
  1411→      {/* Nested groups */}
  1412→      <div className="space-y-2">
  1413→        {group.groups.map((nested, i) => (
  1414→          <GroupEditor
  1415→            key={`${path}.groups[${i}]`}
  1416→            {...({
  1417→              ruleSet,
  1418→              setRuleSet,
  1419→              setGroupAt,
  1420→              timeframes,
  1421→              availableTypes,
  1422→              plannedTypes,
  1423→              typeBySlug,
  1424→              errors,
  1425→            } as RuleSectionProps)}
  1426→            group={nested}
  1427→            path={`${path}.groups[${i}]`}
  1428→            depth={depth + 1}
  1429→          />
  1430→        ))}
  1431→      </div>
  1432→    </div>
  1433→  )
  1434→}

  1436→function ConditionCard({
  1437→  condition,
  1438→  path,
  1439→  index,
  1440→  total,
  1441→  timeframes,
  1442→  availableTypes,
  1443→  plannedTypes,
  1444→  typeBySlug,
  1445→  errors,
  1446→  onChange,
  1447→  onRemove,
  1448→  onMove,
  1449→}: {
  1450→  condition: RuleCondition
  1451→  path: string
  1452→  index: number
  1453→  total: number
  1454→  timeframes: string[]
  1455→  availableTypes: RuleTypeSpec[]
  1456→  plannedTypes: RuleTypeSpec[]
  1457→  typeBySlug: Map<string, RuleTypeSpec>
  1458→  errors: RuleSetValidation['errors']
  1459→  onChange: (next: RuleCondition) => void
  1460→  onRemove: () => void
  1461→  onMove: (dir: -1 | 1) => void
  1462→}) {
  1463→  const spec = typeBySlug.get(condition.type)
  1464→  const ownErrors = errorsForPrefix(errors, path)
  1465→  const paramErrors = (key: string) =>
  1466→    ownErrors.filter((e) => e.path === `${path}.params.${key}`)
  1467→  const generalErrors = ownErrors.filter((e) => !e.path.includes('.params.'))
  1468→
  1469→  function changeType(type: string) {
  1470→    const nextSpec = typeBySlug.get(type)
  1471→    onChange({
  1472→      ...condition,
  1473→      type,
  1474→      lookback: nextSpec?.default_lookback ?? condition.lookback,
  1475→      params: {},
  1476→    })
  1477→  }
  1478→
  1479→  function setParam(key: string, raw: string, schemaType: string) {
  1480→    let value: unknown
  1481→    if (schemaType === 'integer') value = raw === '' ? undefined : parseInt(raw, 10)
  1482→    else if (schemaType === 'number') value = raw === '' ? undefined : parseFloat(raw)
  1483→    else value = raw
  1484→    const params = { ...condition.params }
  1485→    if (value === undefined || value === null || Number.isNaN(value as number)) {
  1486→      delete params[key]
  1487→    } else {
  1488→      params[key] = value
  1489→    }
  1490→    onChange({ ...condition, params })
  1491→  }
  1492→
  1493→  return (
  1494→    <div className="rounded-md border bg-background p-3" data-condition-id={condition.id}>
  1495→      <div className="mb-2 flex flex-wrap items-center justify-between gap-2">
  1496→        <div className="flex flex-wrap items-center gap-2">
  1497→          <Select value={condition.type} onValueChange={changeType}>
  1498→            <SelectTrigger className="h-8 w-56" aria-label={`Rule type of ${condition.id}`}>
  1499→              <SelectValue />
  1500→            </SelectTrigger>
  1501→            <SelectContent>
  1502→              {availableTypes.map((t) => (
  1503→                <SelectItem key={t.type} value={t.type}>
  1504→                  {t.title}
  1505→                </SelectItem>
  1506→              ))}
  1507→              <SelectSeparator />
  1508→              {plannedTypes.map((t) => (
  1509→                <SelectItem key={t.type} value={t.type} disabled>
  1510→                  {t.title} — planned
  1511→                </SelectItem>
  1512→              ))}
  1513→            </SelectContent>
  1514→          </Select>
  1515→          <Badge variant="outline" className="font-mono text-[10px]">
  1516→            {condition.id}
  1517→          </Badge>
  1518→        </div>
  1519→        <TooltipProvider delayDuration={200}>
  1520→          <div className="flex items-center gap-1">
  1521→            <Tooltip>
  1522→              <TooltipTrigger asChild>
  1523→                <Button
  1524→                  size="icon"
  1525→                  variant="ghost"
  1526→                  className="h-7 w-7"
  1527→                  aria-label="Move rule up"
  1528→                  disabled={index === 0}
  1529→                  onClick={() => onMove(-1)}
  1530→                >
  1531→                  <ChevronUp className="h-3.5 w-3.5" aria-hidden />
  1532→                </Button>
  1533→              </TooltipTrigger>
  1534→              <TooltipContent>Rule order (earlier first)</TooltipContent>
  1535→            </Tooltip>
  1536→            <Tooltip>
  1537→              <TooltipTrigger asChild>
  1538→                <Button
  1539→                  size="icon"
  1540→                  variant="ghost"
  1541→                  className="h-7 w-7"
  1542→                  aria-label="Move rule down"
  1543→                  disabled={index === total - 1}
  1544→                  onClick={() => onMove(1)}
  1545→                >
  1546→                  <ChevronDown className="h-3.5 w-3.5" aria-hidden />
  1547→                </Button>
  1548→              </TooltipTrigger>
  1549→              <TooltipContent>Rule order (later last)</TooltipContent>
  1550→            </Tooltip>
  1551→            <Tooltip>
  1552→              <TooltipTrigger asChild>
  1553→                <Button
  1554→                  size="icon"
  1555→                  variant="ghost"
  1556→                  className="h-7 w-7"
  1557→                  aria-label={`Remove rule ${condition.id}`}
  1558→                  onClick={onRemove}
  1559→                >
  1560→                  <Trash2 className="h-3.5 w-3.5 text-destructive" aria-hidden />
  1561→                </Button>
  1562→              </TooltipTrigger>
  1563→              <TooltipContent>Remove rule</TooltipContent>
  1564→            </Tooltip>
  1565→          </div>
  1566→        </TooltipProvider>
  1567→      </div>
  1568→
  1569→      <div className="grid gap-2 sm:grid-cols-[1fr_2fr]">
  1570→        <div className="space-y-2">
  1571→          <div className="space-y-1">
  1572→            <Label htmlFor={`${condition.id}-tf`} className="text-xs">Timeframe</Label>
  1573→            <Select
  1574→              value={condition.timeframe}
  1575→              onValueChange={(v) => onChange({ ...condition, timeframe: v })}
  1576→            >
  1577→              <SelectTrigger id={`${condition.id}-tf`} className="h-8" aria-label="Rule timeframe">
  1578→                <SelectValue />
  1579→              </SelectTrigger>
  1580→              <SelectContent>
  1581→                {timeframes.map((tf) => (
  1582→                  <SelectItem key={tf} value={tf}>
  1583→                    {tf}
  1584→                  </SelectItem>
  1585→                ))}
  1586→              </SelectContent>
  1587→            </Select>
  1588→            {ownErrors.some((e) => e.path === `${path}.timeframe`) && (
  1589→              <p className="text-xs text-destructive">
  1590→                {ownErrors.find((e) => e.path === `${path}.timeframe`)?.message}

  1672→      </div>
  1673→    </div>
