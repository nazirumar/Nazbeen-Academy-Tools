     1→"""Rule-type registry — the single source of truth for condition types.
     2→
     3→Every condition type that a strategy rule set may reference is registered
     4→here with:
     5→
     6→* a machine-readable ``params`` schema (enums, numeric bounds, defaults);
     7→* human-readable semantics (what "satisfied" means, which candles are used);
     8→* availability metadata. Types marked ``available=False`` are **planned**
     9→  extensions: the UI may show them, but validation rejects them with
    10→  ``rule_type_not_available`` and the evaluator never runs them.
    11→
    12→Design rules (Phase 3):
    13→
    14→1. A rule type is "supported" only when it appears here with
    15→   ``available=True`` AND has an evaluator in
    16→   ``app.domains.analysis.evaluator.CONDITION_EVALUATORS`` AND has tests.
    17→2. Adding a new rule type must not require changes to the strategy builder:
    18→   the frontend renders parameter forms from ``params_schema``.
    19→3. No rule definition can execute code — parameters are pure data
    20→   (numbers, enums, strings) validated against the schemas below.
    21→"""
    22→
    23→from __future__ import annotations
    24→

    48→@dataclass(frozen=True)
    49→class RuleTypeSpec:
    50→    """Metadata + parameter schema for one condition type."""
    51→
    52→    type: str
    53→    title: str
    54→    description: str
    55→    semantics: str
    56→    params_schema: dict[str, Any] = field(default_factory=dict)
    57→    required_params: tuple[str, ...] = ()
    58→    available: bool = True
    59→    status: str = "operational"  # operational | planned
    60→    supported_modes: tuple[str, ...] = ("forex", "binary_options")
    61→    default_lookback: int = 20
    62→    # Human note about how much history the rule ideally receives.
    63→    data_note: str = ""
    64→
    65→
    66→def _num(description: str, *, minimum: float | None = None, maximum: float | None = None) -> dict[str, Any]:
    67→    schema: dict[str, Any] = {"type": "number", "description": description}
    68→    if minimum is not None:
    69→        schema["minimum"] = minimum
    70→    if maximum is not None:
    71→        schema["maximum"] = maximum
    72→    return schema
    73→
    74→
    75→def _int(description: str, *, minimum: int | None = None, maximum: int | None = None) -> dict[str, Any]:
    76→    schema: dict[str, Any] = {"type": "integer", "description": description}
    77→    if minimum is not None:
    78→        schema["minimum"] = minimum
    79→    if maximum is not None:
    80→        schema["maximum"] = maximum
    81→    return schema
    82→
    83→
    84→def _enum(description: str, values: tuple[str, ...]) -> dict[str, Any]:
    85→    return {"type": "string", "enum": list(values), "description": description}

   419→    try:
   420→        return REGISTRY[rule_type]

   461→def _validate_param_value(
   462→    rule_type: str,
   463→    key: str,
   464→    schema: dict[str, Any],
   465→    value: Any,
   466→    path: str,
   467→    errors: list[dict[str, str]],
   468→) -> None:
   469→    expected = schema.get("type")
   470→    if expected == "number":
   471→        if isinstance(value, bool) or not isinstance(value, (int, float)):
   472→            errors.append(
   473→                {
   474→                    "path": f"{path}.{key}",
   475→                    "message": f"Parameter '{key}' must be a number.",
   476→                    "code": "invalid_param_type",
   477→                }
   478→            )
   479→            return
   480→        import math
   481→
   482→        if not math.isfinite(float(value)):
   483→            errors.append(
   484→                {
   485→                    "path": f"{path}.{key}",
   486→                    "message": f"Parameter '{key}' must be a finite number.",
   487→                    "code": "invalid_param_type",
   488→                }
   489→            )
   490→            return
   491→        minimum = schema.get("minimum")
   492→        maximum = schema.get("maximum")
   493→        if minimum is not None and float(value) < minimum:
   494→            errors.append(
   495→                {
   496→                    "path": f"{path}.{key}",
   497→                    "message": f"Parameter '{key}' must be >= {minimum}.",
   498→                    "code": "param_out_of_range",
   499→                }
   500→            )
   501→        if maximum is not None and float(value) > maximum:
   502→            errors.append(
   503→                {
   504→                    "path": f"{path}.{key}",
   505→                    "message": f"Parameter '{key}' must be <= {maximum}.",
   506→                    "code": "param_out_of_range",
   507→                }
   508→            )
   509→    elif expected == "integer":
   510→        if isinstance(value, bool) or not isinstance(value, int):
   511→            errors.append(
   512→                {
   513→                    "path": f"{path}.{key}",
   514→                    "message": f"Parameter '{key}' must be an integer.",
   515→                    "code": "invalid_param_type",
   516→                }
   517→            )
   518→            return
   519→        minimum = schema.get("minimum")
   520→        maximum = schema.get("maximum")
   521→        if minimum is not None and value < minimum:
   522→            errors.append(
   523→                {
   524→                    "path": f"{path}.{key}",
   525→                    "message": f"Parameter '{key}' must be >= {minimum}.",
   526→                    "code": "param_out_of_range",
   527→                }
   528→            )
   529→        if maximum is not None and value > maximum:
   530→            errors.append(
   531→                {
   532→                    "path": f"{path}.{key}",
   533→                    "message": f"Parameter '{key}' must be <= {maximum}.",
   534→                    "code": "param_out_of_range",
   535→                }
   536→            )
   537→    elif expected == "string":
   538→        if not isinstance(value, str):
   539→            errors.append(
   540→                {
   541→                    "path": f"{path}.{key}",
   542→                    "message": f"Parameter '{key}' must be a string.",
   543→                    "code": "invalid_param_type",
   544→                }
   545→            )
   546→            return
   547→        enum = schema.get("enum")
   548→        if enum is not None and value not in enum:
   549→            errors.append(
   550→                {
   551→                    "path": f"{path}.{key}",
   552→                    "message": (
   553→                        f"Parameter '{key}' must be one of: {', '.join(enum)}."
   554→                    ),
   555→                    "code": "invalid_param_value",
   556→                }
   557→            )

   559→
   560→def validate_params(
   561→    rule_type: str,
   562→    params: Any,
   563→    *,
   564→    path: str = ".params",
   565→) -> tuple[dict[str, Any], list[dict[str, str]]]:
   566→    """Validate the ``params`` object of a condition against the registry.
   567→
   568→    Returns ``(normalized_params, errors)``. Normalisation applies
   569→    documented defaults for optional parameters. Never raises for content
   570→    problems — errors carry precise paths so the editor can place them
   571→    next to the offending field.
   572→    """
   573→
   574→    errors: list[dict[str, str]] = []
   575→    spec = REGISTRY.get(rule_type)
   576→    if spec is None:
   577→        # Caller (rules.py) already reports the unknown/unavailable type.
   578→        return {}, errors
   579→
   580→    if params is None:
   581→        params = {}
   582→    if not isinstance(params, dict):
   583→        errors.append(
   584→            {
   585→                "path": path,
   586→                "message": "params must be a JSON object.",
   587→                "code": "invalid_param_type",
   588→            }
   589→        )
   590→        return {}, errors
   591→
   592→    unknown = sorted(set(params) - set(spec.params_schema))
   593→    for key in unknown:
   594→        errors.append(
   595→            {
   596→                "path": f"{path}.{key}",
   597→                "message": (
   598→                    f"Unknown parameter '{key}' for rule type '{rule_type}'. "
   599→                    f"Allowed: {', '.join(sorted(spec.params_schema))}."
   600→                ),
   601→                "code": "unknown_param",
   602→            }
   603→        )
   604→
   605→    normalized: dict[str, Any] = {}
   606→    for key in sorted(spec.params_schema):
   607→        if key in params:
   608→            _validate_param_value(
   609→                rule_type, key, spec.params_schema[key], params[key], path, errors
   610→            )
   611→            normalized[key] = params[key]
   612→        elif key in _PARAM_DEFAULTS.get(rule_type, {}):
   613→            normalized[key] = _PARAM_DEFAULTS[rule_type][key]
   614→        elif key in spec.required_params:
   615→            errors.append(
   616→                {
   617→                    "path": f"{path}.{key}",
   618→                    "message": (
   619→                        f"Missing required parameter '{key}' for rule type "
   620→                        f"'{rule_type}'."
   621→                    ),
   622→                    "code": "missing_param",
   623→                }
   624→            )
   625→
   626→    # ---- cross-field rules -------------------------------------------------
   627→    if rule_type == "price_zone":
   628→        low = normalized.get("zone_low")
   629→        high = normalized.get("zone_high")
   630→        if (
   631→            isinstance(low, (int, float))
   632→            and not isinstance(low, bool)
   633→            and isinstance(high, (int, float))
   634→            and not isinstance(high, bool)
   635→            and float(high) < float(low)
   636→        ):
   637→            errors.append(
   638→                {
   639→                    "path": f"{path}.zone_high",
   640→                    "message": "zone_high must be >= zone_low.",
   641→                    "code": "invalid_param_value",
   642→                }
   643→            )
   644→    elif rule_type == "moving_average":
   645→        compare_to = normalized.get("compare_to")
   646→        if compare_to == "value" and "value" not in params:
   647→            errors.append(
   648→                {
   649→                    "path": f"{path}.value",
   650→                    "message": (
   651→                        "compare_to='value' requires the numeric 'value' "
   652→                        "parameter."
   653→                    ),
   654→                    "code": "missing_param",
   655→                }
   656→            )
   657→        if compare_to == "price_field" and "price_field" not in params:
   658→            errors.append(
   659→                {
   660→                    "path": f"{path}.price_field",
   661→                    "message": (
   662→                        "compare_to='price_field' requires the 'price_field' "
   663→                        "parameter."
   664→                    ),
   665→                    "code": "missing_param",
   666→                }
   667→            )
   668→
   669→    return normalized, errors
